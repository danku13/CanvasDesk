  1237→    <script>
  1238→      window.__canvasdesk = (function () {
  1239→        function openDb() {
  1240→          return new Promise((resolve, reject) => {
  1241→            const req = indexedDB.open("canvasdesk", 1);
  1242→            req.onupgradeneeded = () => {
  1243→              const db = req.result;
  1244→              if (!db.objectStoreNames.contains("recent"))
  1245→                db.createObjectStore("recent", { keyPath: "name" });
  1246→              if (!db.objectStoreNames.contains("handles"))
  1247→                db.createObjectStore("handles");
  1248→            };
  1249→            req.onsuccess = () => resolve(req.result);
  1250→            req.onerror = () => reject(req.error);
  1251→          });
  1252→        }
  1253→        async function tx(store, mode, fn) {
  1254→          const db = await openDb();
  1255→          return new Promise((resolve, reject) => {
  1256→            const t = db.transaction(store, mode);
  1257→            const req = fn(t.objectStore(store));
  1258→            req.onsuccess = () => resolve(req.result);
  1259→            req.onerror = () => reject(req.error);
  1260→            t.oncomplete = () => db.close();
  1261→          });
  1262→        }
  1263→        return {
  1264→          recentPut: (name) =>
  1265→            tx("recent", "readwrite", (s) => s.put({ name: name, ts: Date.now() })),
  1266→          recentList: () => tx("recent", "readonly", (s) => s.getAll()),
  1267→          handlePut: (name, handle) =>
  1268→            tx("handles", "readwrite", (s) => s.put(handle, name)),
  1269→          handleGet: (name) => tx("handles", "readonly", (s) => s.get(name)),
  1270→          /* FR-028 v2: read current language from localStorage
  1271→             canvasdesk.config (TOML `language = "ru"` or `"en"`).
  1272→             Tour engine (canvasdesk-tour.js) and dropdown picker use
  1273→             this to choose between ru/en scenario variants. Returns
  1274→             "ru" (default) if localStorage is missing/unparseable. */
  1275→          getLanguage: function () {
  1276→            try {
  1277→              var ls = window.localStorage;
  1278→              if (!ls) return "ru";
  1279→              var text = ls.getItem("canvasdesk.config");
  1280→              if (!text) return "ru";
  1281→              // Minimal TOML parser for `language = "..."` line —
  1282→              // avoids pulling a full TOML lib in JS. If the regex
  1283→              // misses, defaults to "ru" (canvas-app default).
  1284→              var m = text.match(/^language\s*=\s*"([^"]+)"/m);
