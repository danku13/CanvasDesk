    36→    36→    pub c0: [f32; 2],

    41→    41→/// Точка порта — центр соответствующей стороны ноды.
    42→    42→pub fn port_point(node: &Node, side: Side) -> [f32; 2] {
    43→    43→    let cx = node.x + node.width / 2.0;
    44→    44→    let cy = node.y + node.height / 2.0;
    45→    45→    match side {
    46→    46→        Side::Top => [cx, node.y],
    47→    47→        Side::Right => [node.x + node.width, cy],
    48→    48→        Side::Bottom => [cx, node.y + node.height],
    49→    49→        Side::Left => [node.x, cy],
    50→    50→    }
    51→    51→}
    52→    52→
    53→    53→/// Внешняя нормаль стороны (направление «от ноды»).
    54→    54→pub fn side_normal(side: Side) -> [f32; 2] {
    55→    55→    match side {
    56→    56→        Side::Top => [0.0, -1.0],
    57→    57→        Side::Right => [1.0, 0.0],
    58→    58→        Side::Bottom => [0.0, 1.0],
    59→    59→        Side::Left => [-1.0, 0.0],
    60→    60→    }
    61→    61→}
    62→    62→
    63→    63→/// Сторона ноды, ближайшая к точке (доминирующая ось с учётом пропорций).
    64→    64→/// Используется при drop резиновой линии и для None-сторон из файла.
    65→    65→pub fn nearest_side(node: &Node, point: [f32; 2]) -> Side {
    66→    66→    let dx = point[0] - (node.x + node.width / 2.0);
    67→    67→    let dy = point[1] - (node.y + node.height / 2.0);
    68→    68→    let nx = dx.abs() / (node.width / 2.0).max(1.0);
    69→    69→    let ny = dy.abs() / (node.height / 2.0).max(1.0);
    70→    70→    if nx >= ny {
    71→    71→        if dx >= 0.0 {
    72→    72→            Side::Right
    73→    73→        } else {
    74→    74→            Side::Left
    75→    75→        }
    76→    76→    } else if dy >= 0.0 {
    77→    77→        Side::Bottom
    78→    78→    } else {
    79→    79→        Side::Top
    80→    80→    }
    81→    81→}
    82→    82→
    83→    83→/// Кривая между портами двух нод. Контрольные точки смещены от портов
    84→    84→/// по нормалям сторон на max(MIN_CONTROL_OFFSET, 40% расстояния).
    85→    85→pub fn bezier_between(a: &Node, sa: Side, b: &Node, sb: Side) -> CubicBezier {
    86→    86→    let p0 = port_point(a, sa);
    87→    87→    let p1 = port_point(b, sb);
    88→    88→    let dist = ((p1[0] - p0[0]).powi(2) + (p1[1] - p0[1]).powi(2)).sqrt();
    89→    89→    let offset = (dist * 0.4).max(MIN_CONTROL_OFFSET);
    90→    90→    let n0 = side_normal(sa);
    91→    91→    let n1 = side_normal(sb);
    92→    92→    CubicBezier {
    93→    93→        p0,
    94→    94→        c0: [p0[0] + n0[0] * offset, p0[1] + n0[1] * offset],
    95→    95→        c1: [p1[0] + n1[0] * offset, p1[1] + n1[1] * offset],
    96→    96→        p1,
    97→    97→    }
    98→    98→}

   100→   100→/// Точка на кривой по параметру t ∈ [0, 1].
   101→   101→pub fn curve_point(curve: &CubicBezier, t: f32) -> [f32; 2] {
   102→   102→    let mt = 1.0 - t;
   103→   103→    let (b0, b1, b2, b3) = (mt * mt * mt, 3.0 * mt * mt * t, 3.0 * mt * t * t, t * t * t);
   104→   104→    [
   105→   105→        b0 * curve.p0[0] + b1 * curve.c0[0] + b2 * curve.c1[0] + b3 * curve.p1[0],
   106→   106→        b0 * curve.p0[1] + b1 * curve.c0[1] + b2 * curve.c1[1] + b3 * curve.p1[1],
   107→   107→    ]
   108→   108→}
   109→   109→
   110→   110→/// Нормированная касательная к кривой в точке t (для стрелки на конце).
   111→   111→pub fn curve_tangent(curve: &CubicBezier, t: f32) -> [f32; 2] {
   112→   112→    let mt = 1.0 - t;
   113→   113→    let dx = 3.0 * mt * mt * (curve.c0[0] - curve.p0[0])
   114→   114→        + 6.0 * mt * t * (curve.c1[0] - curve.c0[0])
   115→   115→        + 3.0 * t * t * (curve.p1[0] - curve.c1[0]);
   116→   116→    let dy = 3.0 * mt * mt * (curve.c0[1] - curve.p0[1])
   117→   117→        + 6.0 * mt * t * (curve.c1[1] - curve.c0[1])
   118→   118→        + 3.0 * t * t * (curve.p1[1] - curve.c1[1]);
   119→   119→    let len = (dx * dx + dy * dy).sqrt();
   120→   120→    if len < f32::EPSILON {
   121→   121→        [1.0, 0.0]
   122→   122→    } else {
   123→   123→        [dx / len, dy / len]
   124→   124→    }
   125→   125→}

   127→   127→/// Тесселяция кривой в полилинию из `segments` отрезков (segments+1 точек).
   128→   128→pub fn tessellate(curve: &CubicBezier, segments: usize) -> Vec<[f32; 2]> {
   129→   129→    let segments = segments.max(1);
   130→   130→    (0..=segments)
   131→   131→        .map(|i| curve_point(curve, i as f32 / segments as f32))
   132→   132→        .collect()
   133→   133→}
   134→   134→
   135→   135→/// Расстояние от точки до отрезка.
   136→   136→fn distance_point_to_segment(p: [f32; 2], a: [f32; 2], b: [f32; 2]) -> f32 {
   137→   137→    let ab = [b[0] - a[0], b[1] - a[1]];
   138→   138→    let ap = [p[0] - a[0], p[1] - a[1]];
   139→   139→    let len2 = ab[0] * ab[0] + ab[1] * ab[1];
   140→   140→    let t = if len2 < f32::EPSILON {
   141→   141→        0.0
   142→   142→    } else {
   143→   143→        ((ap[0] * ab[0] + ap[1] * ab[1]) / len2).clamp(0.0, 1.0)
   144→   144→    };
   145→   145→    let closest = [a[0] + ab[0] * t, a[1] + ab[1] * t];
   146→   146→    ((p[0] - closest[0]).powi(2) + (p[1] - closest[1]).powi(2)).sqrt()
   147→   147→}

   149→   149→/// Расстояние от точки до полилинии (минимум по сегментам).
   150→   150→pub fn distance_point_to_polyline(point: [f32; 2], points: &[[f32; 2]]) -> f32 {
   151→   151→    points
   152→   152→        .windows(2)
   153→   153→        .map(|seg| distance_point_to_segment(point, seg[0], seg[1]))
   154→   154→        .fold(f32::INFINITY, f32::min)
   155→   155→}

   157→   157→/// CR-008: штраф стоимости пары портов, когда порт «спиной» к другой ноде
   158→   158→/// (нормаль смотрит против направления на соседа) — в долях расстояния.
   159→   159→const BACK_FACING_PENALTY: f32 = 1.5;
   160→   160→
   161→   161→/// CR-008: штраф за одинаковые стороны истока и стока (петлеобразная
   162→   162→/// кривая) — в долях расстояния.
   163→   163→const SAME_SIDE_PENALTY: f32 = 1.0;
   164→   164→
   165→   165→/// Порог «спины» по скалярному произведению нормали и направления на
   166→   166→/// соседа: |dot| ниже порога — боковой порт, штрафа нет.
   167→   167→const FACING_DOT_THRESHOLD: f32 = 0.25;
   168→   168→
   169→   169→/// CR-008: пара сторон с кратчайшим путём между нодами. Перебор всех 4×4
   170→   170→/// пар портов; стоимость = расстояние между портами + штрафы за «спинные»
   171→   171→/// порты (нормаль смотрит от соседа) и за одинаковые стороны. Детерминирован:
   172→   172→/// при равной стоимости остаётся первая пара в порядке перебора (Top, Right,
   173→   173→/// Bottom, Left по истоку; внутренний цикл — по стоку).
   174→   174→pub fn best_sides(from: &Node, to: &Node) -> (Side, Side) {
   175→   175→    let sides = [Side::Top, Side::Right, Side::Bottom, Side::Left];
   176→   176→    let mut best = (Side::Top, Side::Top);
   177→   177→    let mut best_cost = f32::INFINITY;
   178→   178→    for &sa in &sides {
   179→   179→        let pa = port_point(from, sa);
   180→   180→        let na = side_normal(sa);
   181→   181→        for &sb in &sides {
   182→   182→            let pb = port_point(to, sb);
   183→   183→            let dx = pb[0] - pa[0];
   184→   184→            let dy = pb[1] - pa[1];
   185→   185→            let dist = dx.hypot(dy).max(1e-3);
   186→   186→            let ux = dx / dist;
   187→   187→            let uy = dy / dist;
   188→   188→            let mut cost = dist;
   189→   189→            // порт истока «спиной» к получателю
   190→   190→            if na[0] * ux + na[1] * uy < -FACING_DOT_THRESHOLD {
   191→   191→                cost += dist * BACK_FACING_PENALTY;
   192→   192→            }
   193→   193→            // порт получателя «спиной» к истоку (нормаль по направлению связи)
   194→   194→            let nb = side_normal(sb);
   195→   195→            if nb[0] * ux + nb[1] * uy > FACING_DOT_THRESHOLD {
   196→   196→                cost += dist * BACK_FACING_PENALTY;
   197→   197→            }
   198→   198→            if sa == sb {
   199→   199→                cost += dist * SAME_SIDE_PENALTY;
   200→   200→            }
   201→   201→            if cost < best_cost - f32::EPSILON {
   202→   202→                best_cost = cost;
   203→   203→                best = (sa, sb);
   204→   204→            }
   205→   205→        }
   206→   206→    }
   207→   207→    best
   208→   208→}
   209→   209→
   210→   210→/// CR-008: эффективные стороны связи — склейка закреплений и геометрии.
   211→   211→/// Закреплённый конец (`canvasdesk.pin_ports`) берёт сохранённую сторону
   212→   212→/// (None — `best_sides` как фолбэк), свободный — сторону кратчайшего пути.
   213→   213→pub fn effective_sides(edge: &Edge, from: &Node, to: &Node) -> (Side, Side) {
   214→   214→    let (from_pinned, to_pinned) = edge.port_pins();
   215→   215→    let (best_from, best_to) = best_sides(from, to);
   216→   216→    let from_side = if from_pinned {
   217→   217→        edge.from_side.unwrap_or(best_from)
   218→   218→    } else {
   219→   219→        best_from
   220→   220→    };
   221→   221→    let to_side = if to_pinned {
   222→   222→        edge.to_side.unwrap_or(best_to)
   223→   223→    } else {
   224→   224→        best_to
   225→   225→    };
   226→   226→    (from_side, to_side)
   227→   227→}
   228→   228→
   229→   229→/// Кривая связи: резолвит ноды по id; стороны — эффективные (CR-008):
   230→   230→/// закреплённый конец — сохранённая сторона, свободный — кратчайший путь
   231→   231→/// по взаимному положению нод (пересчёт на каждом кадре: drag, автораскладка
   232→   232→/// и загрузка идут через один путь). None, если хотя бы одна нода не найдена
   233→   233→/// (висячая связь).
   234→   234→pub fn edge_curve(canvas: &Canvas, edge: &Edge) -> Option<CubicBezier> {
   235→   235→    let from = canvas.node(&edge.from_node)?;
   236→   236→    let to = canvas.node(&edge.to_node)?;
   237→   237→    let (from_side, to_side) = effective_sides(edge, from, to);
   238→   238→    Some(bezier_between(from, from_side, to, to_side))
   239→   239→}

   241→   241→/// Конец связи для перепривязки (CR-002).
   242→   242→#[derive(Debug, Clone, Copy, PartialEq, Eq)]
   243→   243→pub enum EdgeEnd {
   244→   244→    /// Исток (`fromNode`).
   245→   245→    From,
   246→   246→    /// Сток (`toNode`).
   247→   247→    To,
   248→   248→}
   249→   249→
   250→   250→/// Разрешённый конец связи: сторона и точка порта. None-сторона выводится
   251→   251→/// из взаимного положения нод (резолв как в `edge_curve`) — хэндл перепривязки
   252→   252→/// рисуется/ловится там же, где линия фактически начинается. None — связь
   253→   253→/// висячая (ноды нет) или индекс невалиден.
   254→   254→pub fn edge_endpoint(canvas: &Canvas, edge_index: usize, end: EdgeEnd) -> Option<(Side, [f32; 2])> {
   255→   255→    let edge = canvas.edges.get(edge_index)?;
   256→   256→    let from = canvas.node(&edge.from_node)?;
   257→   257→    let to = canvas.node(&edge.to_node)?;
   258→   258→    // CR-008: сторона хэндла — та же эффективная сторона, по которой
   259→   259→    // рисуется линия (пин учитывается, авто следует геометрии)
   260→   260→    let (eff_from, eff_to) = effective_sides(edge, from, to);
   261→   261→    let (node, side) = match end {
   262→   262→        EdgeEnd::From => (from, eff_from),
   263→   263→        EdgeEnd::To => (to, eff_to),
   264→   264→    };
   265→   265→    Some((side, port_point(node, side)))
   266→   266→}

   268→   268→/// Перепривязать конец связи к ноде `target_node_id` со стороной `side`
   269→   269→/// (CR-002). Чистая функция над моделью: id и настройки связи (лейбл,
   270→   270→/// цвет, стиль, толщина) сохраняются. Отказ (false):
   271→   271→///
   272→   272→/// - связь/индекс невалидны (висячая);
   273→   273→/// - целевая нода не найдена;
   274→   274→/// - цель — противоположный конец (петля или слияние концов).
   275→   275→///
   276→   276→/// Успех — true (поля обновлены на месте).
   277→   277→pub fn retarget_edge(
   278→   278→    canvas: &mut Canvas,
   279→   279→    edge_index: usize,
   280→   280→    end: EdgeEnd,
   281→   281→    target_node_id: &str,
   282→   282→    side: Side,
   283→   283→) -> bool {
   284→   284→    // Проверки ДО mutable-заимствования: цель существует и не противоположный
   285→   285→    // конец (immutable-чтения не конфликтуют друг с другом)
   286→   286→    let opposite = {
   287→   287→        let Some(edge) = canvas.edges.get(edge_index) else {
   288→   288→            return false;
   289→   289→        };
   290→   290→        match end {
   291→   291→            EdgeEnd::From => edge.to_node.clone(),
   292→   292→            EdgeEnd::To => edge.from_node.clone(),
   293→   293→        }
   294→   294→    };
   295→   295→    if opposite == target_node_id {
   296→   296→        return false;
   297→   297→    }
   298→   298→    if canvas.node(target_node_id).is_none() {
   299→   299→        return false;
   300→   300→    }
   301→   301→    let Some(edge) = canvas.edges.get_mut(edge_index) else {
   302→   302→        return false;
   303→   303→    };
   304→   304→    let target = target_node_id.to_owned();
   305→   305→    match end {
   306→   306→        EdgeEnd::From => {
   307→   307→            edge.from_node = target;
   308→   308→            edge.from_side = Some(side);
   309→   309→            // FR-025: перепривязка ИСТОКА на другую ноду сбрасывает
   310→   310→            // построчный исток (индекс строки мог не существовать у новой
   311→   311→            // ноды; v1 — деградация до узлового значения)
   312→   312→            edge.from_line = None;
   313→   313→            // FR-029: именованный выход истока тоже сбрасывается — у новой
   314→   314→            // ноды такого выхода может не быть (деградация до узлового
   315→   315→            // значения, семантика как у from_line)
   316→   316→            edge.from_output = None;
   317→   317→        }
   318→   318→        EdgeEnd::To => {
   319→   319→            edge.to_node = target;
   320→   320→            edge.to_side = Some(side);
   321→   321→        }
   322→   322→    }
   323→   323→    true
   324→   324→}

   326→   326→/// Расстояние от world-точки до кривой связи; None для висячей связи.
   327→   327→/// Hit-test: результат < EDGE_HIT_TOLERANCE — попадание. `avoid` — обход
   328→   328→/// посторонних нод (глобальная настройка): рендер и hit-test ходят по одной
   329→   329→/// и той же полилинии, чтобы кликабельная область совпадала с нарисованным.
   330→   330→pub fn distance_to_edge(canvas: &Canvas, edge: &Edge, point: [f32; 2], avoid: bool) -> Option<f32> {
   331→   331→    let points = edge_polyline(canvas, edge, avoid, TESSELLATION_SEGMENTS)?;
   332→   332→    Some(distance_point_to_polyline(point, &points))
   333→   333→}

   335→   335→/// Полилиния связи для рендера/hit-test'а: тесселяция Безье; при avoid —
   336→   336→/// с огибанием посторонних нод. Не препятствия: концевые ноды и ВСЕ их
   337→   337→/// группы-предки (транзитивно) — линк к ноде внутри группы свободно
   338→   338→/// проходит её границу, линк к группе в целом — её конец. Чужие группы
   339→   339→/// (не содержащие концы) огибаются как обычные ноды.
   340→   340→pub fn edge_polyline(
   341→   341→    canvas: &Canvas,
   342→   342→    edge: &Edge,
   343→   343→    avoid: bool,
   344→   344→    segments: usize,
   345→   345→) -> Option<Vec<[f32; 2]>> {
   346→   346→    let curve = edge_curve(canvas, edge)?;
   347→   347→    let points = tessellate(&curve, segments.max(1));
   348→   348→    if !avoid {
   349→   349→        return Some(points);
   350→   350→    }
   351→   351→    let mut excluded: Vec<usize> = Vec::new();
   352→   352→    for id in [&edge.from_node, &edge.to_node] {
   353→   353→        if let Some(i) = canvas.nodes.iter().position(|node| node.id == *id) {
   354→   354→            excluded.push(i);
   355→   355→            excluded.extend(crate::enclosing_group_indices(canvas, i));
   356→   356→        }
   357→   357→    }
   358→   358→    excluded.sort_unstable();
   359→   359→    excluded.dedup();
   360→   360→    let obstacles: Vec<[f32; 4]> = canvas
   361→   361→        .nodes
   362→   362→        .iter()
   363→   363→        .enumerate()
   364→   364→        .filter(|(index, _)| !excluded.contains(index))
   365→   365→        .map(|(_, node)| [node.x, node.y, node.width, node.height])
   366→   366→        .collect();
   367→   367→    Some(route_polyline(
   368→   368→        &points,
   369→   369→        &obstacles,
   370→   370→        AVOID_MARGIN,
   371→   371→        MAX_DETOURS,
   372→   372→    ))
   373→   373→}
   374→   374→
   375→   375→/// Середина связи по длине дуги (для лейбла и бокса редактирования):
   376→   376→/// при avoid совпадает с видимой огибающей линией.
   377→   377→pub fn edge_midpoint(canvas: &Canvas, edge: &Edge, avoid: bool) -> Option<[f32; 2]> {
   378→   378→    let points = edge_polyline(canvas, edge, avoid, TESSELLATION_SEGMENTS)?;
   379→   379→    let total: f32 = points
   380→   380→        .windows(2)
   381→   381→        .map(|w| (w[1][0] - w[0][0]).hypot(w[1][1] - w[0][1]))
   382→   382→        .sum();
   383→   383→    let half = total / 2.0;
   384→   384→    let mut acc = 0.0;
   385→   385→    for w in points.windows(2) {
   386→   386→        let seg = (w[1][0] - w[0][0]).hypot(w[1][1] - w[0][1]);
   387→   387→        if acc + seg >= half {
   388→   388→            let t = ((half - acc) / seg.max(f32::EPSILON)).clamp(0.0, 1.0);
   389→   389→            return Some([
   390→   390→                w[0][0] + (w[1][0] - w[0][0]) * t,
   391→   391→                w[0][1] + (w[1][1] - w[0][1]) * t,
   392→   392→            ]);
   393→   393→        }
   394→   394→        acc += seg;
   395→   395→    }
   396→   396→    points.last().copied().or_else(|| points.first().copied())
   397→   397→}

   401→   401→/// Rect [x, y, w, h], инфлированный на margin.
   402→   402→fn inflate_rect(rect: [f32; 4], margin: f32) -> [f32; 4] {
   403→   403→    [
   404→   404→        rect[0] - margin,
   405→   405→        rect[1] - margin,
   406→   406→        rect[2] + margin * 2.0,
   407→   407→        rect[3] + margin * 2.0,
   408→   408→    ]
   409→   409→}

   411→   411→/// Точка внутри rect (граница считается внутренностью).
   412→   412→fn point_in_rect(p: [f32; 2], rect: [f32; 4]) -> bool {
   413→   413→    p[0] >= rect[0] && p[0] <= rect[0] + rect[2] && p[1] >= rect[1] && p[1] <= rect[1] + rect[3]
   414→   414→}

   416→   416→/// Пересекается ли отрезок a–b с прямоугольником (slab-метод).
   417→   417→fn segment_hits_rect(a: [f32; 2], b: [f32; 2], rect: [f32; 4]) -> bool {
   418→   418→    let d = [b[0] - a[0], b[1] - a[1]];
   419→   419→    let mut t_enter = 0.0f32;
   420→   420→    let mut t_exit = 1.0f32;
   421→   421→    for axis in 0..2 {
   422→   422→        let (start, dir, lo, hi) = if axis == 0 {
   423→   423→            (a[0], d[0], rect[0], rect[0] + rect[2])
   424→   424→        } else {
   425→   425→            (a[1], d[1], rect[1], rect[1] + rect[3])
   426→   426→        };
   427→   427→        if dir.abs() < f32::EPSILON {
   428→   428→            if start < lo || start > hi {
   429→   429→                return false;
   430→   430→            }
   431→   431→        } else {
   432→   432→            let (t0, t1) = ((lo - start) / dir, (hi - start) / dir);
   433→   433→            let (t0, t1) = (t0.min(t1), t0.max(t1));
   434→   434→            t_enter = t_enter.max(t0);
   435→   435→            t_exit = t_exit.min(t1);
   436→   436→            if t_enter > t_exit {
   437→   437→                return false;
   438→   438→            }
   439→   439→        }
   440→   440→    }
   441→   441→    true
   442→   442→}
   443→   443→
   444→   444→/// Точка входа отрезка a–b в rect (a снаружи): пересечение с ближней гранью.
   445→   445→fn rect_entry(a: [f32; 2], b: [f32; 2], rect: [f32; 4]) -> Option<[f32; 2]> {
   446→   446→    let d = [b[0] - a[0], b[1] - a[1]];
   447→   447→    let mut t_enter = 0.0f32;
   448→   448→    for axis in 0..2 {
   449→   449→        let (start, dir, lo, hi) = if axis == 0 {
   450→   450→            (a[0], d[0], rect[0], rect[0] + rect[2])
   451→   451→        } else {
   452→   452→            (a[1], d[1], rect[1], rect[1] + rect[3])
   453→   453→        };
   454→   454→        if dir.abs() < f32::EPSILON {
   455→   455→            continue;
   456→   456→        }
   457→   457→        let (t0, t1) = ((lo - start) / dir, (hi - start) / dir);
   458→   458→        t_enter = t_enter.max(t0.min(t1));
   459→   459→    }
   460→   460→    (t_enter > 0.0 && t_enter <= 1.0).then(|| [a[0] + d[0] * t_enter, a[1] + d[1] * t_enter])
   461→   461→}
   462→   462→
   463→   463→/// Параметр точки на границе rect вдоль периметра (по часовой от левого
   464→   464→/// верхнего угла): верх → право → низ → лево. Углы: TL=0, TR=w, BR=w+h,
   465→   465→/// BL=2w+h; периметр L = 2(w+h).
   466→   466→fn boundary_param(p: [f32; 2], rect: [f32; 4]) -> f32 {
   467→   467→    let (w, h) = (rect[2], rect[3]);
   468→   468→    let eps = 1e-3;
   469→   469→    if (p[1] - rect[1]).abs() <= eps {
   470→   470→        (p[0] - rect[0]).clamp(0.0, w)
   471→   471→    } else if (p[0] - (rect[0] + w)).abs() <= eps {
   472→   472→        w + (p[1] - rect[1]).clamp(0.0, h)
   473→   473→    } else if (p[1] - (rect[1] + h)).abs() <= eps {
   474→   474→        w + h + (rect[0] + w - p[0]).clamp(0.0, w)
   475→   475→    } else {
   476→   476→        2.0 * w + h + (rect[1] + h - p[1]).clamp(0.0, h)
   477→   477→    }
   478→   478→}
   479→   479→
   480→   480→/// Ближайшая к точке точка границы rect.
   481→   481→fn nearest_boundary_point(p: [f32; 2], rect: [f32; 4]) -> [f32; 2] {
   482→   482→    let (l, t, w, h) = (rect[0], rect[1], rect[2], rect[3]);
   483→   483→    let cx = p[0].clamp(l, l + w);
   484→   484→    let cy = p[1].clamp(t, t + h);
   485→   485→    if cx != p[0] || cy != p[1] {
   486→   486→        return [cx, cy];
   487→   487→    }
   488→   488→    // Внутри: ближайшая грань.
   489→   489→    let candidates = [
   490→   490→        ([l, cy], cx - l),
   491→   491→        ([l + w, cy], l + w - cx),
   492→   492→        ([cx, t], cy - t),
   493→   493→        ([cx, t + h], t + h - cy),
   494→   494→    ];
   495→   495→    candidates
   496→   496→        .into_iter()
   497→   497→        .min_by(|a, b| a.1.partial_cmp(&b.1).unwrap_or(std::cmp::Ordering::Equal))
   498→   498→        .map(|(point, _)| point)
   499→   499→        .unwrap_or([cx, cy])
   500→   500→}
   501→   501→
   502→   502→/// Угловые waypoints вдоль более короткого пути по периметру rect от p до q
   503→   503→/// (обе на границе). Порядок — от p к q, q НЕ включена.
   504→   504→fn detour_corners(rect: [f32; 4], p: [f32; 2], q: [f32; 2]) -> Vec<[f32; 2]> {
   505→   505→    let (l, t, w, h) = (rect[0], rect[1], rect[2], rect[3]);
   506→   506→    let perimeter = 2.0 * (w + h);
   507→   507→    if perimeter <= f32::EPSILON {
   508→   508→        return Vec::new();
   509→   509→    }
   510→   510→    let s_p = boundary_param(p, rect);
   511→   511→    let s_q = boundary_param(q, rect);
   512→   512→    let dist_cw = (s_q - s_p).rem_euclid(perimeter);
   513→   513→    let dist_ccw = perimeter - dist_cw;
   514→   514→    let corners = [
   515→   515→        (0.0f32, [l, t]),
   516→   516→        (w, [l + w, t]),
   517→   517→        (w + h, [l + w, t + h]),
   518→   518→        (2.0 * w + h, [l, t + h]),
   519→   519→    ];
   520→   520→    let mut waypoints: Vec<(f32, [f32; 2])> = corners
   521→   521→        .into_iter()
   522→   522→        .filter_map(|(s, point)| {
   523→   523→            if dist_cw <= dist_ccw {
   524→   524→                let off = (s - s_p).rem_euclid(perimeter);
   525→   525→                (off > 1e-3 && off < dist_cw - 1e-3).then_some((off, point))
   526→   526→            } else {
   527→   527→                let off = (s_p - s).rem_euclid(perimeter);
   528→   528→                (off > 1e-3 && off < dist_ccw - 1e-3).then_some((off, point))
   529→   529→            }
   530→   530→        })
   531→   531→        .collect();
   532→   532→    waypoints.sort_by(|a, b| a.0.partial_cmp(&b.0).unwrap_or(std::cmp::Ordering::Equal));
   533→   533→    waypoints.into_iter().map(|(_, point)| point).collect()
   534→   534→}

   536→   536→/// Огибание препятствий: полилиния обходит инфлированные прямоугольники
   537→   537→/// нод короткой стороной. Детерминировано; при превышении `max_detours`
   538→   538→/// возвращается исходная полилиния (деградация к прямой Безье, без петель).
   539→   539→pub fn route_polyline(
   540→   540→    points: &[[f32; 2]],
   541→   541→    obstacles: &[[f32; 4]],
   542→   542→    margin: f32,
   543→   543→    max_detours: usize,
   544→   544→) -> Vec<[f32; 2]> {
   545→   545→    if points.len() < 2 {
   546→   546→        return points.to_vec();
   547→   547→    }
   548→   548→    let rects: Vec<[f32; 4]> = obstacles
   549→   549→        .iter()
   550→   550→        .map(|rect| inflate_rect(*rect, margin))
   551→   551→        .filter(|rect| rect[2] > 0.0 && rect[3] > 0.0)
   552→   552→        .collect();
   553→   553→    if rects.is_empty() {
   554→   554→        return points.to_vec();
   555→   555→    }
   556→   556→    let mut path = points.to_vec();
   557→   557→    for _ in 0..max_detours {
   558→   558→        // Первое пересечение любого сегмента с любым препятствием.
   559→   559→        let mut hit: Option<(usize, usize, [f32; 2], usize)> = None;
   560→   560→        'scan: for i in 0..path.len() - 1 {
   561→   561→            for (ri, rect) in rects.iter().enumerate() {
   562→   562→                if !segment_hits_rect(path[i], path[i + 1], *rect) {
   563→   563→                    continue;
   564→   564→                }
   565→   565→                let entry = if point_in_rect(path[i], *rect) {
   566→   566→                    path[i]
   567→   567→                } else {
   568→   568→                    rect_entry(path[i], path[i + 1], *rect).unwrap_or(path[i])
   569→   569→                };
   570→   570→                // Индекс первой точки полилинии за пределами rect.
   571→   571→                let mut k = i + 1;
   572→   572→                while k < path.len() && point_in_rect(path[k], *rect) {
   573→   573→                    k += 1;
   574→   574→                }
   575→   575→                hit = Some((i, ri, entry, k));
   576→   576→                break 'scan;
   577→   577→            }
   578→   578→        }
   579→   579→        let Some((i, ri, entry, k)) = hit else {
   580→   580→            break;
   581→   581→        };
   582→   582→        let rect = rects[ri];
   583→   583→        let target = path.get(k).copied().or_else(|| path.last().copied());
   584→   584→        let Some(target) = target else { break };
   585→   585→        let exit = nearest_boundary_point(target, rect);
   586→   586→        let corners = detour_corners(rect, entry, exit);
   587→   587→        if (exit[0] - entry[0]).abs() < f32::EPSILON
   588→   588→            && (exit[1] - entry[1]).abs() < f32::EPSILON
   589→   589→            && corners.is_empty()
   590→   590→        {
   591→   591→            // Касание без огибания — дальше прогресса не будет.
   592→   592→            break;
   593→   593→        }
   594→   594→        // Склейка: путь до входа, угловые waypoints, выход, остаток от k.
   595→   595→        let mut rerouted = Vec::with_capacity(path.len() + corners.len() + 2);
   596→   596→        rerouted.extend_from_slice(&path[..=i]);
   597→   597→        if (entry[0] - path[i][0]).abs() > f32::EPSILON
   598→   598→            || (entry[1] - path[i][1]).abs() > f32::EPSILON
   599→   599→        {
   600→   600→            rerouted.push(entry);
   601→   601→        }
   602→   602→        rerouted.extend_from_slice(&corners);
   603→   603→        rerouted.push(exit);
   604→   604→        rerouted.extend_from_slice(&path[k.min(path.len())..]);
   605→   605→        path = rerouted;
   606→   606→    }
   607→   607→    path
   608→   608→}

   610→   610→/// Порт ноды под курсором: сторона, чья точка порта ближе всего к `point`
   611→   611→/// в пределах допуска `tolerance_px` экранных пикселей (zoom — camera.zoom();
   612→   612→/// CR-003: допуск — параметр, дефолт — `PORT_HIT_PX`).
   613→   613→pub fn port_at(node: &Node, point: [f32; 2], zoom: f32, tolerance_px: f32) -> Option<Side> {
   614→   614→    let tolerance = tolerance_px / zoom.max(1e-3);
   615→   615→    [Side::Top, Side::Right, Side::Bottom, Side::Left]
   616→   616→        .into_iter()
   617→   617→        .map(|side| {
   618→   618→            let port = port_point(node, side);
   619→   619→            let dist = ((point[0] - port[0]).powi(2) + (point[1] - port[1]).powi(2)).sqrt();
   620→   620→            (dist, side)
   621→   621→        })
   622→   622→        .filter(|(dist, _)| *dist <= tolerance)
   623→   623→        .min_by(|a, b| a.0.partial_cmp(&b.0).unwrap_or(std::cmp::Ordering::Equal))
   624→   624→        .map(|(_, side)| side)
   625→   625→}

   702→   702→/// Ближайшая к точке связь в допуске EDGE_HIT_TOLERANCE.
   703→   703→/// Возвращает индекс в `canvas.edges`; None — промах (или все связи висячие).
   704→   704→/// `avoid` — обход посторонних нод (см. `edge_polyline`).
   705→   705→pub fn edge_at(canvas: &Canvas, point: [f32; 2], avoid: bool) -> Option<usize> {
   706→   706→    canvas
   707→   707→        .edges
   708→   708→        .iter()
   709→   709→        .enumerate()
   710→   710→        .filter_map(|(index, edge)| {
   711→   711→            distance_to_edge(canvas, edge, point, avoid).map(|d| (index, d))
   712→   712→        })
   713→   713→        .filter(|(_, dist)| *dist < EDGE_HIT_TOLERANCE)
   714→   714→        .min_by(|a, b| a.1.partial_cmp(&b.1).unwrap_or(std::cmp::Ordering::Equal))
   715→   715→        .map(|(index, _)| index)
   716→   716→}

   718→   718→/// Кривая «резиновой линии» при drag новой связи: от точки порта к курсору.
   719→   719→/// Контрольная точка у порта смещена по нормали стороны, у курсора —
   720→   720→/// совпадает с ним (линия «прилипает» к указателю без выраженного изгиба).
   721→   721→pub fn draft_curve(port: [f32; 2], side: Side, to: [f32; 2]) -> CubicBezier {
   722→   722→    let dist = ((to[0] - port[0]).powi(2) + (to[1] - port[1]).powi(2)).sqrt();
   723→   723→    let offset = (dist * 0.4).max(MIN_CONTROL_OFFSET);
   724→   724→    let normal = side_normal(side);
   725→   725→    CubicBezier {
   726→   726→        p0: port,
   727→   727→        c0: [port[0] + normal[0] * offset, port[1] + normal[1] * offset],
   728→   728→        c1: to,
   729→   729→        p1: to,
   730→   730→    }
   731→   731→}
