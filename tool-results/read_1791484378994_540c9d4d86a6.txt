   113→impl OnboardingState {
   114→    /// Число шагов.
   115→    pub fn len(&self) -> usize {
   116→        ONBOARDING_STEPS.len()
   117→    }
   118→
   119→    /// Тур пуст (недостижимо с непустым ONBOARDING_STEPS — для полноты).
   120→    pub fn is_empty(&self) -> bool {
   121→        ONBOARDING_STEPS.is_empty()
   122→    }
   123→
   124→    /// Шаг вперёд; false на последнем шаге (вызывающий завершает тур —
   125→    /// «Готово» ставит `onboarding_done` и сохраняет конфиг).
   126→    // Не Iterator::next — шаги карусели с клампом, а не выдача элементов
   127→    #[allow(clippy::should_implement_trait)]
   128→    pub fn next(&mut self) -> bool {
   129→        if self.step + 1 < ONBOARDING_STEPS.len() {
   130→            self.step += 1;
   131→            true
   132→        } else {
   133→            false
   134→        }
   135→    }
   136→
   137→    /// Шаг назад; false на первом (кнопки «Назад» там нет — hit-тест None).
   138→    pub fn prev(&mut self) -> bool {
   139→        if self.step == 0 {
   140→            return false;
   141→        }
   142→        self.step -= 1;
   143→        true
   144→    }
   145→
   146→    /// Последний шаг (подпись кнопки меняется на «Готово»).
   147→    pub fn is_last(&self) -> bool {
   148→        self.step + 1 >= ONBOARDING_STEPS.len()
   149→    }
   150→
   151→    /// Ключ подписи правой кнопки: «Далее» / «Готово» на последнем шаге
   152→    /// (текст — таблица [`crate::i18n`], FR-040).
   153→    pub fn next_label_key(&self) -> &'static str {
   154→        if self.is_last() {
   155→            keys::ONBOARDING_DONE
   156→        } else if let Some(action) = ONBOARDING_STEPS
   157→            .get(self.step)
   158→            .and_then(|step| step.action_key)
   159→        {
   160→            // FR-049: шаг с действием — CTA «Попробовать» (галерея схем)
   161→            action
   162→        } else {
   163→            keys::ONBOARDING_NEXT
   164→        }
   165→    }
   166→
   167→    /// Ключ подписи левой кнопки: «Назад» / нет на первом шаге.
   168→    pub fn prev_label_key(&self) -> Option<&'static str> {
   169→        (self.step > 0).then_some(keys::ONBOARDING_BACK)
   170→    }
   171→}

   211→/// Строки тела шага после переноса по ширине карточки — измеренный
   212→/// `TextMeasurer::wrap` (W-e, CR-015: замена эвристики
   213→/// «символы × 0.62»; те же метрики, что у шейпинга рендера). Один источник
   214→/// для высоты карточки ([`card_layout`]) и рендера (overlays.rs) —
   215→/// раскладка и геометрия не разъезжаются.
   216→pub fn body_lines(
   217→    step: usize,
   218→    width: f32,
   219→    language: Language,
   220→    m: &mut TextMeasurer,
   221→    fs: &mut cosmic_text::FontSystem,
   222→) -> Vec<String> {
   223→    let Some(step) = ONBOARDING_STEPS.get(step) else {
   224→        return Vec::new();
   225→    };
   226→    let avail = (width - ONBOARDING_PAD * 2.0).max(10.0);
   227→    let wrapped = m.wrap(
   228→        fs,
   229→        i18n::tr(language, step.body_key),
   230→        FAMILY,
   231→        ONBOARDING_BODY_FONT,
   232→        avail,
   233→    );
   234→    // W-e follow-up (CI windows): сверхширокий токен без переносов
   235→    // («markdown-разметкой» на Windows-метриках — 226.6 > бюджета 192)
   236→    // выходил за карточку и на рендере, и в тесте бюджета. Жёсткий
   237→    // посимвольный разрыв в бюджет: строка либо влезает, либо рвётся
   238→    // глиф за глифом (тексты онбординга — кириллица/латиница).
   239→    let mut lines: Vec<String> = Vec::with_capacity(wrapped.len());
   240→    for line in wrapped {
   241→        if m.width_of(fs, &line, FAMILY, ONBOARDING_BODY_FONT) <= avail {
   242→            lines.push(line);
   243→            continue;
   244→        }
   245→        let mut cur = String::new();
   246→        for ch in line.chars() {
   247→            let mut next = cur.clone();
   248→            next.push(ch);
   249→            if m.width_of(fs, &next, FAMILY, ONBOARDING_BODY_FONT) > avail && !cur.is_empty() {
   250→                lines.push(std::mem::take(&mut cur));
   251→            }
   252→            cur.push(ch);
   253→        }
   254→        if !cur.is_empty() {
   255→            lines.push(cur);
   256→        }
   257→    }
   258→    lines
   259→}

   261→/// Отступ от верха карточки до первой строки тела (заголовок +
   262→/// прогресс-точки + зазор) — общий для геометрии и рендера.
   263→pub fn body_top_offset() -> f32 {
   264→    ONBOARDING_PAD + ONBOARDING_TITLE_LINE_H + ONBOARDING_DOTS_TOP + ONBOARDING_DOT + 10.0
   265→}

   267→/// Зона тела шага в карточке: от якоря первой строки до футера с CTA
   268→/// (минус нижний пад). Общий источник высоты окна видимости скролла для
   269→/// раскладки ([`card_layout`]), колеса ввода и отрисовки (видимые строки).
   270→pub fn body_area(card: [f32; 4]) -> [f32; 4] {
   271→    [
   272→        card[0] + ONBOARDING_PAD,
   273→        card[1] + body_top_offset(),
   274→        (card[2] - ONBOARDING_PAD * 2.0).max(0.0),
   275→        (card[3] - body_top_offset() - ONBOARDING_PAD - ONBOARDING_FOOTER_H).max(0.0),
   276→    ]
   277→}

   279→/// Раскладка карточки шага (W-e): слот = вьюпорт минус поля
   280→/// ([`ONBOARDING_VIEWPORT_MARGIN`]), панель — `kit::modal` (центр слота,
   281→/// как dialog/autolink); ширина — кламп к слоту, высота — по измеренному
   282→/// телу с клампом к слоту (инвариант FR-028 «карточка целиком в окне» —
   283→/// теперь ещё и с полями). При клампе высоты контент ужимается:
   284→/// скролл-состояние тела синхронизируется с контентом/окном (футер с CTA
   285→/// всегда видим — в скролл-зону не входит). Один источник для реестра
   286→/// (hit-rect'ы), ввода (кнопки) и отрисовки — «ввод = тому, что видно».
   287→pub struct OnboardingLayout {
   288→    /// Rect карточки `[x, y, w, h]` (центр окна, кламп с полями).
   289→    pub card: [f32; 4],
   290→    /// Строки тела после измеренного переноса по ширине карточки.
   291→    pub lines: Vec<String>,
   292→}
   293→
   294→/// Раскладка карточки шага: карточка + строки тела + синхронизация
   295→/// скролл-состояния тела (кит [`ScrollState`] — offset сохраняется,
   296→/// content_h/viewport_h обновляются, позиция клампится; паттерн
   297→/// `flow_map_layout`).
   298→pub fn card_layout(
   299→    viewport: [f32; 2],
   300→    step: usize,
   301→    language: Language,
   302→    scroll: &mut ScrollState,
   303→    m: &mut TextMeasurer,
   304→    fs: &mut cosmic_text::FontSystem,
   305→) -> OnboardingLayout {
   306→    let margin = ONBOARDING_VIEWPORT_MARGIN;
   307→    let slot = UiRect::new(
   308→        margin,
   309→        margin,
   310→        (viewport[0] - margin * 2.0).max(0.0),
   311→        (viewport[1] - margin * 2.0).max(0.0),
   312→    );
   313→    let w = ONBOARDING_CARD_WIDTH.min(slot.w);
   314→    let lines = body_lines(step, w, language, m, fs);
   315→    let body_h = lines.len() as f32 * ONBOARDING_BODY_LINE_H;
   316→    let desired_h = body_top_offset() + body_h + ONBOARDING_PAD + ONBOARDING_FOOTER_H;
   317→    // Кламп высоты к слоту: desired пре-клампнут, поэтому min-инвариант
   318→    // модали (приоритетен — parity FR-060) не конфликтует с клампом FR-028.
   319→    let h = desired_h.min(slot.h);
   320→    let size = UiVec2::new(w, h);
   321→    let panel = kit::modal(slot, size, size, size).panel;
   322→    let card = [panel.x, panel.y, panel.w, panel.h];
   323→    // Скролл тела: при клампе высоты viewport_h < content_h — строки
   324→    // прокручиваются (колесо ввода; видимые строки — `kit::list_rows`).
   325→    let body = body_area(card);
   326→    scroll.content_h = lines.len() as f32 * ONBOARDING_BODY_LINE_H;
   327→    scroll.viewport_h = body[3];
   328→    scroll.clamp();
   329→    OnboardingLayout { card, lines }
   330→}

   332→/// Прогресс-точки: центры по горизонтали (рендер — квады-кружки).
   333→pub fn progress_dots(card: [f32; 4]) -> (Vec<f32>, f32) {
   334→    let n = ONBOARDING_STEPS.len() as f32;
   335→    let total = n * ONBOARDING_DOT + (n - 1.0) * ONBOARDING_DOT_GAP;
   336→    let start = card[0] + (card[2] - total) / 2.0;
   337→    let centers: Vec<f32> = (0..ONBOARDING_STEPS.len())
   338→        .map(|i| start + i as f32 * (ONBOARDING_DOT + ONBOARDING_DOT_GAP) + ONBOARDING_DOT / 2.0)
   339→        .collect();
   340→    let y = card[1] + ONBOARDING_PAD + ONBOARDING_TITLE_LINE_H + ONBOARDING_DOTS_TOP;
   341→    (centers, y)
   342→}

   344→/// Rect кнопки карточки (футер прибит к низу карточки — при клампе высоты
   345→/// кнопки остаются достижимы): Prev — слева, Next — справа (оба в футере);
   346→/// Skip — в правом верхнем углу карточки (выход виден всегда — NN/g).
   347→//
   348→// K5/FR-070: `Skip` — это подписанная прямоугольная кнопка 64×22
   349→// («Пропустить»), НЕ каноническая «×» в углу модали. Поэтому `Skip` НЕ
   350→// является миграционным кандидатом для `kit::stage_close_button` (тот
   351→// возвращает квадрат `ICON_BUTTON_SIZE`=26×26 под глиф «×», а Skip —
   352→// широкая кнопка-призрак с подписью). Этот TODO — НЕ close-button
   353→// migration (вопреки FR-070 тегу), а кнопочный layout-миграция.
   354→//
   355→// O2 (Task O / FR-UI-FOOTER): Prev/Next carousel footer migrated to
   356→// `kit::split_footer_buttons` (Agent L's split-footer kit component).
   357→// Pattern: Prev on LEFT (slot.x + inset), Next on RIGHT (slot.right -
   358→// inset - W), inset = ONBOARDING_PAD = 24, gap = GAP_CONTROLS (8,
   359→// unused — 1 button per side). Skip stays manual (top-right corner —
   360→// NOT in the footer slot). Positions are 1:1 with the previous
   361→// hand-rolled formula (footer_y = card.bottom - FOOTER_H + (FOOTER_H -
   362→// BTN_H)/2; Prev.x = card.x + PAD; Next.x = card.right - PAD - W) —
   363→// I-1: zero visual jump.
   364→//
   365→// Будущая миграция Skip — на `kit::button_layout` (измеренный размер
   366→// подписи + `BUTTON_PAD_H` × 2, высота `BUTTON_HEIGHT`=30). Требует:
   367→// 1. Сигнатуру `button_rect(card, button, m, fs, family, size)` —
   368→//    вместо текущей без замерщика (10+ callers в onboarding_ui.rs,
   369→//    app/overlays.rs:3669, app.rs:13310 — большая площадь рефактора).
   370→// 2. UX-ревью: размер меняется с 64×22 на ~(80–100)×30 (ширина
   371→//    измеренной подписи «Пропустить» + 2·BUTTON_PAD_H=12; высота — kit
   372→//    canonical `BUTTON_HEIGHT`=30). Это сдвиг позиции кнопки в карточке
   373→//    (примерно +6px по высоте, +16–36px по ширине).
   374→// 3. Стиль: `kit::button_style(ButtonVariant::Ghost, KitState::Normal,
   375→//    palette)` — ghost-кнопка без заливки/рамки, hover → accent border.
   376→//
   377→// Оставлено как TODO до отдельной волны onboarding button_layout
   378→// (I-1: ноль скачка). Паритет с другими подписанными кнопками-призраками
   379→// в карточках (например suggest-cards).
   380→pub fn button_rect(card: [f32; 4], button: OnboardingButton) -> [f32; 4] {
   381→    match button {
   382→        OnboardingButton::Prev | OnboardingButton::Next => {
   383→            // O2: split footer (Prev left / Next right) via kit.
   384→            let slot = UiRect::new(
   385→                card[0],
   386→                card[1] + card[3] - ONBOARDING_FOOTER_H,
   387→                card[2],
   388→                ONBOARDING_FOOTER_H,
   389→            );
   390→            let btns = kit::split_footer_buttons(
   391→                slot,
   392→                &[ONBOARDING_BUTTON_W],
   393→                &[ONBOARDING_BUTTON_W],
   394→                kit::GAP_CONTROLS,
   395→                ONBOARDING_PAD,
   396→            );
   397→            // btns[0] = (Prev rect, FooterGroup::Left, 0);
   398→            // btns[1] = (Next rect, FooterGroup::Right, 0).
   399→            let idx = match button {
   400→                OnboardingButton::Prev => 0,
   401→                OnboardingButton::Next => 1,
   402→                _ => unreachable!(),
   403→            };
   404→            let r = btns[idx].0;
   405→            [r.x, r.y, r.w, r.h]
   406→        }
   407→        OnboardingButton::Skip => [
   408→            card[0] + card[2] - ONBOARDING_PAD - 64.0,
   409→            card[1] + 10.0,
   410→            64.0,
   411→            22.0,
   412→        ],
   413→    }
   414→}

   416→/// Hit-test кнопки карточки. «Назад» на первом шаге отсутствует (None),
   417→/// «Далее»/«Готово» и «Пропустить» доступны всегда (инвариант карусели).
   418→/// FR-097: на coarse-указателе тач-цели кнопок дотягиваются до 44 лог. px
   419→/// (hit-only, кламп в карточку — расширенная зона не выходит за оверлей;
   420→/// на точном указателе — прежние зоны).
   421→pub fn button_at(
   422→    card: [f32; 4],
   423→    state: &OnboardingState,
   424→    point: [f32; 2],
   425→) -> Option<OnboardingButton> {
   426→    let coarse = crate::touch_targets::pointer_coarse();
   427→    let hit = |rect: [f32; 4]| {
   428→        let rect = if coarse {
   429→            crate::touch_targets::intersect_xywh(
   430→                crate::touch_targets::expand_xywh(rect, crate::touch_targets::MIN_TOUCH_TARGET),
   431→                card,
   432→            )
   433→        } else {
   434→            rect
   435→        };
   436→        point[0] >= rect[0]
   437→            && point[0] <= rect[0] + rect[2]
   438→            && point[1] >= rect[1]
   439→            && point[1] <= rect[1] + rect[3]
   440→    };
   441→    if hit(button_rect(card, OnboardingButton::Next)) {
   442→        return Some(OnboardingButton::Next);
   443→    }
   444→    if hit(button_rect(card, OnboardingButton::Skip)) {
   445→        return Some(OnboardingButton::Skip);
   446→    }
   447→    // «Назад» только не на первом шаге
   448→    if state.step > 0 && hit(button_rect(card, OnboardingButton::Prev)) {
   449→        return Some(OnboardingButton::Prev);
   450→    }
   451→    None
   452→}

   454→/// Клик внутри карточки (но не по кнопкам)? Ввод канваса глотается всей
   455→/// карточкой — клик мимо кнопок не проваливается под оверлей.
   456→pub fn point_in_card(card: [f32; 4], point: [f32; 2]) -> bool {
   457→    point[0] >= card[0]
   458→        && point[0] <= card[0] + card[2]
   459→        && point[1] >= card[1]
   460→        && point[1] <= card[1] + card[3]
   461→}

   614→/// Раскладка экрана AI-онбординга: rect карточки + высота (зависит от

   616→#[derive(Debug, Clone, PartialEq)]
   617→pub struct AiOnboardingLayout {
   618→    /// Rect карточки `[x, y, w, h]` (центр вьюпорта, кламп с полями).
   619→    pub card: [f32; 4],
   620→    /// Rect каждой карточки-режима (Local / Cloud / SelfHosted) — порядок
   621→    /// [`AiOnboardingChoice`]: Local, Cloud, SelfHosted.
   622→    pub mode_cards: [[f32; 4]; 3],
   623→    /// Rect кнопок «Подробнее о privacy» и «Продолжить».
   624→    pub btn_privacy: [f32; 4],
   625→    pub btn_continue: [f32; 4],
   626→    /// Rect блока «Подробнее о privacy» (когда `privacy_open = true`).
   627→    pub privacy_block: [f32; 4],

   630→/// Вычислить раскладку экрана AI-онбординга (чистая функция, без замерщика —
   631→/// высоты строк фиксированы по прототипу F-8). Карточка центрируется во
   632→/// вьюпорте с полями `AI_ONB_VIEWPORT_MARGIN`; высота зависит от
   633→/// `state.privacy_open`.
   634→//
   635→// O1 (Task O / FR-UI-RADIO): mode-card rendering migrated to `kit::radio_card`
   636→// in `app/overlays.rs::ai_onboarding_overlay` (where palette is available —
   637→// this layout function stays a pure geometry function returning slot rects,
   638→// per task step 1 option (b); signature unchanged). The kit's
   639→// `radio_card(slot, label_w, desc_w, selected, state, palette)` is called
   640→// per-card in overlays.rs; `paint_radio_card` emits the card background +
   641→// indicator; `layout.label`/`layout.desc` rects are used for text positions.
   642→// Tag (right corner) stays separate — kit doesn't support tag.
   643→pub fn ai_onboarding_layout(viewport: [f32; 2], state: &AiOnboardingState) -> AiOnboardingLayout {
   644→    use canvas_ui::geometry::{UiRect, UiVec2};
   645→    use canvas_ui::kit;
   646→    let margin = AI_ONB_VIEWPORT_MARGIN;
   647→    let slot = UiRect::new(
   648→        margin,
   649→        margin,
   650→        (viewport[0] - margin * 2.0).max(0.0),
   651→        (viewport[1] - margin * 2.0).max(0.0),
   652→    );
   653→    let w = AI_ONB_CARD_W.min(slot.w);
   654→    // Высота: step + title + sub + 3 mode cards (с зазорами) + actions +
   655→    // опционально privacy block. Поля — top/bottom.
   656→    let modes_h = 3.0 * AI_ONB_MODE_H + 2.0 * AI_ONB_MODE_GAP;
   657→    let mut h = AI_ONB_PAD_TOP
   658→        + AI_ONB_STEP_H
   659→        + AI_ONB_TITLE_H
   660→        + AI_ONB_SUB_H
   661→        + modes_h
   662→        + 18.0 // зазор от режимов до actions (прототип: margin-top: 18px)
   663→        + AI_ONB_ACTIONS_H
   664→        + AI_ONB_PAD_BOTTOM;
   665→    if state.privacy_open {
   666→        h += 15.0 + AI_ONB_PRIV_H; // зазор + блок
   667→    }
   668→    let h = h.min(slot.h);
   669→    let panel = kit::modal(
   670→        slot,
   671→        UiVec2::new(w, h),
   672→        UiVec2::new(w, h),
   673→        UiVec2::new(w, h),
   674→    )
   675→    .panel;
   676→    let card = [panel.x, panel.y, panel.w, panel.h];
   677→    // Y-positions: шаг → заголовок → подзаголовок → 3 карточки → actions → privacy.
   678→    let x = card[0] + AI_ONB_PAD_X;
   679→    let inner_w = (card[2] - AI_ONB_PAD_X * 2.0).max(0.0);
   680→    let mut y = card[1] + AI_ONB_PAD_TOP;
   681→    y += AI_ONB_STEP_H + AI_ONB_TITLE_H + AI_ONB_SUB_H;
   682→    // 3 карточки-режима (Local / Cloud / SelfHosted — порядок AiOnboardingChoice).
   683→    let mut mode_cards = [[0.0f32; 4]; 3];
   684→    for (i, rect) in mode_cards.iter_mut().enumerate() {
   685→        *rect = [
   686→            x,
   687→            y + i as f32 * (AI_ONB_MODE_H + AI_ONB_MODE_GAP),
   688→            inner_w,
   689→            AI_ONB_MODE_H,
   690→        ];
   691→    }
   692→    let actions_y = y + 3.0 * AI_ONB_MODE_H + 2.0 * AI_ONB_MODE_GAP + 18.0;
   693→    // O2 (Task O / FR-UI-FOOTER): Privacy (left) + Continue (right) migrated
   694→    // to `kit::split_footer_buttons` (Agent L's split-footer kit component).
   695→    // Pattern: Privacy on LEFT (slot.x + inset=0), Continue on RIGHT
   696→    // (slot.right - inset=0 - W). Slot height = AI_ONB_BTN_H (30, no vertical
   697→    // centering — y stays at actions_y). Positions are 1:1 with the previous
   698→    // hand-rolled formula (btn_privacy.x = x = card.x + PAD_X;
   699→    // btn_continue.x = x + inner_w - CONTINUE_W = card.right - PAD_X -
   700→    // CONTINUE_W) — I-1: zero visual jump.
   701→    let actions_slot = UiRect::new(x, actions_y, inner_w, AI_ONB_BTN_H);
   702→    let ai_footer_btns = kit::split_footer_buttons(
   703→        actions_slot,
   704→        &[AI_ONB_BTN_PRIV_W],
   705→        &[AI_ONB_BTN_CONTINUE_W],
   706→        kit::GAP_CONTROLS, // unused — 1 button per side
   707→        0.0,               // no inset — both buttons flush to slot edges
   708→    );
   709→    // btns[0] = (Privacy rect, FooterGroup::Left, 0);
   710→    // btns[1] = (Continue rect, FooterGroup::Right, 0).
   711→    let priv_r = ai_footer_btns[0].0;
   712→    let cont_r = ai_footer_btns[1].0;
   713→    let btn_privacy = [priv_r.x, priv_r.y, priv_r.w, priv_r.h];
   714→    let btn_continue = [cont_r.x, cont_r.y, cont_r.w, cont_r.h];
   715→    let privacy_block = if state.privacy_open {
   716→        [
   717→            x,
   718→            actions_y + AI_ONB_ACTIONS_H + 15.0,
   719→            inner_w,
   720→            AI_ONB_PRIV_H,
   721→        ]
   722→    } else {
   723→        [0.0; 4]
   724→    };
   725→    AiOnboardingLayout {
   726→        card,
   727→        mode_cards,
   728→        btn_privacy,
   729→        btn_continue,
   730→        privacy_block,
   731→    }
   732→}

   734→/// Hit-test кнопки экрана AI-онбординга (карточки-режимы — всегда доступны;
   735→/// «Подробнее» — всегда; «Продолжить» — только если `selected` не None).
   736→pub fn ai_onboarding_button_at(
   737→    layout: &AiOnboardingLayout,
   738→    state: &AiOnboardingState,
   739→    point: [f32; 2],
   740→) -> Option<AiOnboardingButton> {
   741→    use canvas_ui::geometry::UiRect;
   742→    let hit = |rect: [f32; 4]| -> bool {
   743→        let r = UiRect::new(rect[0], rect[1], rect[2], rect[3]);
   744→        point[0] >= r.x && point[0] <= r.x + r.w && point[1] >= r.y && point[1] <= r.y + r.h
   745→    };
   746→    // Карточки-режимы (Local / Cloud / SelfHosted — порядок AiOnboardingChoice).
   747→    let choices = [
   748→        AiOnboardingChoice::Local,
   749→        AiOnboardingChoice::Cloud,
   750→        AiOnboardingChoice::SelfHosted,
   751→    ];
   752→    for (i, &choice) in choices.iter().enumerate() {
   753→        if hit(layout.mode_cards[i]) {
   754→            return Some(AiOnboardingButton::Mode(choice));
   755→        }
   756→    }
   757→    if hit(layout.btn_privacy) {
   758→        return Some(AiOnboardingButton::Privacy);
   759→    }
   760→    // «Продолжить» — только если режим выбран (иначе кнопка disabled).
   761→    if state.selected.is_some() && hit(layout.btn_continue) {
   762→        return Some(AiOnboardingButton::Continue);
   763→    }
   764→    None
   765→}

   767→/// Клик внутри карточки AI-онбординга (но не по кнопкам)? Ввод канваса
   768→/// глотается всей карточкой — клик мимо кнопок не проваливается под оверлей.
