     1→//! FR-053 (U3 PRD-0009, F-7): layout-примитивы — immediate-mode функции
     2→//! раскладки от слота родителя (PRD-0009 §7.4 V-5).
     3→//!
     4→//! Ничего не рисуют и не хранят состояния: каждый кадр потребитель
     5→//! передаёт слот (родительский rect) и список детей — примитивы
     6→//! возвращают вычисленные rect'ы. Интерфейс (слоты/constraints/align)
     7→//! выбран совместимым со слотами taffy (PRD-0009 §7.4: эскалация на
     8→//! taffy в v2 возможна без переписывания потребителей).
     9→//!
    10→//! # Политики переполнения
    11→//!
    12→//! `RowPolicy::Fit` — контент занимает слот; переполнение НЕ маскируется
    13→//! (rect'ы выходят за слот) и ловится линтом/тестом G4 — молчаливый срез
    14→//! (бывший `break`-кламп чипов галереи) становится видимым.
    15→//!
    16→//! `RowPolicy::SqueezeTail` — именованная деградация узкого слота: дети
    17→//! получают `min(desired, остаток)`, хвост сжимается до нулевой ширины.
    18→//! Вырожденные rect'ы невидимы и не пикаются (`UiRect::contains`
    19→//! half-open) — дословная семантика бывшего замыкания `take` what-if
    20→//! бара (FR-017/CR-015), вынесенная в именованную политику.
    21→//!
    22→//! `RowPolicy::Wrap { max_rows }` (FR-062 F-15) — жадная упаковка детей
    23→//! по строкам слота: ребёнок не влез в строку — перенос; высота строки =
    24→//! max высот детей строки; строк сверх `max_rows` НЕ маскируются —
    25→//! выходят за нижний край слота (ловится линтом G4), как `Fit`.
    26→//!
    27→//! # Flex-факторы (FR-062 F-14)
    28→//!
    29→//! `Child.grow` — доля свободного места слота (после фиксированных
    30→//! детей, базовых размеров flex-детей и зазоров), распределяемая
    31→//! пропорционально `grow` (CSS flex-grow при flex-basis = базовый
    32→//! размер). `MainAlign::End` — дети прижаты к концу главной оси.
    33→//! Приоритеты: деградация `SqueezeTail` побеждает распределение (grow
    34→//! игнорируется); grow поглощает свободное место — при Σgrow > 0 и
    35→//! свободном месте `SpaceBetween`/`End` деградируют к `Start`/базовому
    36→//! gap. `0.0` — фиксированный ребёнок (байт-в-байт прежнее поведение).
    37→//!
    38→//! # Measured-дети (FR-062 F-13)
    39→//!
    40→//! [`MeasuredItem`] — размер от контента внутри раскладки: ширина/высота
    41→//! текстового ребёнка берутся из [`crate::measure::TextMeasurer`] (тот
    42→//! же шейпинг, что у рендера) — ручная проводка `width_of →
    43→//! Child::fixed` не нужна. Переполнение НЕ маскируется (наследует
    44→//! политику ряда); молчаливого клампа в остаток слота нет — ограничение
    45→//! ширины задаёт вызов явно (`max_w`), отрисовка усечения — ellipsis
    46→//! потребителя (класс CR-015 защищён по построению: ширина не может
    47→//! быть эвристикой — measurer в сигнатуре).
    48→

   161→/// Горизонтальный контейнер (F-7 `Row{gap, align}` + политика переполнения).
   162→#[derive(Debug, Clone, Copy, PartialEq, Default)]
   163→pub struct Row {
   164→    /// Базовый зазор между детьми (ui px; значения — spacing-scale
   165→    /// `design/tokens/dimensions.json`, источник значений — у вызова).
   166→    pub gap: f32,
   167→    pub main: MainAlign,
   168→    pub cross: CrossAlign,
   169→    pub policy: RowPolicy,
   170→}
   171→
   172→impl Row {
   173→    /// Раскладка детей в слоте: возвращает rect'ы (параллельно `items`).
   174→    /// FR-068 W1: делегирует в [`default_backend`] (NativeBackend — поведение
   175→    /// байт-в-байт прежнее; §Контракт-1 — сигнатура стабильна до W3).
   176→    pub fn lay_out(&self, slot: UiRect, items: &[Child]) -> Vec<UiRect> {
   177→        default_backend().lay_out_row(*self, slot, items)
   178→    }
   179→
   180→    /// Раскладка детей в слоте через ЯВНО выбранный backend (FR-068 W1,
   181→    /// ADR-0014 «потребитель выбирает backend осознанно»): pilot-поверхности
   182→    /// передают [`pilot_backend`], остальные — [`default_backend`].
   183→    pub fn lay_out_with(
   184→        &self,
   185→        backend: &dyn LayoutBackend,
   186→        slot: UiRect,
   187→        items: &[Child],
   188→    ) -> Vec<UiRect> {
   189→        backend.lay_out_row(*self, slot, items)
   190→    }
   191→
   192→    /// `Fit`: дети подряд с зазором; свободное место — flex-детям по
   193→    /// `grow` (FR-062 F-14), остаток — по `main`.
   194→    fn fit(&self, slot: UiRect, items: &[Child]) -> Vec<UiRect> {
   195→        let sizes = self.resolve_grow(slot.w, items);
   196→        let n = items.len();
   197→        let total: f32 = sizes.iter().sum();
   198→        let gaps_total = self.gap * n.saturating_sub(1) as f32;
   199→        let extra = (slot.w - total - gaps_total).max(0.0);
   200→        let gap = match self.main {
   201→            MainAlign::Start => self.gap,
   202→            // SpaceBetween: при n>1 свободное место добавляется к зазорам
   203→            MainAlign::SpaceBetween if n > 1 => self.gap + extra / (n - 1) as f32,
   204→            MainAlign::SpaceBetween => self.gap,
   205→            MainAlign::End => self.gap,
   206→        };
   207→        let mut x = match self.main {
   208→            // End: дети прижаты к правому краю (свободное место — слева);
   209→            // при grow свободное место уже съедено — x = slot.x
   210→            MainAlign::End => (slot.right() - total - gaps_total).max(slot.x),
   211→            _ => slot.x,
   212→        };
   213→        items
   214→            .iter()
   215→            .zip(&sizes)
   216→            .map(|(c, &w)| {
   217→                let rect = UiRect::new(x, self.cross_y(slot, c.h), w, c.h);
   218→                x += w + gap;
   219→                rect
   220→            })
   221→            .collect()
   222→    }
   223→
   224→    /// Распределение свободного места слота по `grow` (FR-062 F-14):
   225→    /// свободное = slot.w − Σ(базовые ширины) − Σ(зазоры); flex-дети
   226→    /// получают базу + долю свободного (пропорционально grow). Ширины
   227→    /// фиксированных детей не меняются. Σgrow = 0 или свободного нет —
   228→    /// базовые ширины (байт-в-байт прежнее поведение).
   229→    fn resolve_grow(&self, slot_w: f32, items: &[Child]) -> Vec<f32> {
   230→        let mut sizes: Vec<f32> = items.iter().map(|c| c.w).collect();
   231→        let grow_total: f32 = items.iter().map(|c| c.grow).sum();
   232→        if grow_total <= 0.0 {
   233→            return sizes;
   234→        }
   235→        let basis: f32 = sizes.iter().sum();
   236→        let gaps_total = self.gap * items.len().saturating_sub(1) as f32;
   237→        let free = (slot_w - basis - gaps_total).max(0.0);
   238→        if free <= 0.0 {
   239→            return sizes;
   240→        }
   241→        for (size, child) in sizes.iter_mut().zip(items) {
   242→            if child.grow > 0.0 {
   243→                *size += free * child.grow / grow_total;
   244→            }
   245→        }
   246→        sizes
   247→    }
   248→
   249→    /// `SqueezeTail`: каждый ребёнок получает `min(desired, остаток)`;
   250→    /// хвост сжимается до нуля, за правый край слота ничего не выходит.
   251→    /// Деградация приоритетна: `grow` игнорируется (FR-062 F-14).
   252→    fn squeeze_tail(&self, slot: UiRect, items: &[Child]) -> Vec<UiRect> {
   253→        let mut x = slot.x;
   254→        let limit = slot.right();
   255→        items
   256→            .iter()
   257→            .map(|c| {
   258→                let remaining = (limit - x).max(0.0);
   259→                let w = c.w.min(remaining);
   260→                let rect = UiRect::new(x.min(limit), self.cross_y(slot, c.h), w, c.h);
   261→                x += w + self.gap;
   262→                rect
   263→            })
   264→            .collect()
   265→    }
   266→
   267→    /// Жадная упаковка по строкам (FR-062 F-15, уточнение контракта:
   268→    /// `max_rows` поле убрано — число видимых строк определяет ВЫСОТА
   269→    /// слота; строки ниже слота не маскируются — естественно выходят за
   270→    /// нижний край, что и ловит линт G4; без мёртвого параметра).
   271→    /// Высота строки = max высот детей строки; зазоры: `gap` — по
   272→    /// горизонтали, тот же `gap` — между строками; поперечное
   273→    /// выравнивание — внутри строки (по её высоте).
   274→    fn wrap(&self, slot: UiRect, items: &[Child]) -> Vec<UiRect> {
   275→        let mut rects = vec![UiRect::default(); items.len()];
   276→        let mut row: Vec<(usize, &Child)> = Vec::new();
   277→        let mut row_w = 0.0;
   278→        let mut row_h = 0.0;
   279→        let mut row_y = slot.y;
   280→        let mut flush =
   281→            |row: &mut Vec<(usize, &Child)>, row_w: &mut f32, row_h: &mut f32, row_y: &mut f32| {
   282→                let mut x = slot.x;
   283→                let h = *row_h;
   284→                for (i, c) in row.drain(..) {
   285→                    let y = match self.cross {
   286→                        CrossAlign::Start => *row_y,
   287→                        // поперечное выравнивание внутри строки (по её высоте)
   288→                        CrossAlign::Center => *row_y + (h - c.h).max(0.0) / 2.0,
   289→                        CrossAlign::End => *row_y + (h - c.h).max(0.0),
   290→                    };
   291→                    rects[i] = UiRect::new(x, y, c.w, c.h);
   292→                    x += c.w + self.gap;
   293→                }
   294→                *row_y += h + self.gap;
   295→                *row_w = 0.0;
   296→                *row_h = 0.0;
   297→            };
   298→        for (i, c) in items.iter().enumerate() {
   299→            let next_w = if row.is_empty() {
   300→                c.w
   301→            } else {
   302→                row_w + self.gap + c.w
   303→            };
   304→            // ребёнок шире слота — строка из одного (переполнение вправо
   305→            // видно, как в Fit); непоместившийся в строку — перенос
   306→            if !row.is_empty() && next_w > slot.w + f32::EPSILON {
   307→                flush(&mut row, &mut row_w, &mut row_h, &mut row_y);
   308→            }
   309→            row.push((i, c));
   310→            row_w = if row.len() == 1 {
   311→                c.w
   312→            } else {
   313→                row_w + self.gap + c.w
   314→            };
   315→            row_h = row_h.max(c.h);
   316→        }
   317→        if !row.is_empty() {
   318→            flush(&mut row, &mut row_w, &mut row_h, &mut row_y);
   319→        }
   320→        rects
   321→    }
   322→
   323→    /// FR-062 F-13: раскладка measured-детей (размер от контента —
   324→    /// через [`TextMeasurer`]); та же политика, что [`Row::lay_out`]
   325→    /// (в т.ч. `Wrap`); flex-факторы у measured-детей не задаются
   326→    /// (сигнатура F-13 минимальна — фиксированные размеры + текст).
   327→    pub fn lay_out_measured(
   328→        &self,
   329→        slot: UiRect,
   330→        items: &[MeasuredItem],
   331→        m: &mut TextMeasurer,
   332→        fs: &mut cosmic_text::FontSystem,
   333→        family: &str,
   334→        size: f32,
   335→    ) -> Vec<UiRect> {
   336→        default_backend().lay_out_measured(*self, slot, items, m, fs, family, size)
   337→    }

   339→    /// FR-062 F-13 через ЯВНО выбранный backend (FR-068 W1): см.
   340→    /// [`Row::lay_out_with`]. Замер текста ([`MeasuredItem::resolve`], тот же
   341→    /// [`TextMeasurer`]) выполняется ДО адаптера у обоих backend'ов — шейпинг
   342→    /// идентичен, golden-оракул побитовый.
   343→    #[allow(clippy::too_many_arguments)] // плоский контракт F-13 (заморожен §Контракт-1)
   344→    pub fn lay_out_measured_with(
   345→        &self,
   346→        backend: &dyn LayoutBackend,
   347→        slot: UiRect,
   348→        items: &[MeasuredItem],
   349→        m: &mut TextMeasurer,
   350→        fs: &mut cosmic_text::FontSystem,
   351→        family: &str,
   352→        size: f32,
   353→    ) -> Vec<UiRect> {
   354→        backend.lay_out_measured(*self, slot, items, m, fs, family, size)
   355→    }
   356→
   357→    fn cross_y(&self, slot: UiRect, h: f32) -> f32 {
   358→        match self.cross {
   359→            CrossAlign::Start => slot.y,
   360→            CrossAlign::Center => slot.y + (slot.h - h).max(0.0) / 2.0,
   361→            CrossAlign::End => slot.y + (slot.h - h).max(0.0),
   362→        }
   363→    }
   364→}

   394→
   395→    /// Распределение свободного места слота по `grow` по высоте
   396→    /// (FR-062 F-14; симметрично [`Row::resolve_grow`]).
   397→    fn resolve_grow(&self, slot_h: f32, items: &[Child]) -> Vec<f32> {
   398→        let mut sizes: Vec<f32> = items.iter().map(|c| c.h).collect();
   399→        let grow_total: f32 = items.iter().map(|c| c.grow).sum();
   400→        if grow_total <= 0.0 {
   401→            return sizes;
   402→        }
   403→        let basis: f32 = sizes.iter().sum();
   404→        let gaps_total = self.gap * items.len().saturating_sub(1) as f32;
   405→        let free = (slot_h - basis - gaps_total).max(0.0);
   406→        if free <= 0.0 {
   407→            return sizes;
   408→        }
   409→        for (size, child) in sizes.iter_mut().zip(items) {
   410→            if child.grow > 0.0 {
   411→                *size += free * child.grow / grow_total;
   412→            }
   413→        }
   414→        sizes
   415→    }
   416→
   417→    fn cross_x(&self, slot: UiRect, w: f32) -> f32 {
   418→        match self.cross {
   419→            CrossAlign::Start => slot.x,
   420→            CrossAlign::Center => slot.x + (slot.w - w).max(0.0) / 2.0,
   421→            CrossAlign::End => slot.x + (slot.w - w).max(0.0),
   422→        }
   423→    }

   469→/// Ребёнок measured-раскладки (FR-062 F-13): размер от контента.
   470→///
   471→/// Ширина/высота текстового ребёнка берутся из [`TextMeasurer`] (тот же
   472→/// шейпинг, что у screen-текстов рендера); ограничение ширины — только
   473→/// явное (`max_w`) — молчаливого клампа в остаток слота нет (G5):
   474→/// усечение подписи — ellipsis потребителя по той же измеренной ширине.
   475→#[derive(Debug, Clone, Copy, PartialEq)]
   476→pub enum MeasuredItem<'a> {
   477→    /// Фиксированный размер (эквивалент [`Child::fixed`]) — оракул
   478→    /// эквивалентности с ручной проводкой `width_of → Child::fixed`.
   479→    Fixed { w: f32, h: f32 },
   480→    /// Размер от текста: ширина = `width_of` + `pad_x` (кламп в `max_w`,
   481→    /// если задан, и подъём до `min_w`); высота = явная `h` (дизайн-константа
   482→    /// потребителя, например `CHIP_HEIGHT`) или измеренная (строки · кегль ·
   483→    /// `SCREEN_LINE_FACTOR`).
   484→    Text {
   485→        text: &'a str,
   486→        /// Явный потолок ширины (ellipsis — решение потребителя).
   487→        max_w: Option<f32>,
   488→        /// Пол ширины (например, минимальная ширина кнопки) — уже текста
   489→        /// с падом не будет.
   490→        min_w: f32,
   491→        /// Суммарный горизонтальный пад (лево+право), добавляемый к
   492→        /// измеренной ширине текста (W3.3 — пад-семантика F-13, названное
   493→        /// условие миграции проводок «width_of → размер ребёнка»:
   494→        /// чип/кнопка = один `Text { pad_x }` без ручного замера;
   495→        /// бит-в-бит эквивалент `Fixed { w: width_of + pad_x }`;
   496→        /// `pad_x: 0.0` — прежняя семантика дословно).
   497→        pad_x: f32,
   498→        /// Явная высота (дизайн-константа чипа/кнопки — `Some(h)`);
   499→        /// `None` — измеренная (прежнее поведение дословно). Нужно там,
   500→        /// где константа высоты ≠ измеренной (чип whatif:
   501→        /// `CHIP_HEIGHT 26` ≠ `13·1.3 = 16.9`) — иначе миграция меняет
   502→        /// визуал (нарушение I-1).
   503→        h: Option<f32>,
   504→    },
   505→    /// Распорка (эквивалент [`Child::spacer`]).
   506→    Spacer(f32),
   507→}
   508→
   509→impl MeasuredItem<'_> {
   510→    /// Разрешить в фиксированного [`Child`] измерениями (точка интеграции
   511→    /// TextMeasurer; ширины/высоты — реальные, не эвристика).
   512→    pub fn resolve(
   513→        &self,
   514→        m: &mut TextMeasurer,
   515→        fs: &mut cosmic_text::FontSystem,
   516→        family: &str,
   517→        size: f32,
   518→    ) -> Child {
   519→        match *self {
   520→            MeasuredItem::Fixed { w, h } => Child::fixed(w, h),
   521→            MeasuredItem::Spacer(len) => Child::spacer(len),
   522→            MeasuredItem::Text {
   523→                text,
   524→                max_w,
   525→                min_w,
   526→                pad_x,
   527→                h,
   528→            } => {
   529→                let spec = crate::measure::TextSpec {
   530→                    text,
   531→                    family,
   532→                    size,
   533→                    max_width: f32::INFINITY,
   534→                    // UI-раскладка — пропорциональный sans (Weight 500,
   535→                    // паритет sans_attrs рендера; см. TextSpec::weight).
   536→                    weight: cosmic_text::Weight::MEDIUM,
   537→                };
   538→                let measured = m.measure(fs, &spec);
   539→                let mut w = (measured.width + pad_x).max(min_w);
   540→                if let Some(max) = max_w {
   541→                    w = w.min(max.max(min_w));
   542→                }
   543→                Child::fixed(w, h.unwrap_or(measured.height))
   544→            }
   545→        }
   546→    }
   547→}

   591→/// Разместить блок фиксированного размера в слоте (F-7 `Stack{align}`):
   592→/// панель галереи по центру вьюпорта, empty-карточка и т.п.
   593→pub fn stack(slot: UiRect, size: UiVec2, h: HAlign, v: VAlign) -> UiRect {
   594→    let x = match h {
   595→        HAlign::Start => slot.x,
   596→        HAlign::Center => slot.x + (slot.w - size.x).max(0.0) / 2.0,
   597→        HAlign::End => slot.x + (slot.w - size.x).max(0.0),
   598→    };
   599→    let y = match v {
   600→        VAlign::Start => slot.y,
   601→        VAlign::Center => slot.y + (slot.h - size.y).max(0.0) / 2.0,
   602→        VAlign::End => slot.y + (slot.h - size.y).max(0.0),
   603→    };
   604→    UiRect::new(x, y, size.x.max(0.0), size.y.max(0.0))
   605→}

   607→/// Кламп размера в min/max границы (F-7 `Constrain(min/max)`); нижняя
   608→/// граница приоритетна над верхней (`max` срезается до `min` —
   609→/// противоречивые границы дают `min`).
   610→pub fn constrain(min: UiVec2, max: UiVec2, desired: UiVec2) -> UiVec2 {
   611→    let lo_x = min.x.max(0.0);
   612→    let lo_y = min.y.max(0.0);
   613→    let hi_x = max.x.max(lo_x);
   614→    let hi_y = max.y.max(lo_y);
   615→    UiVec2::new(desired.x.clamp(lo_x, hi_x), desired.y.clamp(lo_y, hi_y))
   616→}

   772→impl LayoutBackend for NativeBackend {
   773→    fn features(&self) -> LayoutFeatures {
   774→        // Native: grow/basis/wrap/grid/measured — есть; CSS flex_shrink —
   775→        // нет (`SqueezeTail` — именованная деградация, не flex_shrink,
   776→        // §Контракт-4); percent/aspect/overflow/sticky — нет (территория
   777→        // расширенной сцены [`SceneNode`], в W1 — TaffyBackend).

   779→            .union(LayoutFeatures::FLEX_BASIS)
   780→            .union(LayoutFeatures::FLEX_WRAP)
   781→            .union(LayoutFeatures::GRID_2D)
   782→            .union(LayoutFeatures::AUTO_SIZE)

   785→    fn lay_out_row(&self, row: Row, slot: UiRect, items: &[Child]) -> Vec<UiRect> {
   786→        match row.policy {
   787→            RowPolicy::Fit => row.fit(slot, items),
   788→            RowPolicy::SqueezeTail => row.squeeze_tail(slot, items),
   789→            RowPolicy::Wrap => row.wrap(slot, items),
   790→        }
   791→    }
   792→
   793→    fn lay_out_column(&self, column: Column, slot: UiRect, items: &[Child]) -> Vec<UiRect> {
   794→        // Тело перенесено из `Column::lay_out` 1:1 (FR-062 F-14).
   795→        let sizes = column.resolve_grow(slot.h, items);
   796→        let n = items.len();
   797→        let total: f32 = sizes.iter().sum();
   798→        let gaps_total = column.gap * n.saturating_sub(1) as f32;
   799→        let extra = (slot.h - total - gaps_total).max(0.0);
   800→        let gap = match column.main {
   801→            MainAlign::Start => column.gap,
   802→            MainAlign::SpaceBetween if n > 1 => column.gap + extra / (n - 1) as f32,
   803→            MainAlign::SpaceBetween => column.gap,
   804→            MainAlign::End => column.gap,
   805→        };
   806→        let mut y = match column.main {
   807→            MainAlign::End => (slot.bottom() - total - gaps_total).max(slot.y),
   808→            _ => slot.y,
   809→        };
   810→        items
   811→            .iter()
   812→            .zip(&sizes)
   813→            .map(|(c, &h)| {
   814→                let rect = UiRect::new(column.cross_x(slot, c.w), y, c.w, h);
   815→                y += h + gap;
   816→                rect
   817→            })
   818→            .collect()
   819→    }

   821→    fn lay_out_measured(
   822→        &self,
   823→        row: Row,
   824→        slot: UiRect,
   825→        items: &[MeasuredItem],
   826→        m: &mut TextMeasurer,
   827→        fs: &mut cosmic_text::FontSystem,
   828→        family: &str,
   829→        size: f32,
   830→    ) -> Vec<UiRect> {
   831→        // Тот же resolve, что до W1 (`MeasuredItem::resolve` — единая точка
   832→        // замера): backend получает уже фиксированные Child'ы.
   833→        let children: Vec<Child> = items
   834→            .iter()
   835→            .map(|item| item.resolve(m, fs, family, size))
   836→            .collect();
   837→        self.lay_out_row(row, slot, &children)
   838→    }

   840→    fn lay_out_grid(
   841→        &self,
   842→        slot: UiRect,
   843→        cols: &[f32],
   844→        rows: usize,
   845→        row_h: f32,
   846→        gap: UiVec2,
   847→    ) -> Vec<UiRect> {
   848→        // Тело перенесено из `grid_cells` 1:1 (FR-062 F-16).
   849→        let mut rects = Vec::with_capacity(cols.len().saturating_mul(rows));
   850→        let row_h = row_h.max(0.0);
   851→        for r in 0..rows {
   852→            let y = slot.y + r as f32 * (row_h + gap.y.max(0.0));
   853→            let mut x = slot.x;
   854→            for &w in cols {
   855→                let w = w.max(0.0);
   856→                rects.push(UiRect::new(x, y, w, row_h));
   857→                x += w + gap.x.max(0.0);
   858→            }
   859→        }
   860→        rects
   861→    }
   862→}
