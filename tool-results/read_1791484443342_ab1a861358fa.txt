   158→/// flyout справа). Поле ввода — своя лёгкая модель (однострочная,
   159→/// как `SearchInput`), НЕ `EditingSession`.
   160→///
   161→/// Ревизия FR-025: `open` — развёрнут ли док (по умолчанию false — свёрнута
   162→/// в полосу категорий); `focused` — принимает ли панель клавиатуру (фокус в
   163→/// поиске: Ctrl+P или клик по полю фильтра). Клик по канвасу мимо панели
   164→/// фокус снимает, док не закрывает.
   165→#[derive(Debug, Clone, PartialEq)]
   166→pub struct TemplatePanel {
   167→    pub open: bool,
   168→    /// Клавиатурный фокус: true — клавиши уходят в панель (фильтр, стрелки,
   169→    /// Enter, Esc), false — в канвас (панель видна, но не перехватывает).
   170→    pub focused: bool,
   171→    /// Строка фильтра (подстрока без учёта регистра по имени/описанию/id).
   172→    pub filter: String,
   173→    /// Байтовая позиция каретки в `filter`.
   174→    pub cursor: usize,
   175→    /// Фильтр категории (клик по чипу); None — все категории.
   176→    pub category: Option<String>,
   177→    /// Выбранная строка-шаблон (ординал среди [`PanelRow::Template`]).
   178→    pub selected: usize,
   179→    /// Первая видимая строка (индекс в векторе [`PanelRow`], прокрутка).
   180→    pub scroll_top: usize,
   181→}

   183→impl TemplatePanel {
   184→    /// Новая панель — СВЁРНУТАЯ (ревизия FR-025: палитра примарно свёрнута,
   185→    /// полоса категорий по центру слева; развёрнутый док — по Ctrl+P),
   186→    /// клавиатурный фокус снят.
   187→    pub fn new() -> Self {
   188→        Self {
   189→            open: false,
   190→            focused: false,
   191→            filter: String::new(),
   192→            cursor: 0,
   193→            category: None,
   194→            selected: 0,
   195→            scroll_top: 0,
   196→        }
   197→    }
   198→
   199→    /// Развернуть док с клавиатурным фокусом в поиске (сброс фильтров —
   200→    /// каждый вызов с чистого листа). Вызывается по Ctrl+P.
   201→    pub fn open(&mut self) {
   202→        self.open = true;
   203→        self.focused = true;
   204→        self.filter.clear();
   205→        self.cursor = 0;
   206→        self.category = None;
   207→        self.selected = 0;
   208→        self.scroll_top = 0;
   209→    }
   210→
   211→    /// Свернуть док в полосу-ручку (FR-025; Esc). Фокус снимается.
   212→    pub fn close(&mut self) {
   213→        self.open = false;
   214→        self.focused = false;
   215→    }
   216→
   217→    /// Развернуть док без сброса фильтров и без клавиатурного фокуса
   218→    /// (клик по полосе-ручке свёрнутого дока).
   219→    pub fn expand(&mut self) {
   220→        self.open = true;
   221→    }
   222→
   223→    /// Клавиатурный фокус в поиск (Ctrl+P по развёрнутому доку: фильтр
   224→    /// сохраняется, каретка — в конец строки).
   225→    pub fn focus_search(&mut self) {
   226→        self.focused = true;
   227→        self.cursor = self.filter.len();
   228→    }
   229→
   230→    /// Снять клавиатурный фокус (клик по канвасу): док остаётся развёрнут.
   231→    pub fn unfocus(&mut self) {
   232→        self.focused = false;
   233→    }
   234→
   235→    /// Вставка строки в каретку (печать символа, IME).
   236→    pub fn insert_str(&mut self, text: &str) {
   237→        let byte = self.cursor.min(self.filter.len());
   238→        self.filter.insert_str(byte, text);
   239→        self.cursor += text.len();
   240→    }
   241→
   242→    /// Backspace: удалить символ перед кареткой (true — было изменение).
   243→    pub fn backspace(&mut self) -> bool {
   244→        if self.cursor == 0 {
   245→            return false;
   246→        }
   247→        let byte = self.cursor.min(self.filter.len());
   248→        let prev = self.filter[..byte]
   249→            .char_indices()
   250→            .next_back()
   251→            .map(|(i, _)| i)
   252→            .unwrap_or(0);
   253→        self.filter.replace_range(prev..byte, "");
   254→        self.cursor = prev;
   255→        true
   256→    }
   257→
   258→    pub fn move_left(&mut self) {
   259→        let byte = self.cursor.min(self.filter.len());
   260→        if let Some(i) = self.filter[..byte]
   261→            .char_indices()
   262→            .next_back()
   263→            .map(|(i, _)| i)
   264→        {
   265→            self.cursor = i;
   266→        }
   267→    }
   268→
   269→    pub fn move_right(&mut self) {
   270→        let byte = self.cursor.min(self.filter.len());
   271→        if let Some(i) = self.filter[byte..]
   272→            .char_indices()
   273→            .nth(1)
   274→            .map(|(i, _)| byte + i)
   275→        {
   276→            self.cursor = i;
   277→        }
   278→    }
   279→
   280→    /// Сдвиг выделения (стрелки): нумеруются ТОЛЬКО строки-шаблоны
   281→    /// (секции — заголовки, не цели); true — было изменение. Прокрутка
   282→    /// следует за выделением по ИЗМЕРЕННОЙ высоте окна строк
   283→    /// `rows_area_h` (W-e, аудит ui-kit §8 п.11: прежнее константное окно
   284→    /// `SCROLL_WINDOW` не зависело от высоты панели и могло оставить
   285→    /// выделение за пределами экрана): вверх — строка становится первой
   286→    /// (с заголовком секции над ней), вниз — выделение прижимается к
   287→    /// нижней границе окна (максимум контекста сверху).
   288→    pub fn move_selection(&mut self, delta: i32, rows: &[PanelRow], rows_area_h: f32) -> bool {
   289→        let total = template_row_count(rows);
   290→        if total == 0 {
   291→            return false;
   292→        }
   293→        let next = (self.selected as i32 + delta).clamp(0, total as i32 - 1);
   294→        if next == self.selected as i32 {
   295→            return false;
   296→        }
   297→        self.selected = next as usize;
   298→        let Some(row_idx) = row_of_ordinal(rows, self.selected) else {
   299→            return true;
   300→        };
   301→        if row_idx < self.scroll_top {
   302→            // Заголовок секции над строкой — показать и его
   303→            self.scroll_top = if row_idx > 0 && matches!(rows[row_idx - 1], PanelRow::Section(_)) {
   304→                row_idx - 1
   305→            } else {
   306→                row_idx
   307→            };
   308→        } else {
   309→            // Строка ниже окна? Видимость от scroll_top: сумма шагов
   310→            // scroll_top..row_idx + карточка выделенной строки ≤ высота окна.
   311→            let steps_to_row: f32 = rows[self.scroll_top..row_idx].iter().map(row_step).sum();
   312→            if steps_to_row + row_card_h(&rows[row_idx]) > rows_area_h {
   313→                // Прижать выделение к нижней границе: наибольшее s ≤ row_idx,
   314→                // при котором выделенная строка целиком видна (хвост
   315→                // s..row_idx помещается в окно).
   316→                let mut s = row_idx;
   317→                let mut tail = row_card_h(&rows[row_idx]);
   318→                while s > 0 {
   319→                    let prev = row_step(&rows[s - 1]);
   320→                    if tail + prev > rows_area_h {
   321→                        break;
   322→                    }
   323→                    tail += prev;
   324→                    s -= 1;
   325→                }
   326→                self.scroll_top = s;
   327→            }
   328→        }
   329→        true
   330→    }
   331→
   332→    /// FR-082: прокрутка списка развёрнутой панели колесом — кламп к
   333→    /// `[0, max_scroll]` (max_scroll считает `panel_layout` — «приклейка
   334→    /// хвоста» по измеренной высоте окна строк; W-e: кэп
   335→    /// `MAX_VISIBLE_ROWS` удалён — предел задаёт только высота). Знак —
   336→    /// как у списков: колесо от себя (LineDelta y < 0) увеличивает
   337→    /// scroll_top; шаг строк — `PANEL_WHEEL_LINES`.
   338→    pub fn scroll_by(&mut self, delta: i32, max_scroll: usize) {
   339→        let cur = self.scroll_top as i32;
   340→        self.scroll_top = (cur + delta).clamp(0, max_scroll as i32) as usize;
   341→    }
   342→}

   351→/// FR-025: нажатие на строку шаблона палитры — кандидат в drag: без
   352→/// движения порога это клик (вставка в центр viewport), с движением —
   353→/// drag с ghost-превью (вставка в точку курсора). Отпускание решает.
   354→#[derive(Debug, Clone, PartialEq)]
   355→pub struct PanelDrag {
   356→    /// Индекс шаблона в `registry.list()`.
   357→    pub index: usize,
   358→    /// Точка нажатия (screen, логические px).
   359→    pub press: Vec2,
   360→    /// Порог пройден — идёт drag (ghost-превью, вставка в точку курсора).
   361→    pub active: bool,
   362→}
   363→
   364→impl PanelDrag {
   365→    /// Порог перевода нажатия в drag, логические px (как рамка выделения).
   366→    pub const THRESHOLD: f32 = 4.0;
   367→
   368→    /// Обновление по позиции курсора: true — только что перешло в drag.
   369→    pub fn update(&mut self, cursor: Vec2) -> bool {
   370→        if self.active {
   371→            return false;
   372→        }
   373→        let moved = (cursor[0] - self.press[0]).abs() > Self::THRESHOLD
   374→            || (cursor[1] - self.press[1]).abs() > Self::THRESHOLD;
   375→        if moved {
   376→            self.active = true;
   377→        }
   378→        moved
   379→    }
   380→}

   382→/// Число строк-шаблонов в наборе строк панели (секции не считаются).
   383→pub fn template_row_count(rows: &[PanelRow]) -> usize {
   384→    rows.iter()
   385→        .filter(|row| matches!(row, PanelRow::Template(_)))
   386→        .count()
   387→}
   388→
   389→/// Позиция строки-шаблона по ординалу выделения (k-я строка-шаблон →
   390→/// индекс в `rows`). Секции пропускаются.
   391→pub fn row_of_ordinal(rows: &[PanelRow], ordinal: usize) -> Option<usize> {
   392→    let mut seen = 0_usize;
   393→    for (index, row) in rows.iter().enumerate() {
   394→        if matches!(row, PanelRow::Template(_)) {
   395→            if seen == ordinal {
   396→                return Some(index);
   397→            }
   398→            seen += 1;
   399→        }
   400→    }
   401→    None
   402→}

   404→/// Строки панели (FR-024): при пустом фильтре и без чипа категории —
   405→/// группировка по категориям с заголовками секций (порядок реестра,
   406→/// паттерн Miro Template picker); при поиске/фильтре — плоский список
   407→/// совпадений (секции не имеют смысла в результатах поиска).
   408→///
   409→/// FR-040 v2: фильтр — двуязычный: совпадение по `name_en` ИЛИ `name_ru` ИЛИ
   410→/// `description` ИЛИ `description_en` ИЛИ `id`. Английский пользователь,
   411→/// набирающий «load», находит «Load Balancer» (`name_en`) и шаблоны с
   412→/// этим словом в `description_en`; русский — «баланс» — находит по
   413→/// `name_ru`/`description`. Снапшот имени на момент вставки берётся
   414→/// по активному языку (`TemplateManifest::display_name(language)` в
   415→/// рендере карточки и `instantiate_with_language`).
   416→///
   417→/// FR-087 (роль/класс): `visible` — allow-list raw-токенов категорий
   418→/// ([`crate::App::visible_template_categories`]: ручной фильтр настроек >
   419→/// дефолт роли > все). Поиск ведётся ТОЛЬКО по видимым категориям —
   420→/// скрытые роли не «всплывают» через строку поиска. Индексы строк —
   421→/// как и прежде, позиция в `registry.list()` (полный реестр: вставка
   422→/// и `find` не зависят от фильтра).
   423→pub fn panel_rows(
   424→    registry: &TemplateRegistry,
   425→    panel: &TemplatePanel,
   426→    language: canvas_core::Language,
   427→    visible: &[String],
   428→) -> Vec<PanelRow> {
   429→    let query = panel.filter.to_lowercase();
   430→    let matches = |manifest: &TemplateManifest| -> bool {
   431→        if !visible
   432→            .iter()
   433→            .any(|category| category == &manifest.category)
   434→        {
   435→            return false;
   436→        }
   437→        if let Some(category) = &panel.category {
   438→            if &manifest.category != category {
   439→                return false;
   440→            }
   441→        }
   442→        query.is_empty()
   443→            || manifest.name.to_lowercase().contains(&query)
   444→            || manifest
   445→                .name_ru
   446→                .as_deref()
   447→                .is_some_and(|n| n.to_lowercase().contains(&query))
   448→            || manifest.description.to_lowercase().contains(&query)
   449→            || manifest
   450→                .description_en
   451→                .as_deref()
   452→                .is_some_and(|d| d.to_lowercase().contains(&query))
   453→            || manifest.id.to_lowercase().contains(&query)
   454→    };
   455→    // (аргумент language — индикатор намерения: фильтр билингвален и
   456→    //  работает для любого языка; в будущем — упреждающее ранжирование)
   457→    let _ = language;
   458→    let grouped = panel.filter.is_empty() && panel.category.is_none();
   459→    if !grouped {
   460→        return registry
   461→            .list()
   462→            .iter()
   463→            .enumerate()
   464→            .filter(|(_, manifest)| matches(manifest))
   465→            .map(|(index, _)| PanelRow::Template(index))
   466→            .collect();
   467→    }
   468→    let mut rows = Vec::new();
   469→    for category in registry.categories() {
   470→        // FR-087: секции скрытых категорий не показываются вовсе
   471→        if !visible.iter().any(|visible| visible == category) {
   472→            continue;
   473→        }
   474→        let indexes: Vec<usize> = registry
   475→            .list()
   476→            .iter()
   477→            .enumerate()
   478→            .filter(|(_, manifest)| manifest.category == *category)
   479→            .map(|(index, _)| index)
   480→            .collect();
   481→        if indexes.is_empty() {
   482→            continue;
   483→        }
   484→        rows.push(PanelRow::Section((*category).to_owned()));
   485→        rows.extend(indexes.into_iter().map(PanelRow::Template));
   486→    }
   487→    rows
   488→}

   490→/// Геометрия панели на кадр (логические px) — левый док во всю высоту

   497→    pub input_rect: [f32; 4],

   508→    /// Кнопка сворачивания дока в шапке (FR-025; 22×22 у правого края

   511→    /// W-e: применённая первая видимая строка (индекс в полном наборе

   517→    /// W-e: окно строк — измеренная доступная высота панели (между чипами

   536→/// Ширина полосы категорий свёрнутой палитры (ревизия FR-025, 2026-09-16),
   537→/// логические px. Ширина полосы — максимум ширин чипов категорий, но не
   538→/// уже этой границы (впритык под короткие имена не сжимаем).
   539→pub const STRIP_MIN_W: f32 = 64.0;
   540→/// Вертикальный внутренний паддинг полосы (над первой строкой и под шевроном).
   541→pub const STRIP_PAD_V: f32 = 6.0;
   542→/// Горизонтальный внутренний паддинг полосы.
   543→pub const STRIP_PAD_H: f32 = 4.0;
   544→/// Запас ширины полосы под счётчик шаблонов («backend · 10») рядом с именем.
   545→pub const STRIP_COUNT_SLACK: f32 = 28.0;
   546→/// Зазор между полосой категорий и flyout, логические px.
   547→pub const FLYOUT_GAP: f32 = 6.0;
   548→/// Минимальная ширина flyout, логические px.
   549→pub const FLYOUT_MIN_W: f32 = 260.0;
   550→/// Вертикальный внутренний паддинг flyout (над первой/под последней строкой).
   551→pub const FLYOUT_PAD_V: f32 = 6.0;
   552→/// Горизонтальный внутренний паддинг flyout.
   553→pub const FLYOUT_PAD_H: f32 = 8.0;
   554→
   555→/// Ширина flyout: уже дока на 40 px, но не уже [`FLYOUT_MIN_W`].
   556→pub fn flyout_width() -> f32 {
   557→    (PANEL_WIDTH - 40.0).max(FLYOUT_MIN_W)
   558→}

   560→/// Геометрия свёрнутой палитры (ревизия FR-025): вертикальная полоса
   561→/// категорий у левого края, центрированная по вертикали окна. Все rect'ы —
   562→/// `[x, y, w, h]` (конвенция `point_in_rect`). Единый источник для рендера
   563→/// и hit-test: что нарисовано, по тому и клик/наведение.
   564→#[derive(Debug, Clone, PartialEq)]
   565→pub struct StripLayout {
   566→    /// Прямоугольник полосы целиком (включая шеврон).
   567→    pub rect: [f32; 4],
   568→    /// Строки категорий: (rect, имя категории), порядок — порядок реестра.
   569→    pub rows: Vec<([f32; 4], String)>,
   570→    /// Шеврон «развернуть док» — строка внизу полосы.
   571→    pub chevron_rect: [f32; 4],
   572→}
   573→
   574→/// Раскладка свёрнутой полосы категорий: ширина — по самому длинному чипу
   575→/// (+ запас под счётчик), строки по [`CATEGORY_ROW_H`], полоса центрирована
   576→/// по вертикали окна и клампится отступом [`PANEL_TOP_MARGIN`] сверху/снизу
   577→/// (в маленьком окне не уходит за края выше верхнего отступа).
   578→pub fn dock_strip_layout(
   579→    categories: &[String],
   580→    window_h: f32,
   581→    m: &mut TextMeasurer,
   582→    fs: &mut cosmic_text::FontSystem,
   583→) -> StripLayout {
   584→    // Ширина полосы — КОНТЕЙЕРНЫЙ замер (первый проход): максимум ширин
   585→    // самых широких чипов (+ запас под счётчик). Это не проводка в ребёнка —
   586→    // строки полосы получают готовую ширину полосы (ниже), поэтому
   587→    // width_of здесь легитимен (FR-068 W3.3).
   588→    let width = categories
   589→        .iter()
   590→        .map(|name| category_chip_width(name, m, fs) + STRIP_COUNT_SLACK)
   591→        .fold(STRIP_MIN_W, f32::max);
   592→    let rows_h = categories.len() as f32 * CATEGORY_ROW_H;
   593→    // + строка шеврона внизу полосы
   594→    let height = rows_h + STRIP_PAD_V * 2.0 + CATEGORY_ROW_H;
   595→    let y = ((window_h - height) / 2.0).clamp(
   596→        PANEL_TOP_MARGIN,
   597→        (window_h - height - PANEL_TOP_MARGIN).max(PANEL_TOP_MARGIN),
   598→    );
   599→    let x = PANEL_MARGIN;
   600→    // W-c (аудит ui-kit §10): строки категорий — окно kit::list_rows
   601→    // (ScrollState без прокрутки: вьюпорт = контент — вся полоса видна);
   602→    // прежняя стопка «y + PAD_V + i·CATEGORY_ROW_H» воспроизводится
   603→    // дословно (hit-тест реестра зовёт эту же функцию — hit==draw).
   604→    let rows_area = canvas_ui::geometry::UiRect::new(
   605→        x + STRIP_PAD_H,
   606→        y + STRIP_PAD_V,
   607→        (width - STRIP_PAD_H * 2.0).max(0.0),
   608→        rows_h,
   609→    );
   610→    let strip_scroll = canvas_ui::kit::ScrollState {
   611→        offset: 0.0,
   612→        content_h: rows_h,
   613→        viewport_h: rows_h,
   614→    };
   615→    let rows = canvas_ui::kit::list_rows(
   616→        rows_area,
   617→        &strip_scroll,
   618→        CATEGORY_ROW_H,
   619→        0.0,
   620→        categories.len(),
   621→    )
   622→    .into_iter()
   623→    .zip(categories.iter())
   624→    .map(|((_, r), name)| ([r.x, r.y, r.w, r.h], name.clone()))
   625→    .collect();
   626→    StripLayout {
   627→        rect: [x, y, width, height],
   628→        rows,
   629→        chevron_rect: [
   630→            x + STRIP_PAD_H,
   631→            y + STRIP_PAD_V + rows_h,
   632→            width - STRIP_PAD_H * 2.0,
   633→            CATEGORY_ROW_H,
   634→        ],
   635→    }
   636→}
   637→
   638→/// Геометрия flyout — выпадающего списка шаблонов категории справа от
   639→/// полосы. Rect'ы — `[x, y, w, h]`. `row_rects` — только видимое окно строк
   640→/// (обрезка по скроллу как у строк дока); `scroll_top` в раскладке уже
   641→/// клампнут к `max_scroll`.
   642→#[derive(Debug, Clone, PartialEq)]
   643→pub struct FlyoutLayout {
   644→    /// Прямоугольник flyout.
   645→    pub rect: [f32; 4],
   646→    /// Видимые строки (начиная с клампнутого `scroll_top`).
   647→    pub row_rects: Vec<[f32; 4]>,
   648→    /// Клампнутая первая видимая строка.
   649→    pub scroll_top: usize,
   650→    /// Максимум прокрутки в строках.
   651→    pub max_scroll: usize,
   652→}
   653→
   654→/// Инсет строки-карточки flyout от её шага ([`ROW_HEIGHT`]): карточка ниже
   655→/// соседней на 4 px (прежний литерал `ROW_HEIGHT - 4.0`), зазор между
   656→/// карточками — те же 4 px (шаг строки `list_rows` = row_h + gap =
   657→/// [`ROW_HEIGHT`]).
   658→pub const FLYOUT_ROW_INSET: f32 = 4.0;
   659→
   660→/// Раскладка flyout справа от строки-категории (`strip_row_rect`, xywh).
   661→///
   662→/// W-c (аудит ui-kit §8 п.11, §10 — «flyout — ручные клампы»): ручные
   663→/// клампы координат у краёв заменены kit-dropdown
   664→/// `canvas_ui::kit::dropdown_menu` («якорь + flip + финальный кламп к
   665→/// вьюпорту», паттерн `hints_ui::popup_rect`):
   666→/// - якорь — тонкая линия у ПРАВОГО края строки категории: flyout
   667→///   раскрывается справа от полосы с прежним зазором [`FLYOUT_GAP`]
   668→///   (зазор зашит в позицию якоря), по вертикали якорь = строка категории;
   669→/// - вьюпорт — окно с полями [`PANEL_TOP_MARGIN`] по вертикали и
   670→///   [`PANEL_MARGIN`] справа: кит сам опускает меню ПОД строку, у нижнего
   671→///   края разворачивает НАД ней (flip), переполнение прижимает к низу и
   672→///   финально клампит к вьюпорту — строки никогда не вылезают за область
   673→///   видимости;
   674→/// - ширина — [`flyout_width`]: горизонтальный кламп кита держит правый
   675→///   край за [`PANEL_MARGIN`] (у узкого окна сдвигает меню влево, совсем
   676→///   не влезающее — обрезает до вьюпорта).
   677→///
   678→/// Строки — окно `kit::list_rows` ([`ScrollState`]): шаг [`ROW_HEIGHT`]
   679→/// (карточка = `ROW_HEIGHT − FLYOUT_ROW_INSET`, зазор = `FLYOUT_ROW_INSET` —
   680→/// прежняя стопка дословно); рисуются только ПОЛНОСТЬЮ видимые строки
   681→/// (частичный низ — прежний контракт flyout: полоса скроллится построчно).
   682→pub fn flyout_layout(
   683→    strip_row_rect: [f32; 4],
   684→    item_count: usize,
   685→    window_w: f32,
   686→    window_h: f32,
   687→    scroll_top: usize,
   688→) -> FlyoutLayout {
   689→    use canvas_ui::geometry::{UiRect, UiVec2};
   690→    use canvas_ui::kit::{self, ScrollState};
   691→
   692→    // Вьюпорт: поля PANEL_TOP_MARGIN сверху/снизу, PANEL_MARGIN справа
   693→    // (левая граница — край окна: flyout и так правее полосы).
   694→    let viewport = UiRect::new(
   695→        0.0,
   696→        PANEL_TOP_MARGIN,
   697→        (window_w - PANEL_MARGIN).max(0.0),
   698→        (window_h - PANEL_TOP_MARGIN * 2.0).max(0.0),
   699→    );
   700→    // Якорь — правый край строки категории + FLYOUT_GAP, высота = строка.
   701→    let anchor = UiRect::new(
   702→        strip_row_rect[0] + strip_row_rect[2] + FLYOUT_GAP,
   703→        strip_row_rect[1],
   704→        1.0,
   705→        strip_row_rect[3],
   706→    );
   707→    // Желаемый контент: ширина flyout, высота — все строки + вертикальные
   708→    // поля (кламп высоты к вьюпорту — внутри dropdown_menu).
   709→    let content = UiVec2::new(
   710→        flyout_width(),
   711→        item_count as f32 * ROW_HEIGHT + FLYOUT_PAD_V * 2.0,
   712→    );
   713→    let menu = kit::dropdown_menu(anchor, viewport, content).menu;
   714→    // Окно строк — из ФИНАЛЬНОЙ высоты меню (кламп кита учтён): прежняя
   715→    // формула visible_count по клампнутой высоте.
   716→    let visible_count = if item_count == 0 {
   717→        0
   718→    } else {
   719→        (((menu.h - FLYOUT_PAD_V * 2.0) / ROW_HEIGHT).floor() as usize)
   720→            .max(1)
   721→            .min(item_count)
   722→    };
   723→    let max_scroll = item_count.saturating_sub(visible_count);
   724→    let top = scroll_top.min(max_scroll);
   725→    // Строки — kit::list_rows: offset построчный (шаг ROW_HEIGHT), карточка
   726→    // ниже шага на FLYOUT_ROW_INSET. Частично видимый низ (остаток деления
   727→    // высоты окна на шаг) не рисуется — фильтр полной видимости (у кита
   728→    // нет scissor'а на этом проходе, прежний контракт «целые строки»).
   729→    let area = UiRect::new(
   730→        menu.x + FLYOUT_PAD_H,
   731→        menu.y + FLYOUT_PAD_V,
   732→        (menu.w - FLYOUT_PAD_H * 2.0).max(0.0),
   733→        (menu.h - FLYOUT_PAD_V * 2.0).max(0.0),
   734→    );
   735→    let scroll = ScrollState {
   736→        offset: top as f32 * ROW_HEIGHT,
   737→        content_h: item_count as f32 * ROW_HEIGHT,
   738→        viewport_h: area.h,
   739→    };
   740→    let row_rects = kit::list_rows(
   741→        area,
   742→        &scroll,
   743→        ROW_HEIGHT - FLYOUT_ROW_INSET,
   744→        FLYOUT_ROW_INSET,
   745→        item_count,
   746→    )
   747→    .into_iter()
   748→    .filter(|(_, r)| r.bottom() <= area.bottom() + 0.01)
   749→    .map(|(_, r)| [r.x, r.y, r.w, r.h])
   750→    .collect();
   751→    FlyoutLayout {
   752→        rect: [menu.x, menu.y, menu.w, menu.h],
   753→        row_rects,
   754→        scroll_top: top,
   755→        max_scroll,
   756→    }
   757→}

   759→/// Задержка открытия flyout по наведению (hover-intent, как у
   760→/// `PaletteHover`): провод курсора через полосу к канвасу не мигает списками.
   761→pub const STRIP_OPEN_DELAY_MS: u64 = 150;
   762→/// Отсрочка закрытия после ухода курсора с полосы/flyout (grace period).
   763→pub const STRIP_CLOSE_DELAY_MS: u64 = 300;
   764→
   765→/// Состояние hover-раскрытия категорий свёрнутой полосы палитры (ревизия
   766→/// FR-025). Механика — как у палитры выделения (`PaletteHover`, FR-009/010):
   767→/// hover-intent 150 мс на открытие, grace 300 мс на закрытие, пин по клику
   768→/// (WAI-ARIA menu button), удержание пока курсор внутри flyout. Скролл
   769→/// flyout — колесом мыши, кламп к `max_scroll` (max_scroll зависит от числа
   770→/// шаблонов категории — передаётся снаружи).
   771→#[derive(Debug, Clone)]
   772→pub struct StripHover {
   773→    /// Раскрытая категория (индекс строки в `StripLayout::rows`).
   774→    pub open: Option<usize>,
   775→    /// Пин по клику: раскрытие держится после ухода курсора до повторного
   776→    /// клика/Esc.
   777→    pub pinned: bool,
   778→    /// Первая видимая строка flyout (прокрутка колесом).
   779→    pub scroll_top: usize,
   780→    /// Строка полосы под курсором (None — мимо полосы) — для подсветки;
   781→    /// смена строки — тоже повод для перерисовки.
   782→    pub hovered_row: Option<usize>,
   783→    /// (категория, момент входа курсора) — накопление hover-intent.
   784→    trigger_since: Option<(usize, Instant)>,
   785→    /// Момент ухода курсора с открытой зоны — отсрочка закрытия.
   786→    left_since: Option<Instant>,
   787→}
   788→
   789→impl Default for StripHover {
   790→    fn default() -> Self {
   791→        Self::new()
   792→    }
   793→}
   794→
   795→impl StripHover {
   796→    pub fn new() -> Self {
   797→        Self {
   798→            open: None,
   799→            pinned: false,
   800→            scroll_top: 0,
   801→            hovered_row: None,
   802→            trigger_since: None,
   803→            left_since: None,
   804→        }
   805→    }
   806→
   807→    /// Идут кадры ожидания (hover-intent / grace) — `about_to_wait` держит
   808→    /// цикл перерисовки, иначе задержки не сработают при неподвижном курсоре.
   809→    pub fn pending(&self) -> bool {
   810→        self.trigger_since.is_some() || self.left_since.is_some()
   811→    }
   812→
   813→    /// Кадровое обновление: `hovered` — строка категории под курсором,
   814→    /// `in_flyout` — курсор внутри rect'а flyout раскрытой категории.
   815→    /// true — раскрытие или строка под курсором изменились (нужна
   816→    /// перерисовка); при неподвижном курсоре стабильно false — без
   817→    /// самоподдерживающегося цикла кадров.
   818→    pub fn update_at(&mut self, hovered: Option<usize>, in_flyout: bool, now: Instant) -> bool {
   819→        let open_delay = Duration::from_millis(STRIP_OPEN_DELAY_MS);
   820→        let close_delay = Duration::from_millis(STRIP_CLOSE_DELAY_MS);
   821→        let row_changed = self.hovered_row != hovered;
   822→        self.hovered_row = hovered;
   823→        let mut changed = false;
   824→        match hovered {
   825→            Some(cat) => {
   826→                self.left_since = None;
   827→                match self.open {
   828→                    Some(open) if open == cat => self.trigger_since = None,
   829→                    _ => {
   830→                        // Курсор на другой/закрытой категории — копим intent
   831→                        match self.trigger_since {
   832→                            Some((target, since)) if target == cat => {
   833→                                if now.duration_since(since) >= open_delay {
   834→                                    self.open = Some(cat);
   835→                                    self.scroll_top = 0;
   836→                                    self.trigger_since = None;
   837→                                    changed = true;
   838→                                }
   839→                            }
   840→                            _ => self.trigger_since = Some((cat, now)),
   841→                        }
   842→                    }
   843→                }
   844→            }
   845→            None => {
   846→                self.trigger_since = None;
   847→                if in_flyout || self.pinned {
   848→                    // Внутри flyout или закреплено кликом — держим открытым
   849→                    self.left_since = None;
   850→                } else if self.open.is_some() {
   851→                    match self.left_since {
   852→                        None => self.left_since = Some(now),
   853→                        Some(since) => {
   854→                            if now.duration_since(since) >= close_delay {
   855→                                self.reset();
   856→                                changed = true;
   857→                            }
   858→                        }
   859→                    }
   860→                }
   861→            }
   862→        }
   863→        changed || row_changed
   864→    }
   865→
   866→    /// Клик по строке категории: пин-переключение раскрытия (открывает без
   867→    /// задержки — намерение явное; повторный клик — закрыть).
   868→    pub fn toggle_trigger(&mut self, category: usize) {
   869→        if self.open == Some(category) && self.pinned {
   870→            self.reset();
   871→        } else {
   872→            self.open = Some(category);
   873→            self.pinned = true;
   874→            self.scroll_top = 0;
   875→            self.trigger_since = None;
   876→            self.left_since = None;
   877→        }
   878→    }
   879→
   880→    /// Прокрутка flyout на `delta` строк с клампом к `[0, max_scroll]`.
   881→    pub fn scroll_by(&mut self, delta: i32, max_scroll: usize) {
   882→        let cur = self.scroll_top as i32;
   883→        self.scroll_top = (cur + delta).clamp(0, max_scroll as i32) as usize;
   884→    }
   885→
   886→    /// Сброс: flyout закрыт, пин и прокрутка сняты.
   887→    pub fn reset(&mut self) {
   888→        self.open = None;
   889→        self.pinned = false;
   890→        self.scroll_top = 0;
   891→        self.hovered_row = None;
   892→        self.trigger_since = None;
   893→        self.left_since = None;
   894→    }
   895→}

   954→    let chip_strip = head_flow[2];

  1038→        offset,
  1039→        content_h,
  1040→        viewport_h: rows_area_h,

  1087→/// Шаг строки списка панели (расстояние до следующей): секция —
  1088→/// [`SECTION_HEIGHT`], шаблон — [`ROW_HEIGHT`] (карточка + зазор 4 px).
  1089→fn row_step(row: &PanelRow) -> f32 {
  1090→    match row {
  1091→        PanelRow::Section(_) => SECTION_HEIGHT,
  1092→        PanelRow::Template(_) => ROW_HEIGHT,
  1093→    }
  1094→}
  1095→
  1096→/// Высота карточки строки: секция — заголовок во весь шаг, шаблон —
  1097→/// карточка ниже шага на 4 px (зазор, прежний литерал `ROW_HEIGHT − 4`).
  1098→fn row_card_h(row: &PanelRow) -> f32 {
  1099→    match row {
  1100→        PanelRow::Section(_) => SECTION_HEIGHT,
  1101→        PanelRow::Template(_) => ROW_HEIGHT - 4.0,
  1102→    }
  1103→}

  1105→impl PanelLayout {
  1106→    /// FR-082/W-e: «приклейка хвоста» — наибольшее `scroll_top`, при
  1107→    /// котором панель показывает хвост списка ЦЕЛИКОМ (последняя строка
  1108→    /// у нижней границы окна строк). W-e (аудит §8 п.11): предел задаёт
  1109→    /// только измеренная высота окна строк — кэп `MAX_VISIBLE_ROWS`
  1110→    /// удалён (раньше max = max(по высоте, по кэпу) и на высоких окнах
  1111→    /// список обрывался дюжиной строк). Условие монотонно по старту
  1112→    /// хвоста (хвост только укорачивается); suffix-суммы дают O(n) без
  1113→    /// повторных прогонов раскладки. Условие видимости строки i при
  1114→    /// старте s — то же, что в цикле выше:
  1115→    /// `rows_top + Σ шагов(s..i) + card_h(i) ≤ bottom_limit` ⟹ весь
  1116→    /// хвост s..len виден ⟺ `Σ шагов(s..len−1) + card_h(len−1) ≤
  1117→    /// rows_area`.
  1118→    fn max_scroll_of(rows: &[PanelRow], rows_top: f32, bottom_limit: f32) -> usize {
  1119→        let len = rows.len();
  1120→        if len == 0 {
  1121→            return 0;
  1122→        }
  1123→        let rows_area = (bottom_limit - rows_top).max(0.0);
  1124→        let last_card_h = row_card_h(rows.last().expect("len > 0"));
  1125→        // suffix_steps[i] — сумма шагов строк i..len
  1126→        let mut suffix_steps = vec![0.0f32; len + 1];
  1127→        for i in (0..len).rev() {
  1128→            suffix_steps[i] = suffix_steps[i + 1] + row_step(&rows[i]);
  1129→        }
  1130→        // Граница по высоте: наименьшее s, при котором высота хвоста
  1131→        // s..len целиком ≤ rows_area; len — даже последняя строка не
  1132→        // влезает (панель теснее одной строки) — прокрутки нет
  1133→        let by_height = (0..len)
  1134→            .find(|&s| suffix_steps[s] - suffix_steps[len - 1] + last_card_h <= rows_area)
  1135→            .unwrap_or(len);
  1136→        if by_height >= len {
  1137→            return 0;
  1138→        }
  1139→        by_height
  1140→    }
  1141→}

  1145→// Рестайл FR-022 (2026-09-16) по скриншоту пользователя: «прямоугольные
  1146→// плашки на концентрических кольцах» заменены donut-секторами в стиле
  1147→// circular-menu (https://www.npmjs.com/package/circular-menu — ориентир

  1149→// отверстием, в каждом секторе иконка + короткая подпись, между секторами

  1152→// полярной математикой (`sectors::angle_gap` — Rust-зеркало WGSL), поэтому
  1153→// расхождения «клик мимо нарисованного» нет (раньше угловой тест расходился
  1154→// с AABB плашек — заменён осознанно).

  1156→/// Толщина кольца wheel (радиальная ширина секторов), лог. px.
  1157→pub const WHEEL_RING_THICKNESS: f32 = 68.0;
  1158→/// Межкольцевой зазор (радиальный), лог. px.
  1159→pub const WHEEL_RING_GAP: f32 = 8.0;
  1160→/// Угловой зазор между соседними секторами кольца, радианы (≈2.5°).
  1161→pub const WHEEL_SECTOR_GAP: f32 = 0.043_633_23;
  1162→/// Минимальная дуга сектора на среднем радиусе кольца, лог. px — из неё
  1163→/// выводится вместимость кольца (окружность / (дуга + угловой зазор в px)).
  1164→pub const WHEEL_MIN_SECTOR_ARC: f32 = 60.0;
  1165→/// Максимум секторов в одном кольце; большее — концентрические под-кольца
  1166→/// (внешние вместительнее — окружность больше).
  1167→pub const WHEEL_RING_CAP: usize = 6;
  1168→/// Радиус центрального отверстия (дырка donut — глотает клик мимо секторов).
  1169→pub const WHEEL_HUB_R: f32 = 24.0;
  1170→/// FR-022: диаметр кнопки-хаба (клик = «назад»/«закрыть»). Крупная цель
  1171→/// ≥ 44 лог. px — гайдлайн сенсорных целей (Big Medium); совпадает с
  1172→/// удвоенным [`WHEEL_HUB_R`] — зона хаба и кнопка — одно и то же.
  1173→pub const WHEEL_HUB_D: f32 = 48.0;
  1174→/// Отступ wheel от краёв окна при клампе центра.
  1175→pub const WHEEL_SCREEN_MARGIN: f32 = 8.0;
  1176→/// Ширина слота подписи шаблона в секторе wheel (лог. px) — дизайн-граница
  1177→/// переноса, единая для раскладки ([`split_two_lines`]) и отрисовки
  1178→/// (`overlays.rs`: подпись центрируется в слоте этой ширины при кегле
  1179→/// [`WHEEL_TPL_FONT`]). CR-015: прежний символьный бюджет
  1180→/// `WHEEL_TPL_TEXT_CHARS = 15` (~96 px кегля 11) расходился со слотом
  1181→/// отрисовки 64 px — строки переноса клипались рендером; теперь строки
  1182→/// переноса обязаны влезать в слот ПО ИЗМЕРЕННОЙ ШИРИНЕ.
  1183→pub const WHEEL_TPL_TEXT_W: f32 = 64.0;
  1184→/// Кегль подписи шаблона в секторе wheel — паритет отрисовке (overlays.rs,
  1185→/// screen-тексты подписи сектора).
  1186→pub const WHEEL_TPL_FONT: f32 = 11.0;

  1188→/// Состояние радиального меню шаблонов (FR-018, `Shift+клик` по пустому

  1200→/// Цель попадания в wheel.
  1201→#[derive(Debug, Clone, PartialEq)]
  1202→pub enum WheelHit {
  1203→    /// Сектор категории (индекс в `registry.categories()`).
  1204→    Category(usize),
  1205→    /// Сектор шаблона (индекс ВНУТРИ категории — `registry.by_category`;
  1206→    /// сквозной по кольцам, кольца — только план раскладки).
  1207→    Template(usize),
  1208→}

  1210→/// Donut-сектор wheel: дуговой интервал [a0, a1] (радианы, 12 часов = -π/2,
  1211→/// по часовой) на кольце [r0, r1] (лог. px от центра). Единый источник
  1212→/// правды рендера (`SectorInstance` в `wheel_overlay`) и hit-test
  1213→/// ([`WheelGeometry::hit`]) — инвариант WYSIWYG.
  1214→#[derive(Debug, Clone, PartialEq)]
  1215→pub struct WheelSector {
  1216→    pub a0: f32,
  1217→    pub a1: f32,
  1218→    pub r0: f32,
  1219→    pub r1: f32,
  1220→    pub hit: WheelHit,
  1221→}
  1222→
  1223→impl WheelSector {
  1224→    /// Середина дуги сектора (для иконки/подписи и hit-пробы тестов).
  1225→    pub fn mid_angle(&self) -> f32 {
  1226→        (self.a0 + self.a1) / 2.0
  1227→    }
  1228→    /// Средний радиус кольца сектора (для иконки/подписи).
  1229→    pub fn mid_radius(&self) -> f32 {
  1230→        (self.r0 + self.r1) / 2.0
  1231→    }
  1232→}

  1234→/// Геометрия wheel на кадр: центр (с клампом к окну), сектора (шаблоны —
  1235→/// внутренние кольца от дырки наружу, категории — внешние), внешний радиус,
  1236→/// квадрат кнопки-хаба. Строится чистой функцией [`wheel_geometry`] — и
  1237→/// рендер (`wheel_overlay`), и клики ([`WheelGeometry::hit`]) вызывают её с
  1238→/// теми же аргументами, поэтому расхождений быть не может.
  1239→#[derive(Debug, Clone, PartialEq)]
  1240→pub struct WheelGeometry {
  1241→    /// Центр меню: точка клика, сдвинутая клампом внутрь окна, если wheel
  1242→    /// целиком не влезает.
  1243→    pub center: Vec2,
  1244→    /// Сектора: шаблоны (изнутри наружу), затем категории.
  1245→    pub sectors: Vec<WheelSector>,
  1246→    /// Внешний радиус (максимальный r1 секторов; для «клик заметно дальше —
  1247→    /// закрыть» и клампа центра).
  1248→    pub extent: f32,
  1249→    /// FR-022: квадрат кнопки-хаба (`WHEEL_HUB_D × WHEEL_HUB_D`) вокруг
  1250→    /// центра — единый источник для рендера круга и клика «назад/закрыть».
  1251→    pub hub: [f32; 4],
  1252→}
  1253→
  1254→impl WheelGeometry {
  1255→    /// FR-022: курсор на кнопке-хабе? Тест по квадрату хаба (визуальный
  1256→    /// круг вписан в него; углы квадрата прощаются — Fitts).
  1257→    pub fn hub_hit(&self, cursor: Vec2) -> bool {
  1258→        let [x, y, w, h] = self.hub;
  1259→        cursor[0] >= x && cursor[0] <= x + w && cursor[1] >= y && cursor[1] <= y + h
  1260→    }
  1261→
  1262→    /// Сектор под курсором (screen px): полярный hit-test — кольцо по
  1263→    /// радиусу (r0 ≤ r ≤ r1), угловая принадлежность — знаковое угловое
  1264→    /// расстояние `sectors::angle_gap` (тот же код, что в WGSL-шейдере
  1265→    /// рендера, поэтому клик всегда попадает в нарисованное). В угловом
  1266→    /// зазоре, в дырке (r < WHEEL_HUB_R) и за внешним радиусом — None.
  1267→    pub fn hit(&self, cursor: Vec2) -> Option<WheelHit> {
  1268→        let dx = cursor[0] - self.center[0];
  1269→        let dy = cursor[1] - self.center[1];
  1270→        let r = dx.hypot(dy);
  1271→        let theta = norm_angle(dy.atan2(dx));
  1272→        self.sectors
  1273→            .iter()
  1274→            .find(|s| r >= s.r0 && r <= s.r1 && angle_gap(theta, s.a0, s.a1) <= 0.0)
  1275→            .map(|s| s.hit.clone())
  1276→    }
  1277→}
  1278→
  1279→/// Радиусы кольца `k` (0 — кольцо у дырки): r0 = HUB_R + k·(толщина+зазор),
  1280→/// r1 = r0 + толщина.
  1281→fn ring_radii(k: usize) -> (f32, f32) {
  1282→    let r0 = WHEEL_HUB_R + k as f32 * (WHEEL_RING_THICKNESS + WHEEL_RING_GAP);
  1283→    (r0, r0 + WHEEL_RING_THICKNESS)
  1284→}
  1285→
  1286→/// Вместимость кольца `k`: число секторов, чья дуга на среднем радиусе
  1287→/// ≥ [`WHEEL_MIN_SECTOR_ARC`] с учётом углового зазора между соседями.
  1288→/// Гарантированно ≥ 2 (при HUB_R=24/толщине 68 кольцо 0 вмещает 5).
  1289→pub fn ring_capacity(k: usize) -> usize {
  1290→    let (r0, r1) = ring_radii(k);
  1291→    let r_mid = (r0 + r1) / 2.0;
  1292→    let arc_gap = WHEEL_SECTOR_GAP * r_mid;
  1293→    ((std::f32::consts::TAU * r_mid / (WHEEL_MIN_SECTOR_ARC + arc_gap)).floor() as usize)
  1294→        .clamp(2, WHEEL_RING_CAP)
  1295→}
  1296→
  1297→/// План колец: `count` секторов по кольцам вместимости [`ring_capacity`]
  1298→/// (у внешних колец она больше), сбалансированно: 10 → [5, 5], 11 → [5, 6],
  1299→/// 7 → [3, 4], 15 → [5, 5, 5]. Каждая часть ≤ вместимости своего кольца.
  1300→pub fn wheel_ring_plan(count: usize) -> Vec<usize> {
  1301→    if count == 0 {
  1302→        return Vec::new();
  1303→    }
  1304→    // Минимальное число колец, чья суммарная вместимость покрывает count
  1305→    let mut rings = 0_usize;
  1306→    let mut total = 0_usize;
  1307→    while total < count {
  1308→        total += ring_capacity(rings);
  1309→        rings += 1;
  1310→    }
  1311→    let base = count / rings;
  1312→    let extra = count % rings;
  1313→    (0..rings)
  1314→        .map(|i| base + usize::from(i + extra >= rings))
  1315→        .collect()
  1316→}
  1317→
  1318→/// Сектора одного кольца: `count` секторов на кольце `k`, цель каждого —
  1319→/// через `hit_of` (нарастающий сквозной индекс — за пределами функции).
  1320→/// Старт — 12 часов, по часовой; между соседними секторами угловой зазор
  1321→/// [`WHEEL_SECTOR_GAP`].
  1322→fn ring_sectors(k: usize, count: usize, hit_of: impl FnMut(usize) -> WheelHit) -> Vec<WheelSector> {
  1323→    let (r0, r1) = ring_radii(k);
  1324→    let step = std::f32::consts::TAU / count as f32;
  1325→    let start = -std::f32::consts::FRAC_PI_2;
  1326→    let mut hit_of = hit_of;
  1327→    (0..count)
  1328→        .map(|i| {
  1329→            let a0 = start + step * i as f32 + WHEEL_SECTOR_GAP / 2.0;
  1330→            let a1 = start + step * (i + 1) as f32 - WHEEL_SECTOR_GAP / 2.0;
  1331→            WheelSector {
  1332→                a0,
  1333→                a1,
  1334→                r0,
  1335→                r1,
  1336→                hit: hit_of(i),
  1337→            }
  1338→        })
  1339→        .collect()
  1340→}
  1341→
  1342→/// Перенос имени шаблона в подписи сектора: имя ШИРЕ бюджета `max_w`
  1343→/// (по измеренной ширине при кегле подписи [`WHEEL_TPL_FONT`] — CR-015)
  1344→/// делится на две строки по самому позднему подходящему разделителю
  1345→/// (пробел или дефис) так, чтобы ОБЕ части влезали в бюджет по измеренной
  1346→/// ширине; не получилось — одна строка (хвост клипается рендером по ширине
  1347→/// слота [`WHEEL_TPL_TEXT_W`]). Замерщик/FontSystem — внешние (тот же пул
  1348→/// шейпинга, что у draw-путей — FR-094).
  1349→pub fn split_two_lines(
  1350→    m: &mut TextMeasurer,
  1351→    fs: &mut cosmic_text::FontSystem,
  1352→    name: &str,
  1353→    max_w: f32,
  1354→) -> (String, Option<String>) {
  1355→    if m.width_of(fs, name, FAMILY, WHEEL_TPL_FONT) <= max_w {
  1356→        return (name.to_owned(), None);
  1357→    }
  1358→    let mut best: Option<usize> = None; // байт-позиция начала второй строки
  1359→    for (byte, ch) in name.char_indices() {
  1360→        if ch != ' ' && ch != '-' {
  1361→            continue;
  1362→        }
  1363→        let split = byte + ch.len_utf8();
  1364→        let (n1, n2) = (&name[..split], &name[split..]);
  1365→        // Обе части влезают в бюджет ПО ИЗМЕРЕННОЙ ШИРИНЕ и не вырождены
  1366→        // (минимум 2 символа — семантическая страховка от обрывков «Т-»).
  1367→        if m.width_of(fs, n1, FAMILY, WHEEL_TPL_FONT) <= max_w
  1368→            && m.width_of(fs, n2, FAMILY, WHEEL_TPL_FONT) <= max_w
  1369→            && n1.chars().count() >= 2
  1370→            && n2.chars().count() >= 2
  1371→            && best.map_or(true, |b| split > b)
  1372→        {
  1373→            best = Some(split);
  1374→        }
  1375→    }
  1376→    match best {
  1377→        Some(split) => (
  1378→            name[..split].trim_end().to_owned(),
  1379→            Some(name[split..].to_owned()),
  1380→        ),
  1381→        None => (name.to_owned(), None),
  1382→    }
  1383→}
  1384→
  1385→/// Геометрия wheel (рестайл FR-022, donut-сектора): кольца фиксированной
  1386→/// толщины [`WHEEL_RING_THICKNESS`] с межкольцевым зазором
  1387→/// [`WHEEL_RING_GAP`]; вместимость каждого кольца выводится из минимальной
  1388→/// дуги сектора ([`ring_capacity`]), переполнение уходит во внешние
  1389→/// под-кольца ([`wheel_ring_plan`]). Шаблоны — кольца от дырки наружу,
  1390→/// категории — внешние кольца. Угловые зазоры между секторами —
  1391→/// [`WHEEL_SECTOR_GAP`], они же формируют визуальные промежутки рендера.
  1392→/// `screen` — точка клика; `window_w/h` — размер окна для клампа центра
  1393→/// (клик у края не должен обрезать меню).
  1394→pub fn wheel_geometry(
  1395→    screen: Vec2,
  1396→    window_w: f32,
  1397→    window_h: f32,
  1398→    category_count: usize,
  1399→    template_count: usize,
  1400→) -> WheelGeometry {
  1401→    // 1. Кольца шаблонов — от дырки наружу. Индекс шаблона — сквозной по
  1402→    //    категории (кольца — только план раскладки).
  1403→    let plan = wheel_ring_plan(template_count);
  1404→    let mut sectors: Vec<WheelSector> = Vec::new();
  1405→    let mut template_index = 0_usize;
  1406→    for (k, count) in plan.iter().enumerate() {
  1407→        sectors.extend(ring_sectors(k, *count, |_| {
  1408→            let hit = WheelHit::Template(template_index);
  1409→            template_index += 1;
  1410→            hit
  1411→        }));
  1412→    }
  1413→    // 2. Кольца категорий — снаружи шаблонных (индексы по порядку реестра).
  1414→    let cat_plan = wheel_ring_plan(category_count);
  1415→    let mut category_index = 0_usize;
  1416→    for (j, count) in cat_plan.iter().enumerate() {
  1417→        let k = plan.len() + j;
  1418→        sectors.extend(ring_sectors(k, *count, |_| {
  1419→            let hit = WheelHit::Category(category_index);
  1420→            category_index += 1;
  1421→            hit
  1422→        }));
  1423→    }
  1424→
  1425→    // 3. Внешний радиус, кнопка-хаб (FR-022) и кламп центра к окну.
  1426→    let extent = sectors.iter().map(|s| s.r1).fold(WHEEL_HUB_R, f32::max);
  1427→    let lo = extent + WHEEL_SCREEN_MARGIN;
  1428→    let center: Vec2 = if window_w >= lo * 2.0 && window_h >= lo * 2.0 {
  1429→        [
  1430→            screen[0].clamp(lo, window_w - lo),
  1431→            screen[1].clamp(lo, window_h - lo),
  1432→        ]
  1433→    } else {
  1434→        [window_w / 2.0, window_h / 2.0]
  1435→    };
  1436→    let hub = [
  1437→        center[0] - WHEEL_HUB_D / 2.0,
  1438→        center[1] - WHEEL_HUB_D / 2.0,
  1439→        WHEEL_HUB_D,
  1440→        WHEEL_HUB_D,
  1441→    ];
  1442→    WheelGeometry {
  1443→        center,
  1444→        sectors,
  1445→        extent,
  1446→        hub,
  1447→    }
  1448→}

  1450→/// Точка на кольце для рендера сектора (центр сектора на среднем радиусе).
  1451→pub fn sector_point(center: Vec2, angle: f32, radius: f32) -> Vec2 {
  1452→    [
  1453→        center[0] + radius * angle.cos(),
  1454→        center[1] + radius * angle.sin(),
  1455→    ]
  1456→}
