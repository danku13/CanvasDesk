    19→    19→/// Замер кнопки по подписи: ширина = текст + 2·BUTTON_PAD_H, высота = const.
    20→    20→/// Пустая подпись → квадрат минимальной ширины BUTTON_HEIGHT.
    21→    21→pub fn button_size(
    22→    22→    label: &str,
    23→    23→    m: &mut TextMeasurer,
    24→    24→    fs: &mut cosmic_text::FontSystem,
    25→    25→    family: &str,
    26→    26→    size: f32,
    27→    27→) -> UiVec2 {
    28→    28→    let text_w = if label.is_empty() {
    29→    29→        0.0
    30→    30→    } else {
    31→    31→        m.width_of(fs, label, family, size)
    32→    32→    };
    33→    33→    let w = (text_w + BUTTON_PAD_H * 2.0).max(BUTTON_HEIGHT);
    34→    34→    UiVec2::new(w, BUTTON_HEIGHT)
    35→    35→}
    36→    36→
    37→    37→/// Кнопка в слоте: измеренный размер, позиция — stack по выравниванию;
    38→    38→/// подпись усекается `ellipsis`, если слот уже кнопки (деградация видна
    39→    39→/// потребителю и линту, не молчаливый break).
    40→    40→pub struct ButtonLayout {
    41→    41→    /// Rect кнопки.
    42→    42→    pub rect: UiRect,
    43→    43→    /// Подпись после ellipsis (может отличаться от исходной).
    44→    44→    pub label: String,
    45→    45→}
    46→    46→
    47→    47→pub fn button_layout(
    48→    48→    slot: UiRect,
    49→    49→    label: &str,
    50→    50→    align: (HAlign, VAlign),
    51→    51→    m: &mut TextMeasurer,
    52→    52→    fs: &mut cosmic_text::FontSystem,
    53→    53→    family: &str,
    54→    54→    size: f32,
    55→    55→) -> ButtonLayout {
    56→    56→    let desired = button_size(label, m, fs, family, size);
    57→    57→    let max_w = slot.w;
    58→    58→    // Деградация узкого слота: усечение по фактической ширине (не срез);
    59→    59→    // кнопка не вылезает за слот — сжимается до его ширины.
    60→    60→    let width = desired.x.min(max_w);
    61→    61→    let shown = if desired.x > max_w {
    62→    62→        m.ellipsis(
    63→    63→            fs,
    64→    64→            label,
    65→    65→            family,
    66→    66→            size,
    67→    67→            (max_w - BUTTON_PAD_H * 2.0).max(0.0),
    68→    68→        )
    69→    69→    } else {
    70→    70→        label.to_owned()
    71→    71→    };
    72→    72→    let rect = stack(slot, UiVec2::new(width, BUTTON_HEIGHT), align.0, align.1);
    73→    73→    ButtonLayout { rect, label: shown }
    74→    74→}
    75→    75→
    76→    76→/// Стиль кнопки: слот заливки по варианту, слот hover/pressed по состоянию,
    77→    77→/// текст — слот по состоянию (disabled — свой слот). Никакой арифметики
    78→    78→/// над цветами: только выбор слота.
    79→    79→pub fn button_style(variant: ButtonVariant, state: KitState, p: &KitPalette) -> ControlStyle {
    80→    80→    // Wave T §5.1.3: + Tertiary, Text, Inverse variants.
    81→    81→    let base_fill = match variant {
    82→    82→        ButtonVariant::Primary => p.control_primary,
    83→    83→        ButtonVariant::Danger => p.control_danger,
    84→    84→        ButtonVariant::Secondary | ButtonVariant::Ghost | ButtonVariant::Tertiary => p.control_fill,
    85→    85→        // Text — прозрачный контейнер (M3 text button).
    86→    86→        ButtonVariant::Text => [0.0, 0.0, 0.0, 0.0],
    87→    87→        // Inverse — на цветном фоне; используем accent как «inverse» fill.
    88→    88→        ButtonVariant::Inverse => p.accent,
    89→    89→    };
    90→    90→    let fill = match state {
    91→    91→        KitState::Normal
    92→    92→        | KitState::Selected
    93→    93→        | KitState::Focused
    94→    94→        | KitState::Dragged
    95→    95→        | KitState::Error => base_fill,
    96→    96→        KitState::Hovered => match variant {
    97→    97→            ButtonVariant::Primary | ButtonVariant::Inverse => p.primary_hover_fill,
    98→    98→            ButtonVariant::Secondary
    99→    99→            | ButtonVariant::Ghost
   100→   100→            | ButtonVariant::Tertiary
   101→   101→            | ButtonVariant::Text
   102→   102→            | ButtonVariant::Danger => p.hover_fill,
   103→   103→        },
   104→   104→        KitState::Pressed => match variant {
   105→   105→            ButtonVariant::Primary | ButtonVariant::Inverse => p.primary_hover_fill,
   106→   106→            _ => p.hover_fill,
   107→   107→        },
   108→   108→        KitState::Disabled => base_fill,
   109→   109→    };
   110→   110→    let border = match (variant, state) {
   111→   111→        (ButtonVariant::Ghost | ButtonVariant::Text, KitState::Hovered | KitState::Pressed) => {
   112→   112→            p.accent
   113→   113→        }
   114→   114→        (ButtonVariant::Ghost | ButtonVariant::Text, _) => [0.0, 0.0, 0.0, 0.0],
   115→   115→        _ => p.control_border,
   116→   116→    };
   117→   117→    let text = match state {
   118→   118→        KitState::Disabled => p.disabled_text,
   119→   119→        _ => match variant {
   120→   120→            ButtonVariant::Ghost | ButtonVariant::Text => p.text,
   121→   121→            ButtonVariant::Inverse => p.text_title,
   122→   122→            _ => p.text_title,
   123→   123→        },
   124→   124→    };
   125→   125→    ControlStyle {
   126→   126→        fill,
   127→   127→        border,
   128→   128→        text,
   129→   129→        radius: canvas_core::tokens::RADIUS_CHIP,
   130→   130→        elevation: super::Elevation::None,
   131→   131→    }
   132→   132→}

   153→   153→/// Замер чипа: ширина = текст + 2·CHIP_PAD_H, высота = const.
   154→   154→pub fn chip_size(
   155→   155→    label: &str,
   156→   156→    m: &mut TextMeasurer,
   157→   157→    fs: &mut cosmic_text::FontSystem,
   158→   158→    family: &str,
   159→   159→    size: f32,
   160→   160→) -> UiVec2 {
   161→   161→    let text_w = m.width_of(fs, label, family, size);
   162→   162→    UiVec2::new(text_w + CHIP_PAD_H * 2.0, CHIP_HEIGHT)
   163→   163→}
   164→   164→
   165→   165→/// Чип от левого края `origin_x` (ряд чипов компонует потребитель через
   166→   166→/// `Row`/SqueezeTail); усечение подписи — ellipsis по `max_width`.
   167→   167→pub struct ChipLayout {
   168→   168→    pub rect: UiRect,
   169→   169→    pub label: String,
   170→   170→}
   171→   171→
   172→   172→pub fn chip_layout(
   173→   173→    origin: UiPoint,
   174→   174→    label: &str,
   175→   175→    max_width: f32,
   176→   176→    m: &mut TextMeasurer,
   177→   177→    fs: &mut cosmic_text::FontSystem,
   178→   178→    family: &str,
   179→   179→    size: f32,
   180→   180→) -> ChipLayout {
   181→   181→    let desired = chip_size(label, m, fs, family, size);
   182→   182→    let shown = if desired.x > max_width {
   183→   183→        m.ellipsis(
   184→   184→            fs,
   185→   185→            label,
   186→   186→            family,
   187→   187→            size,
   188→   188→            (max_width - CHIP_PAD_H * 2.0).max(0.0),
   189→   189→        )
   190→   190→    } else {
   191→   191→        label.to_owned()
   192→   192→    };
   193→   193→    let w = if shown == label { desired.x } else { max_width };
   194→   194→    ChipLayout {
   195→   195→        rect: UiRect::new(origin.x, origin.y, w, CHIP_HEIGHT),
   196→   196→        label: shown,
   197→   197→    }
   198→   198→}

   203→   203→    let fill = match state {

   206→   206→        _ => p.control_fill,
   207→   207→    };
   208→   208→    ControlStyle {
   209→   209→        fill,
   210→   210→        border: p.control_border,
   211→   211→        text: match state {
   212→   212→            KitState::Disabled => p.disabled_text,
   213→   213→            _ => p.text,
   214→   214→        },
   215→   215→        radius: canvas_core::tokens::RADIUS_CHIP,
   216→   216→        elevation: super::Elevation::None,
   217→   217→    }
   218→   218→}

   222→   222→/// Раскладка Switch (тогла): трек (pill) + бегунок + стиль трека + заливка
   223→   223→/// бегунка. `on` определяет позицию бегунка (вправо) и слот заливки трека.
   224→   224→#[derive(Debug, Clone, Copy, PartialEq)]
   225→   225→pub struct SwitchLayout {
   226→   226→    /// Rect трека (pill).
   227→   227→    pub track: UiRect,
   228→   228→    /// Rect бегунка (квадрат внутри трека).
   229→   229→    pub knob: UiRect,
   230→   230→    /// Стиль трека — слот заливки/рамки/радиуса (RADIUS_PILL).
   231→   231→    pub track_style: ControlStyle,
   232→   232→    /// Заливка бегунка (слот `text_title`/`disabled_text`).
   233→   233→    pub knob_fill: [f32; 4],
   234→   234→}
   235→   235→
   236→   236→/// Switch в слоте: трек (pill) + бегунок. `on` — позиция бегунка и слот
   237→   237→/// заливки трека (`control_primary` on / `control_fill` off); `state` —
   238→   238→/// hover/pressed/disabled слоты; `knob_fill` — `text_title` (disabled —
   239→   239→/// `disabled_text`).
   240→   240→pub fn switch(slot: UiRect, on: bool, state: KitState, p: &KitPalette) -> SwitchLayout {
   241→   241→    let track = stack(
   242→   242→        slot,
   243→   243→        UiVec2::new(SWITCH_W, SWITCH_H),
   244→   244→        HAlign::Center,
   245→   245→        VAlign::Center,
   246→   246→    );
   247→   247→    // Бегунок: квадрат side = SWITCH_H - 2·pad; позиция — влево/вправо по `on`.
   248→   248→    let knob_side = (SWITCH_H - 2.0 * SWITCH_KNOB_PAD).max(0.0);
   249→   249→    let knob_x = if on {
   250→   250→        track.right() - SWITCH_KNOB_PAD - knob_side
   251→   251→    } else {
   252→   252→        track.x + SWITCH_KNOB_PAD
   253→   253→    };
   254→   254→    let knob_y = track.y + SWITCH_KNOB_PAD;
   255→   255→    let knob = UiRect::new(knob_x, knob_y, knob_side, knob_side);
   256→   256→    // Стиль трека: заливка — primary on / control_fill off; hover/pressed —
   257→   257→    // свои слоты; disabled — базовая. Радиус RADIUS_PILL (pill).
   258→   258→    let base_fill = if on {
   259→   259→        p.control_primary
   260→   260→    } else {
   261→   261→        p.control_fill
   262→   262→    };
   263→   263→    let fill = match state {
   264→   264→        KitState::Hovered | KitState::Pressed => {
   265→   265→            if on {
   266→   266→                p.primary_hover_fill
   267→   267→            } else {
   268→   268→                p.hover_fill
   269→   269→            }
   270→   270→        }
   271→   271→        _ => base_fill,
   272→   272→    };
   273→   273→    let track_style = ControlStyle {
   274→   274→        fill,
   275→   275→        border: p.control_border,
   276→   276→        text: p.text, // не используется у Switch (без подписи)
   277→   277→        radius: canvas_core::tokens::RADIUS_PILL,
   278→   278→        elevation: super::Elevation::None,
   279→   279→    };
   280→   280→    // Бегунок: text_title (white/light); disabled — приглушён.
   281→   281→    let knob_fill = match state {
   282→   282→        KitState::Disabled => p.disabled_text,
   283→   283→        _ => p.text_title,
   284→   284→    };
   285→   285→    SwitchLayout {
   286→   286→        track,
   287→   287→        knob,
   288→   288→        track_style,
   289→   289→        knob_fill,
   290→   290→    }
   291→   291→}

   355→   355→/// Кнопка «×» в правом-верхнем углу панели/модали (FR-070, audit §6.1 —
   356→   356→/// самый повторяющийся паттерн: 7+ файлов дублировали ручную формулу
   357→   357→/// `x = panel.right() - 28.0, y = panel.y + 6.0` с дрейфом inset/size
   358→   358→/// между 22/24/26/28/30 — канонический размер [`ICON_BUTTON_SIZE`],
   359→   359→/// inset — [`canvas_core::tokens::SPACING_SM`] = 8).
   360→   360→///
   361→   361→/// Возвращает `UiRect` угловой кнопки в слоте `slot` (panel/modal rect);
   362→   362→/// позиция — «прижата к правому-верхнему углу слота» минус inset, размер —
   363→   363→/// квадрат `ICON_BUTTON_SIZE × ICON_BUTTON_SIZE` (та же метрика, что у
   364→   364→/// [`icon_button_rect`] с `HAlign::End, VAlign::Start`, но с явным inset —
   365→   365→/// не вплотную к углу).
   366→   366→///
   367→   367→/// Контракт F-8: цвет/стиль кнопки — забота потребителя
   368→   368→/// ([`icon_button_style`] с `KitState`/`KitPalette`); здесь — только
   369→   369→/// геометрия (как у [`icon_button_rect`]).
   370→   370→pub fn stage_close_button(slot: UiRect) -> UiRect {
   371→   371→    let s = ICON_BUTTON_SIZE;
   372→   372→    let inset = canvas_core::tokens::SPACING_SM;
   373→   373→    UiRect::new(slot.right() - s - inset, slot.y + inset, s, s)
   374→   374→}
   375→   375→
   376→   376→/// LG-вариант [`stage_close_button`] — 30×30 (вместо канонического
   377→   377→/// [`ICON_BUTTON_SIZE`] = 26), тот же inset [`canvas_core::tokens::SPACING_SM`]
   378→   378→/// = 8. Выделен для dialog-шапок с HEADER_H = 56–58 (autolink/explain —
   379→   379→/// FR-060/FR-059 documented deviation): крупная кнопка визуально
   380→   380→/// сбалансирована с высоким заголовком; канонический 26 выглядит мелко.
   381→   381→///
   382→   382→/// **Визуальный скачок при миграции** с hand-rolled (inset 14 от прежнего
   383→   383→/// `panel.right() - 14 - 30`/`panel.y + 14`) на канон `SPACING_SM`=8 —
   384→   384→/// позиция кнопки сдвигается на ~6px по диагонали (вверх+вправо, к углу
   385→   385→/// панели). Это каноническое направление kit (FR-070/§6.1 — «× в углу
   386→   386→/// панели с inset `SPACING_SM`»); прежний inset 14 был наследием ручной
   387→   387→/// вертикальной центровки в HEADER_H=58 (где `(58−30)/2=14`). Когда
   388→   388→/// `HEADER_H` мигрирует на `PANEL_HEADER_H_L`=44 (отдельный TODO W-d),
   389→   389→/// `(44−30)/2`=7 — почти совпадает с `SPACING_SM`=8, и LG-вариант можно
   390→   390→/// будет пересмотреть (возможно, свернуть обратно в [`stage_close_button`]).
   391→   391→///
   392→   392→/// Контракт F-8 (как у [`stage_close_button`]): цвет/стиль — забота
   393→   393→/// потребителя (через [`icon_button_style`] с `KitState`/`KitPalette`);
   394→   394→/// здесь — только геометрия.
   395→   395→pub fn stage_close_button_lg(slot: UiRect) -> UiRect {

   397→   397→    let inset = canvas_core::tokens::SPACING_SM;

   435→   435→pub struct Button {

   443→   443→impl Button {
   444→   444→    /// Новая кнопка: `props.disabled` прошивается в состояние виджета —
   445→   445→    /// дальше стиль (`paint`) и ребро клика (`WidgetState::clicked`)
   446→   446→    /// читают одно и то же состояние. Смена `disabled` между кадрами —
   447→   447→    /// через пересоздание или `state.set_disabled` потребителем.
   448→   448→    pub fn new(props: ButtonProps) -> Self {
   449→   449→        let mut state = WidgetState::default();
   450→   450→        state.set_disabled(props.disabled);
   451→   451→        Self { props, state }
   452→   452→    }
   453→   453→}

   455→   455→impl Component for Button {
   456→   456→    type Props = ButtonProps;
   457→   457→
   458→   458→    fn props(&self) -> &ButtonProps {
   459→   459→        &self.props
   460→   460→    }
   461→   461→
   462→   462→    /// Один rect — сама кнопка: делегирование [`button_layout`] 1:1
   463→   463→    /// (размер = подпись + 2·[`BUTTON_PAD_H`], минимум [`BUTTON_HEIGHT`];
   464→   464→    /// узкий слот — сжатие до ширины слота + ellipsis подписи). Порядок
   465→   465→    /// rects: `rects[0]` — кнопка; других rect'ов нет.
   466→   466→    ///
   467→   467→    /// Выравнивание в слоте — Center/Center (как у кнопок диалога/шапки);
   468→   468→    /// семейство/кегль — [`BUTTON_FAMILY`]/[`BUTTON_FONT_SIZE`].
   469→   469→    /// `backend` не участвует: `button_layout` — stack одного блока,
   470→   470→    /// контейнерной вёрстки нет (сигнатура kit-функции НЕ менялась —
   471→   471→    /// §Контракт-1 FR-068).
   472→   472→    ///
   473→   473→    /// Standalone-семантика: измеритель и шрифтовой пул создаются на
   474→   474→    /// вызов (immediate, без retained-состояния — решение FR-068 §W3);
   475→   475→    /// горячий путь потребителя — [`button_layout`] с собственным пулом
   476→   476→    /// (владелец пула — canvas-render::text, PRD-0009 §14/Q6).
   477→   477→    fn layout(&self, _backend: &dyn LayoutBackend, slot: UiRect) -> Vec<UiRect> {
   478→   478→        let mut m = TextMeasurer::new();
   479→   479→        let mut fs = cosmic_text::FontSystem::new();
   480→   480→        let bl = button_layout(
   481→   481→            slot,
   482→   482→            &self.props.label,
   483→   483→            (HAlign::Center, VAlign::Center),
   484→   484→            &mut m,
   485→   485→            &mut fs,
   486→   486→            BUTTON_FAMILY,
   487→   487→            BUTTON_FONT_SIZE,
   488→   488→        );
   489→   489→        vec![bl.rect]
   490→   490→    }
   491→   491→
   492→   492→    /// Рисует `rects[0]` (rect из [`Component::layout`]) контролом:
   493→   493→    /// заливка/рамка/радиус — [`button_style`] по варианту и состоянию
   494→   494→    /// [`WidgetState::kit_state`] (disabled прошит в state при
   495→   495→    /// [`Button::new`]). Подпись — забота потребителя (`Painter::label`
   496→   496→    /// слотом `style.text`; усечённая подпись живёт в [`button_layout`],
   497→   497→    /// кит цвет от геометрии не отделяет — F-8). Пустой `rects` — no-op
   498→   498→    /// (деградация, не паника).
   499→   499→    fn paint(&self, painter: &mut Painter, rects: &[UiRect]) {
   500→   500→        let Some(&rect) = rects.first() else {
   501→   501→            return;
   502→   502→        };
   503→   503→        let style = button_style(
   504→   504→            self.props.variant,
   505→   505→            self.state.kit_state(),
   506→   506→            &self.props.palette,
   507→   507→        );
   508→   508→        painter.control(rect, &style);
   509→   509→    }
   510→   510→
   511→   511→    // hit_test — дефолтный из [`Component`] (первый rect, содержащий
   512→   512→    // точку): у кнопки rect один — этого достаточно, переопределение
   513→   513→    // не требуется (план FR-068 §W3).
   514→   514→}

   838→   838→    #[test]
   839→   839→    fn component_button_layout_single_rect_in_slot() {
   840→   840→        let slot = UiRect::new(10.0, 20.0, 400.0, 60.0);
   841→   841→        let btn = Button::new(ButtonProps {
   842→   842→            label: "ОО".to_owned(),
   843→   843→            variant: ButtonVariant::Primary,
   844→   844→            palette: palette_a(),
   845→   845→            disabled: false,
   846→   846→        });
   847→   847→        let rects = btn.layout(default_backend(), slot);
   848→   848→        assert_eq!(rects.len(), 1, "кнопка — один rect");
   849→   849→        let r = rects[0];
   850→   850→        assert_eq!(r.h, BUTTON_HEIGHT);
   851→   851→        assert!(r.w >= BUTTON_HEIGHT, "ширина не ниже минимума");
   852→   852→        assert!(
   853→   853→            r.x >= slot.x && r.right() <= slot.right() + 0.01,
   854→   854→            "в слоте по X"

   857→   857→            r.y >= slot.y && r.bottom() <= slot.bottom() + 0.01,
   858→   858→            "в слоте по Y"

   862→   862→        assert!((r.y - slot.y - (slot.h - r.h) / 2.0).abs() < 0.01);
   863→   863→        // Пустая подпись — квадрат BUTTON_HEIGHT×BUTTON_HEIGHT (без
   864→   864→        // шейпинга: ширина текста 0 — детерминировано на любом CI).
   865→   865→        let empty = Button::new(ButtonProps {
   866→   866→            label: String::new(),
   867→   867→            ..btn.props.clone()
   868→   868→        });
   869→   869→        let rects = empty.layout(default_backend(), slot);
   870→   870→        assert_eq!(rects[0].w, BUTTON_HEIGHT);
   871→   871→        assert_eq!(rects[0].h, BUTTON_HEIGHT);
   872→   872→    }
