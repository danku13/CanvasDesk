     1→//! FR-045 R-5 (приоритет владельца, сессия 2026-09-21) — квалифицированная
     2→//! адресация «Объект.Поле»: **единственная точка** сборки отображаемых
     3→//! путей входных параметров. FR-044 (§Changes-1) переиспользует этот модуль
     4→//! для пилюль веера main stage и панели «Как считается» — определение
     5→//! `QualifiedRef` живёт здесь, а не в `bundles.rs` (сметчивание 2026-09-21).
     6→//!
     7→//! Правила (FR-045 §Решения Р-5, проверены на прототипе):
     8→//! - объект = имя ноды-истока ([`node_display_name`]; data-нода FR-045 R-2 —
     9→//!   имя ноды-объекта, колонка приходит через `fromOutput`);
    10→//! - поле = имя выхода (`fromOutput`) | «строка N» (`fromLine`, 1-based для
    11→//!   отображения; полный текст — в `tooltip`) | fallback `edge.id`
    12→//!   (ребро без адресации — значение ноды целиком);
    13→//! - имена отображаются **дословно** (нормализация/транслитерация не
    14→//!   выполняются); коллизия имён нод — fallback «Имя (node_id)» (§Q2);
    15→//! - display-level: легаси-синтаксис (`$1`, `$параметр`) при отображении
    16→//!   подставляется путями ([`formula_displays`]); именованный синтаксис
    17→//!   Numi «Объект.Поле» — в исходнике с FR-050 Р-6 (решение владельца
    18→//!   «сразу вариант Б», FR-044 §Q1 закрыт): пути резолвит вычислитель
    19→//!   (`Env.qualified`), а [`formula_displays`] читает их же для
    20→//!   рёбер-операндов подсветки (FR-044 Р-5).
    21→//!
    22→//! Чистый Rust, без I/O и глобального состояния; wasm-гейт (ADR-0011).
    23→
    24→use std::collections::BTreeMap;
    25→
    26→use crate::expr::NumiLineKind;

    28→use crate::model::CanvasdeskExt;

    31→/// Квалифицированный путь «Объект.Поле» (FR-045 Р-5, FR-044 Р-3).
    32→#[derive(Debug, Clone, PartialEq, Eq)]
    33→pub struct QualifiedRef {
    34→    /// Объект — имя ноды-истока (дословно; коллизия — «Имя (node_id)»).
    35→    pub obj: String,
    36→    /// Поле — выход (`fromOutput`), «строка N» (`fromLine`) или `edge.id`.
    37→    pub field: String,
    38→    /// Дополнительные данные для тултипа (FR-044 §Проверка: `fromLine` →
    39→    /// полное «строка N»; короткая форма на канвасе — полный путь).
    40→    pub tooltip: Option<String>,
    41→}
    42→
    43→impl QualifiedRef {
    44→    /// Полный путь `Объект.Поле` — для stage, формул и тултипов (FR-044 Р-3).
    45→    pub fn path(&self) -> String {
    46→        format!("{}.{}", self.obj, self.field)
    47→    }
    48→
    49→    /// Короткая форма «Поле» — для плотного канваса вне stage (компромисс
    50→    /// FR-044 Р-3: полный путь — в [`QualifiedRef::path`]/тултипе).
    51→    pub fn short(&self) -> &str {
    52→        &self.field
    53→    }
    54→}
    55→
    56→/// Отображаемое имя ноды (дословно, FR-045 Р-5): `label` (программное имя —
    57→/// по нему формулы ссылаются на объект, MUST быть первым) → явный заголовок
    58→/// (FR-072, волна 1) → имя снимка шаблона → первая непустая строка текста →
    59→/// `id`. Детерминировано, без нормализации.
    60→pub fn node_display_name(canvas: &Canvas, node_id: &str) -> String {
    61→    let Some(node) = canvas.node(node_id) else {
    62→        return node_id.to_owned();
    63→    };
    64→    // Волна 1 (решение владельца): явный заголовок (FR-072) — ВТОРОЙ в
    65→    // цепочке (после label): qualified-пути тултипов показывают то же имя,
    66→    // что шапка карточки; label главнее — по нему формулы ссылаются на
    67→    // объект («Статьи.marketing»): миграция заголовка (title = «Статьи
    68→    // бюджета») не должна рвать resolution ссылок. Фолбэк «первая непустая
    69→    // строка» — последний рубеж для не-мигрированных контекстов (MCP);
    70→    // шапкой он больше не становится (cards::title_for).
    71→    if let Some(label) = node.label.as_deref() {
    72→        if !label.trim().is_empty() {
    73→            return label.to_owned();
    74→        }
    75→    }
    76→    if let Some(title) = node.title() {
    77→        if !title.trim().is_empty() {
    78→            return title.to_owned();
    79→        }
    80→    }
    81→    if let Some(name) = node.template().and_then(|t| t.name) {
    82→        if !name.trim().is_empty() {
    83→            return name;
    84→        }
    85→    }
    86→    if let Some(text) = node.text.as_deref() {
    87→        if let Some(line) = text.lines().map(str::trim).find(|l| !l.is_empty()) {
    88→            return line.to_owned();
    89→        }
    90→    }
    91→    node.id.clone()
    92→}
    93→
    94→/// Счётчик отображаемых имён (для детекции коллизий §Q2): имя → сколько нод
    95→/// его носят. Публичен для переиспользования рендером (один обход на кадр).
    96→pub fn display_name_counts(canvas: &Canvas) -> BTreeMap<String, usize> {
    97→    let mut counts: BTreeMap<String, usize> = BTreeMap::new();
    98→    for node in &canvas.nodes {
    99→        *counts
   100→            .entry(node_display_name(canvas, &node.id))
   101→            .or_insert(0) += 1;
   102→    }
   103→    counts
   104→}
   105→
   106→/// Объект для [`display_ref`]: имя ноды с разрешением коллизии — если
   107→/// имя носят несколько нод, добавляется ` (node_id)` (FR-045 §Q2, базово).
   108→pub fn qualified_obj_name(
   109→    canvas: &Canvas,
   110→    node_id: &str,
   111→    counts: &BTreeMap<String, usize>,
   112→) -> String {
   113→    let name = node_display_name(canvas, node_id);
   114→    if counts.get(&name).copied().unwrap_or(0) > 1 {
   115→        format!("{name} ({node_id})")
   116→    } else {
   117→        name
   118→    }
   119→}
   120→
   121→/// Единая точка сборки отображаемого пути value-ребра (FR-044 §Changes-1,
   122→/// тесты §Проверка; правила — FR-045 Р-5). `edge_index` вне диапазона —
   123→/// пустой путь (без паники; чистая функция не падает).
   124→pub fn display_ref(canvas: &Canvas, edge_index: usize) -> QualifiedRef {
   125→    let Some(edge) = canvas.edges.get(edge_index) else {
   126→        return QualifiedRef {
   127→            obj: String::new(),
   128→            field: String::new(),
   129→            tooltip: None,
   130→        };
   131→    };
   132→    display_ref_for_edge(canvas, edge, &display_name_counts(canvas))
   133→}
   134→
   135→/// То же, но с переиспользованием готового счётчика имён (горячий путь
   136→/// рендера: один [`display_name_counts`] на кадр вместо обхода на ребро).
   137→pub fn display_ref_for_edge(
   138→    canvas: &Canvas,
   139→    edge: &Edge,
   140→    counts: &BTreeMap<String, usize>,
   141→) -> QualifiedRef {
   142→    let obj = qualified_obj_name(canvas, &edge.from_node, counts);
   143→    // Приоритет адресации — как в flow::edge_source_value (FR-025/FR-029):
   144→    // fromLine → fromOutput → значение ноды целиком.
   145→    if let Some(line) = edge.from_line {
   146→        return QualifiedRef {
   147→            obj,
   148→            // Отображение 1-based (человек считает строки с единицы);
   149→            // в модели `from_line` — индекс листа (0-based).
   150→            field: format!("строка {}", line + 1),
   151→            tooltip: Some(format!("строка {}", line + 1)),
   152→        };
   153→    }
   154→    if let Some(name) = &edge.from_output {
   155→        return QualifiedRef {
   156→            obj,
   157→            field: name.clone(),
   158→            tooltip: None,
   159→        };
   160→    }
   161→    // Ребро без адресации: значение ноды целиком — fallback `edge.id`
   162→    // (FR-044 Р-3/инвариант 5: «безымянный выход → edge.id»).
   163→    QualifiedRef {
   164→        obj,
   165→        field: edge.id.clone(),
   166→        tooltip: None,
   167→    }
   168→}
   169→
   170→/// Вход приёмника для групп «Переменные» (FR-044 Р-4, FR-045 Р-4):
   171→/// ребро + его позиция в адресации + квалифицированный путь.
   172→#[derive(Debug, Clone, PartialEq, Eq)]
   173→pub struct InputRef {
   174→    /// Индекс ребра в `canvas.edges`.
   175→    pub edge_index: usize,
   176→    /// Позиционный слот `$N` (0-based индекс среди позиционных рёбер);
   177→    /// `None` — проливание в параметр.
   178→    pub slot: Option<usize>,
   179→    /// Имя параметра (`toParam`); `None` — позиционный слот.
   180→    pub param: Option<String>,
   181→    /// Квалифицированный путь истока.
   182→    pub r: QualifiedRef,
   183→}
   184→
   185→/// Все входящие value-рёбра ноды с квалифицированными путями — порядок
   186→/// `canvas.edges` (детерминирован); позиционные и проливания в параметры.
   187→/// Единственный источник группы «Переменные · входящие значения» для
   188→/// панели main stage (FR-044 Р-4) и рейлов ноды (FR-045 Р-4).
   189→pub fn input_refs(canvas: &Canvas, node_id: &str) -> Vec<InputRef> {
   190→    let counts = display_name_counts(canvas);
   191→    let mut slot: usize = 0;
   192→    canvas
   193→        .edges
   194→        .iter()
   195→        .enumerate()
   196→        .filter(|(_, edge)| {
   197→            edge.to_node == node_id && edge.flow_kind() == crate::flow::FlowKind::Value
   198→        })
   199→        .map(|(edge_index, edge)| {
   200→            let (slot_no, param) = if edge.to_param.is_none() {
   201→                let no = slot;
   202→                slot += 1;
   203→                (Some(no), None)
   204→            } else {
   205→                (None, edge.to_param.clone())
   206→            };
   207→            InputRef {
   208→                edge_index,
   209→                slot: slot_no,
   210→                param,
   211→                r: display_ref_for_edge(canvas, edge, &counts),
   212→            }
   213→        })
   214→        .collect()
   215→}

   217→/// Строка расчёта приёмника (FR-044 Р-4/Р-5, FR-045 Р-4): исходник,
   218→/// display-подстановка квалифицированных путей вместо `$N`/`$параметр`
   219→/// и рёбра-операнды (для подсветки формула⇄переменные⇄рёбра, FR-044 Р-5).
   220→#[derive(Debug, Clone, PartialEq, Eq)]
   221→pub struct FormulaDisplay {
   222→    /// Индекс строки тела (для template-ноды — 0: формула снимка).
   223→    pub line: usize,
   224→    /// Исходный текст строки (дословно).
   225→    pub raw: String,
   226→    /// Текст с подставленными путями `Объект.Поле`; неразрешённые токены
   227→    /// (`$N` вне диапазона, параметр без рёбер) остаются как есть.
   228→    pub display: String,
   229→    /// Индексы рёбер-операндов в порядке первого упоминания (без дублей).
   230→    pub operand_edges: Vec<usize>,
   231→}
   232→
   233→/// Строки расчёта ноды (группа «Расчёт · формулы», FR-044 Р-4):
   234→/// - template-нода (снимок с непустым `expr`) — одна строка: формула снимка;
   235→/// - текстовая нода — строки тела с родом `Assignment`/`Expression`
   236→///   ([`NumiLineKind`], детектор FR-021) в порядке тела; проза пропускается.
   237→///
   238→/// Детерминировано; подстановка — display-level, исходник не меняется.
   239→///
   240→/// FR-044 Р-5 (именованный синтаксис, решение владельца 2026-09-22 —
   241→/// FR-050 Р-6 «сразу вариант Б», Q1 закрыт): операндами становятся не
   242→/// только `$`-токены (`$N`/`$in`/`$параметр`), но и qualified-пути
   243→/// исходника «Объект.Поле» — матчинг по всем адресным формам имени
   244→/// истока × поле (`flow::QualifiedNames::edge_keys` — те же ключи, что
   245→/// резолвит `Env.qualified`; последнее value-ребро побеждает — зеркало
   246→/// insert в `inbound_values`). Локальные переменные присваиваний (bare-
   247→/// идентификаторы без `.Поля`) и числа (`0.6`) операндами не являются;
   248→/// неразрешённый путь остаётся в display как написан (диагностика Р-3).
   249→pub fn formula_displays(canvas: &Canvas, node_id: &str) -> Vec<FormulaDisplay> {
   250→    let Some(node) = canvas.node(node_id) else {
   251→        return Vec::new();
   252→    };
   253→    let counts = display_name_counts(canvas);
   254→    // Карта адресации приёмника: позиционные слоты и проливания.
   255→    let mut slot_edges: Vec<usize> = Vec::new();
   256→    let mut param_edges: BTreeMap<String, Vec<usize>> = BTreeMap::new();
   257→    // FR-044 Р-5: qualified-ключи value-рёбер приёмника → индекс ребра.
   258→    let obj_names = crate::flow::QualifiedNames::build(canvas);
   259→    let mut qkeys: std::collections::HashMap<(String, String), usize> =
   260→        std::collections::HashMap::new();
   261→    for (index, edge) in canvas.edges.iter().enumerate() {
   262→        if edge.to_node != node_id || edge.flow_kind() != crate::flow::FlowKind::Value {
   263→            continue;
   264→        }
   265→        match &edge.to_param {
   266→            None => slot_edges.push(index),
   267→            Some(name) => param_edges.entry(name.clone()).or_default().push(index),
   268→        }
   269→        for key in obj_names.edge_keys(canvas, edge) {
   270→            qkeys.insert(key, index);
   271→        }
   272→    }
   273→    // Единый проход по строке: `$`-токены (слот/`$in`/параметр — прежняя
   274→    // семантика) и qualified-пути именованного синтаксиса; операнды — в
   275→    // порядке первого упоминания без дублей (документированный контракт).
   276→    let render = move |raw: &str, operands: &mut Vec<usize>| -> String {
   277→        let chars: Vec<char> = raw.chars().collect();
   278→        let mut out = String::with_capacity(raw.len() + 16);
   279→        let mut i = 0;
   280→        while i < chars.len() {
   281→            if chars[i] != '$' {
   282→                // FR-044 Р-5: qualified-путь «Объект.Поле» (не цифра —
   283→                // числа с точкой не пути; bare-идентификатор — локальная
   284→                // переменная/функция, не операнд)
   285→                if is_base_char(chars[i]) && !chars[i].is_ascii_digit() {
   286→                    if let Some((obj, field, next)) = scan_qualified(&chars, i) {
   287→                        if let Some(&index) = qkeys.get(&(obj, field)) {
   288→                            if !operands.contains(&index) {
   289→                                operands.push(index);
   290→                            }
   291→                        }
   292→                        // display: путь остаётся как написан
   293→                        while i < next {
   294→                            out.push(chars[i]);
   295→                            i += 1;
   296→                        }
   297→                        continue;
   298→                    }
   299→                    // plain идентификатор — копируем как есть
   300→                    let mut j = i + 1;
   301→                    while j < chars.len() && is_base_char(chars[j]) {
   302→                        j += 1;
   303→                    }
   304→                    while i < j {
   305→                        out.push(chars[i]);
   306→                        i += 1;
   307→                    }
   308→                    continue;
   309→                }
   310→                out.push(chars[i]);
   311→                i += 1;
   312→                continue;
   313→            }
   314→            // Токен: максимум цифр; если после цифр идёт буква/`_`/`-` —
   315→            // это имя параметра, а не слот (детерминированное правило).
   316→            let mut j = i + 1;
   317→            while j < chars.len() && chars[j].is_ascii_digit() {
   318→                j += 1;
   319→            }
   320→            let digits_only = j > i + 1;
   321→            let ident_next = j < chars.len() && is_ident_char(chars[j]);
   322→            if digits_only && !ident_next {
   323→                let slot_no: usize = chars[i + 1..j]
   324→                    .iter()
   325→                    .collect::<String>()
   326→                    .parse()
   327→                    .unwrap_or(0);
   328→                let edge_index = if slot_no >= 1 {
   329→                    slot_edges.get(slot_no - 1).copied()
   330→                } else {
   331→                    None
   332→                };
   333→                match edge_index {
   334→                    Some(index) => {
   335→                        operands.push(index);
   336→                        out.push_str(
   337→                            &display_ref_for_edge(canvas, &canvas.edges[index], &counts).path(),
   338→                        );
   339→                    }
   340→                    None => {
   341→                        // Неразрешённый слот — токен остаётся как есть.
   342→                        out.push('$');
   343→                        out.push_str(&chars[i + 1..j].iter().collect::<String>());
   344→                    }
   345→                }
   346→                i = j;
   347→                continue;
   348→            }
   349→            // Имя параметра: `$` + идентификатор.
   350→            let mut k = i + 1;
   351→            while k < chars.len() && is_ident_char(chars[k]) {
   352→                k += 1;
   353→            }
   354→            if k > i + 1 {
   355→                let name: String = chars[i + 1..k].iter().collect();
   356→                // Спец-токен `$in` (FR-014): ровно одно входящее ребро.
   357→                if name == "in" && slot_edges.len() == 1 {
   358→                    let index = slot_edges[0];
   359→                    operands.push(index);
   360→                    out.push_str(
   361→                        &display_ref_for_edge(canvas, &canvas.edges[index], &counts).path(),
   362→                    );
   363→                    i = k;
   364→                    continue;
   365→                }
   366→                if let Some(edges) = param_edges.get(&name) {
   367→                    // Все рёбра параметра — операнды (последнее побеждает
   368→                    // в проливе; для подсветки значимы все — конфликты
   369→                    // предупреждаются отдельно, FR-029).
   370→                    for (n, index) in edges.iter().enumerate() {
   371→                        if n == 0 {
   372→                            out.push_str(
   373→                                &display_ref_for_edge(canvas, &canvas.edges[*index], &counts)
   374→                                    .path(),
   375→                            );
   376→                        }
   377→                        operands.push(*index);
   378→                    }
   379→                } else {
   380→                    out.push('$');
   381→                    out.push_str(&name);
   382→                }
   383→                i = k;
   384→                continue;
   385→            }
   386→            // Одиночный `$` без токена — литерал.
   387→            out.push('$');
   388→            i += 1;
   389→        }
   390→        out
   391→    };
   392→    // Источник строк: снимок шаблона или текст тела.
   393→    if let Some(expr) = node
   394→        .template()
   395→        .map(|t| t.expr)
   396→        .filter(|e| !e.trim().is_empty())
   397→    {
   398→        let mut operands = Vec::new();
   399→        let display = render(&expr, &mut operands);
   400→        dedup_keep_first(&mut operands);
   401→        return vec![FormulaDisplay {
   402→            line: 0,
   403→            raw: expr,
   404→            display,
   405→            operand_edges: operands,
   406→        }];
   407→    }
   408→    let Some(text) = node.text.as_deref() else {
   409→        return Vec::new();
   410→    };
   411→    text.lines()
   412→        .enumerate()
   413→        .filter(|(_, line)| {
   414→            matches!(
   415→                crate::expr::line_kind(line),
   416→                NumiLineKind::Assignment { .. } | NumiLineKind::Expression
   417→            )
   418→        })
   419→        .map(|(line, raw)| {
   420→            let mut operands = Vec::new();
   421→            let display = render(raw, &mut operands);
   422→            dedup_keep_first(&mut operands);
   423→            FormulaDisplay {
   424→                line,
   425→                raw: raw.to_owned(),
   426→                display,
   427→                operand_edges: operands,
   428→            }
   429→        })

   439→/// Базовый символ идентификатора — зеркало цикла `lex_ident` expr.rs:
   440→/// alphanumeric (включая кириллицу) + `_`; дефис НЕ входит (`a-b` —
   441→/// вычитание).
   442→fn is_base_char(c: char) -> bool {
   443→    c.is_alphanumeric() || c == '_'
   444→}

   446→/// FR-044 Р-5: сканировать qualified-путь с позиции `start` (первый символ
   447→/// — базовый, не цифра). Ветвь А: `Объект.Поле`; ветвь Б: алиас коллизии
   448→/// «Объект (N)»/«Объект (id)» + `.Поле` (§Q2). Поле — [`scan_field`].
   449→/// Возвращает `(объект, поле, позиция за путём)` — зеркало правил
   450→/// `lex_ident`/`try_qualified`/`lex_qualified_field` expr.rs (FR-050 Р-6),
   451→/// чтобы матчинг операндов читал ровно то, что резолвит вычислитель.
   452→fn scan_qualified(chars: &[char], start: usize) -> Option<(String, String, usize)> {
   453→    let mut j = start + 1;
   454→    while j < chars.len() && is_base_char(chars[j]) {
   455→        j += 1;
   456→    }
   457→    let base: String = chars[start..j].iter().collect();
   458→    // Ветвь А: `.Поле` сразу за идентификатором.
   459→    if let Some((field, next)) = scan_field(chars, j) {
   460→        return Some((base, field, next));
   461→    }
   462→    // Ветвь Б: суффикс ` (N)`/` (id)` (пробелы/табы допускаются) + `.Поле`.
   463→    let mut k = j;
   464→    while k < chars.len() && (chars[k] == ' ' || chars[k] == '\t') {
   465→        k += 1;
   466→    }
   467→    if k < chars.len() && chars[k] == '(' {
   468→        let open = k;
   469→        let mut m = open + 1;
   470→        while m < chars.len() && (chars[m].is_alphanumeric() || chars[m] == '_' || chars[m] == '-')
   471→        {
   472→            m += 1;
   473→        }
   474→        if m < chars.len() && chars[m] == ')' && m > open + 1 {
   475→            let content: String = chars[open + 1..m].iter().collect();
   476→            let mut p = m + 1;
   477→            while p < chars.len() && (chars[p] == ' ' || chars[p] == '\t') {
   478→                p += 1;
   479→            }
   480→            if let Some((field, next)) = scan_field(chars, p) {
   481→                return Some((format!("{base} ({content})"), field, next));
   482→            }
   483→        }
   484→    }
   485→    None
   486→}
   487→
   488→/// `.Поле` с позиции `dot` (`chars[dot] == '.'`); поле начинается с
   489→/// буквы/`_` (точка перед цифрой — дробное число, не путь), продолжается
   490→/// alphanumeric/`_`; дефис продолжается именем, если сразу за ним
   491→/// буква/цифра/`_` («Кол-во»; `Кол - во` — вычитание, поле «Кол»).
   492→/// Возвращает `(поле, позиция за полем)`.
   493→fn scan_field(chars: &[char], dot: usize) -> Option<(String, usize)> {
   494→    if chars.get(dot) != Some(&'.') {
   495→        return None;
   496→    }
   497→    let first = *chars.get(dot + 1)?;
   498→    if !(first.is_alphabetic() || first == '_') {
   499→        return None;
   500→    }
   501→    let mut j = dot + 2;
   502→    while j < chars.len() {
   503→        let c = chars[j];
   504→        let continues = c.is_alphanumeric()
   505→            || c == '_'
   506→            || (c == '-'
   507→                && chars
   508→                    .get(j + 1)
   509→                    .is_some_and(|n| n.is_alphanumeric() || *n == '_'));
   510→        if continues {
   511→            j += 1;
   512→        } else {
   513→            break;
   514→        }
   515→    }
   516→    Some((chars[dot + 1..j].iter().collect(), j))
   517→}
   518→
   519→fn is_ident_char(c: char) -> bool {
   520→    c.is_alphanumeric() || c == '_' || c == '-'
   521→}
