    19→/// Замер кнопки по подписи: ширина = текст + 2·BUTTON_PAD_H, высота = const.
    20→/// Пустая подпись → квадрат минимальной ширины BUTTON_HEIGHT.
    21→pub fn button_size(
    22→    label: &str,
    23→    m: &mut TextMeasurer,
    24→    fs: &mut cosmic_text::FontSystem,
    25→    family: &str,
    26→    size: f32,
    27→) -> UiVec2 {
    28→    let text_w = if label.is_empty() {
    29→        0.0
    30→    } else {
    31→        m.width_of(fs, label, family, size)
    32→    };
    33→    let w = (text_w + BUTTON_PAD_H * 2.0).max(BUTTON_HEIGHT);
    34→    UiVec2::new(w, BUTTON_HEIGHT)
    35→}
    36→
    37→/// Кнопка в слоте: измеренный размер, позиция — stack по выравниванию;
    38→/// подпись усекается `ellipsis`, если слот уже кнопки (деградация видна
    39→/// потребителю и линту, не молчаливый break).
    40→pub struct ButtonLayout {
    41→    /// Rect кнопки.
    42→    pub rect: UiRect,
    43→    /// Подпись после ellipsis (может отличаться от исходной).
    44→    pub label: String,
    45→}
    46→
    47→pub fn button_layout(
    48→    slot: UiRect,
    49→    label: &str,
    50→    align: (HAlign, VAlign),
    51→    m: &mut TextMeasurer,
    52→    fs: &mut cosmic_text::FontSystem,
    53→    family: &str,
    54→    size: f32,
    55→) -> ButtonLayout {
    56→    let desired = button_size(label, m, fs, family, size);
    57→    let max_w = slot.w;
    58→    // Деградация узкого слота: усечение по фактической ширине (не срез);
    59→    // кнопка не вылезает за слот — сжимается до его ширины.
    60→    let width = desired.x.min(max_w);
    61→    let shown = if desired.x > max_w {
    62→        m.ellipsis(
    63→            fs,
    64→            label,
    65→            family,
    66→            size,
    67→            (max_w - BUTTON_PAD_H * 2.0).max(0.0),
    68→        )
    69→    } else {
    70→        label.to_owned()
    71→    };
    72→    let rect = stack(slot, UiVec2::new(width, BUTTON_HEIGHT), align.0, align.1);
    73→    ButtonLayout { rect, label: shown }
    74→}
    75→
    76→/// Стиль кнопки: слот заливки по варианту, слот hover/pressed по состоянию,
    77→/// текст — слот по состоянию (disabled — свой слот). Никакой арифметики
    78→/// над цветами: только выбор слота.
    79→pub fn button_style(variant: ButtonVariant, state: KitState, p: &KitPalette) -> ControlStyle {
    80→    // Wave T §5.1.3: + Tertiary, Text, Inverse variants.
    81→    let base_fill = match variant {
    82→        ButtonVariant::Primary => p.control_primary,
    83→        ButtonVariant::Danger => p.control_danger,
    84→        ButtonVariant::Secondary | ButtonVariant::Ghost | ButtonVariant::Tertiary => p.control_fill,
    85→        // Text — прозрачный контейнер (M3 text button).
    86→        ButtonVariant::Text => [0.0, 0.0, 0.0, 0.0],
    87→        // Inverse — на цветном фоне; используем accent как «inverse» fill.
    88→        ButtonVariant::Inverse => p.accent,
    89→    };
    90→    let fill = match state {
    91→        KitState::Normal
    92→        | KitState::Selected
    93→        | KitState::Focused
    94→        | KitState::Dragged
    95→        | KitState::Error => base_fill,
    96→        KitState::Hovered => match variant {
    97→            ButtonVariant::Primary | ButtonVariant::Inverse => p.primary_hover_fill,
    98→            ButtonVariant::Secondary
    99→            | ButtonVariant::Ghost
   100→            | ButtonVariant::Tertiary
   101→            | ButtonVariant::Text
   102→            | ButtonVariant::Danger => p.hover_fill,
   103→        },
   104→        KitState::Pressed => match variant {
   105→            ButtonVariant::Primary | ButtonVariant::Inverse => p.primary_hover_fill,
   106→            _ => p.hover_fill,
   107→        },
   108→        KitState::Disabled => base_fill,
   109→    };
   110→    let border = match (variant, state) {
   111→        (ButtonVariant::Ghost | ButtonVariant::Text, KitState::Hovered | KitState::Pressed) => {
   112→            p.accent
   113→        }
   114→        (ButtonVariant::Ghost | ButtonVariant::Text, _) => [0.0, 0.0, 0.0, 0.0],
   115→        _ => p.control_border,
   116→    };
   117→    let text = match state {
   118→        KitState::Disabled => p.disabled_text,
   119→        _ => match variant {
   120→            ButtonVariant::Ghost | ButtonVariant::Text => p.text,
   121→            ButtonVariant::Inverse => p.text_title,
   122→            _ => p.text_title,
   123→        },
   124→    };
   125→    ControlStyle {
   126→        fill,
   127→        border,
   128→        text,
   129→        radius: canvas_core::tokens::RADIUS_CHIP,
   130→        elevation: super::Elevation::None,
   131→    }
   132→}

   153→/// Замер чипа: ширина = текст + 2·CHIP_PAD_H, высота = const.
   154→pub fn chip_size(
   155→    label: &str,
   156→    m: &mut TextMeasurer,
   157→    fs: &mut cosmic_text::FontSystem,
   158→    family: &str,
   159→    size: f32,
   160→) -> UiVec2 {
   161→    let text_w = m.width_of(fs, label, family, size);
   162→    UiVec2::new(text_w + CHIP_PAD_H * 2.0, CHIP_HEIGHT)
   163→}
   164→
   165→/// Чип от левого края `origin_x` (ряд чипов компонует потребитель через
   166→/// `Row`/SqueezeTail); усечение подписи — ellipsis по `max_width`.
   167→pub struct ChipLayout {
   168→    pub rect: UiRect,
   169→    pub label: String,
   170→}
   171→
   172→pub fn chip_layout(
   173→    origin: UiPoint,
   174→    label: &str,
   175→    max_width: f32,
   176→    m: &mut TextMeasurer,
   177→    fs: &mut cosmic_text::FontSystem,
   178→    family: &str,
   179→    size: f32,
   180→) -> ChipLayout {
   181→    let desired = chip_size(label, m, fs, family, size);
   182→    let shown = if desired.x > max_width {
   183→        m.ellipsis(
   184→            fs,
   185→            label,
   186→            family,
   187→            size,
   188→            (max_width - CHIP_PAD_H * 2.0).max(0.0),
   189→        )
   190→    } else {
   191→        label.to_owned()
   192→    };
   193→    let w = if shown == label { desired.x } else { max_width };
   194→    ChipLayout {
   195→        rect: UiRect::new(origin.x, origin.y, w, CHIP_HEIGHT),
   196→        label: shown,
   197→    }
   198→}

   200→/// Стиль чипа: selected → слот selected_fill, hovered → hover_fill,

   222→/// Раскладка Switch (тогла): трек (pill) + бегунок + стиль трека + заливка
   223→/// бегунка. `on` определяет позицию бегунка (вправо) и слот заливки трека.
   224→#[derive(Debug, Clone, Copy, PartialEq)]
   225→pub struct SwitchLayout {
   226→    /// Rect трека (pill).
   227→    pub track: UiRect,
   228→    /// Rect бегунка (квадрат внутри трека).
   229→    pub knob: UiRect,
   230→    /// Стиль трека — слот заливки/рамки/радиуса (RADIUS_PILL).
   231→    pub track_style: ControlStyle,
   232→    /// Заливка бегунка (слот `text_title`/`disabled_text`).
   233→    pub knob_fill: [f32; 4],
   234→}
   235→
   236→/// Switch в слоте: трек (pill) + бегунок. `on` — позиция бегунка и слот
   237→/// заливки трека (`control_primary` on / `control_fill` off); `state` —
   238→/// hover/pressed/disabled слоты; `knob_fill` — `text_title` (disabled —
   239→/// `disabled_text`).
   240→pub fn switch(slot: UiRect, on: bool, state: KitState, p: &KitPalette) -> SwitchLayout {
   241→    let track = stack(
   242→        slot,
   243→        UiVec2::new(SWITCH_W, SWITCH_H),
   244→        HAlign::Center,
   245→        VAlign::Center,
   246→    );
   247→    // Бегунок: квадрат side = SWITCH_H - 2·pad; позиция — влево/вправо по `on`.
   248→    let knob_side = (SWITCH_H - 2.0 * SWITCH_KNOB_PAD).max(0.0);
   249→    let knob_x = if on {
   250→        track.right() - SWITCH_KNOB_PAD - knob_side
   251→    } else {
   252→        track.x + SWITCH_KNOB_PAD
   253→    };
   254→    let knob_y = track.y + SWITCH_KNOB_PAD;
   255→    let knob = UiRect::new(knob_x, knob_y, knob_side, knob_side);
   256→    // Стиль трека: заливка — primary on / control_fill off; hover/pressed —
   257→    // свои слоты; disabled — базовая. Радиус RADIUS_PILL (pill).
   258→    let base_fill = if on {
   259→        p.control_primary
   260→    } else {
   261→        p.control_fill
   262→    };
   263→    let fill = match state {
   264→        KitState::Hovered | KitState::Pressed => {
   265→            if on {
   266→                p.primary_hover_fill
   267→            } else {
   268→                p.hover_fill
   269→            }
   270→        }
   271→        _ => base_fill,
   272→    };
   273→    let track_style = ControlStyle {
   274→        fill,
   275→        border: p.control_border,
   276→        text: p.text, // не используется у Switch (без подписи)
   277→        radius: canvas_core::tokens::RADIUS_PILL,
   278→        elevation: super::Elevation::None,
   279→    };
   280→    // Бегунок: text_title (white/light); disabled — приглушён.
   281→    let knob_fill = match state {
   282→        KitState::Disabled => p.disabled_text,
   283→        _ => p.text_title,
   284→    };
   285→    SwitchLayout {
   286→        track,
   287→        knob,
   288→        track_style,
   289→        knob_fill,
   290→    }
   291→}

   319→/// Глиф иконки — `&'static str` для `TextMeasurer`/`Painter::label`. Маппинг

   326→        Icon::Search => "⌕",
   327→        Icon::Plus => "+",
   328→        Icon::ArrowLeft => "←",
   329→        Icon::ArrowRight => "→",
   330→        Icon::Refresh => "↻",

   355→/// Кнопка «×» в правом-верхнем углу панели/модали (FR-070, audit §6.1 —
   356→/// самый повторяющийся паттерн: 7+ файлов дублировали ручную формулу
   357→/// `x = panel.right() - 28.0, y = panel.y + 6.0` с дрейфом inset/size
   358→/// между 22/24/26/28/30 — канонический размер [`ICON_BUTTON_SIZE`],
   359→/// inset — [`canvas_core::tokens::SPACING_SM`] = 8).
   360→///
   361→/// Возвращает `UiRect` угловой кнопки в слоте `slot` (panel/modal rect);
   362→/// позиция — «прижата к правому-верхнему углу слота» минус inset, размер —
   363→/// квадрат `ICON_BUTTON_SIZE × ICON_BUTTON_SIZE` (та же метрика, что у
   364→/// [`icon_button_rect`] с `HAlign::End, VAlign::Start`, но с явным inset —
   365→/// не вплотную к углу).
   366→///
   367→/// Контракт F-8: цвет/стиль кнопки — забота потребителя
   368→/// ([`icon_button_style`] с `KitState`/`KitPalette`); здесь — только
   369→/// геометрия (как у [`icon_button_rect`]).
   370→pub fn stage_close_button(slot: UiRect) -> UiRect {
   371→    let s = ICON_BUTTON_SIZE;
   372→    let inset = canvas_core::tokens::SPACING_SM;
   373→    UiRect::new(slot.right() - s - inset, slot.y + inset, s, s)
   374→}

   376→/// LG-вариант [`stage_close_button`] — 30×30 (вместо канонического
   377→/// [`ICON_BUTTON_SIZE`] = 26), тот же inset [`canvas_core::tokens::SPACING_SM`]
   378→/// = 8. Выделен для dialog-шапок с HEADER_H = 56–58 (autolink/explain —
   379→/// FR-060/FR-059 documented deviation): крупная кнопка визуально
   380→/// сбалансирована с высоким заголовком; канонический 26 выглядит мелко.
   381→///
   382→/// **Визуальный скачок при миграции** с hand-rolled (inset 14 от прежнего
   383→/// `panel.right() - 14 - 30`/`panel.y + 14`) на канон `SPACING_SM`=8 —
   384→/// позиция кнопки сдвигается на ~6px по диагонали (вверх+вправо, к углу
   385→/// панели). Это каноническое направление kit (FR-070/§6.1 — «× в углу
   386→/// панели с inset `SPACING_SM`»); прежний inset 14 был наследием ручной
   387→/// вертикальной центровки в HEADER_H=58 (где `(58−30)/2=14`). Когда
   388→/// `HEADER_H` мигрирует на `PANEL_HEADER_H_L`=44 (отдельный TODO W-d),
   389→/// `(44−30)/2`=7 — почти совпадает с `SPACING_SM`=8, и LG-вариант можно
   390→/// будет пересмотреть (возможно, свернуть обратно в [`stage_close_button`]).
   391→///
   392→/// Контракт F-8 (как у [`stage_close_button`]): цвет/стиль — забота
   393→/// потребителя (через [`icon_button_style`] с `KitState`/`KitPalette`);
   394→/// здесь — только геометрия.
   395→pub fn stage_close_button_lg(slot: UiRect) -> UiRect {
   396→    let s = 30.0;
   397→    let inset = canvas_core::tokens::SPACING_SM;
   398→    UiRect::new(slot.right() - s - inset, slot.y + inset, s, s)
   399→}

   430→/// Кнопка как [`Component`] — retained-объект (FR-068 §W3): `props` —
   431→/// вход кадра, `state` — машина состояний FR-057 (потребитель двигает
   432→/// `set_pointer`/`clicked` по событиям указателя; интерактивность —
   433→/// потому `WidgetState` обязателен).
   434→#[derive(Debug, Clone, PartialEq)]
   435→pub struct Button {
   436→    /// Свойства кадра.
   437→    pub props: ButtonProps,
   438→    /// Состояние виджета (hover/pressed/disabled/... — матрица
   439→    /// приоритетов [`WidgetState::kit_state`]).
   440→    pub state: WidgetState,
   441→}

   443→impl Button {
   444→    /// Новая кнопка: `props.disabled` прошивается в состояние виджета —
   445→    /// дальше стиль (`paint`) и ребро клика (`WidgetState::clicked`)
   446→    /// читают одно и то же состояние. Смена `disabled` между кадрами —
   447→    /// через пересоздание или `state.set_disabled` потребителем.
   448→    pub fn new(props: ButtonProps) -> Self {
   449→        let mut state = WidgetState::default();
   450→        state.set_disabled(props.disabled);
   451→        Self { props, state }
   452→    }

   456→    type Props = ButtonProps;
   457→
   458→    fn props(&self) -> &ButtonProps {
   459→        &self.props
   460→    }
   461→
   462→    /// Один rect — сама кнопка: делегирование [`button_layout`] 1:1
   463→    /// (размер = подпись + 2·[`BUTTON_PAD_H`], минимум [`BUTTON_HEIGHT`];
   464→    /// узкий слот — сжатие до ширины слота + ellipsis подписи). Порядок
   465→    /// rects: `rects[0]` — кнопка; других rect'ов нет.
   466→    ///
   467→    /// Выравнивание в слоте — Center/Center (как у кнопок диалога/шапки);
   468→    /// семейство/кегль — [`BUTTON_FAMILY`]/[`BUTTON_FONT_SIZE`].
   469→    /// `backend` не участвует: `button_layout` — stack одного блока,
   470→    /// контейнерной вёрстки нет (сигнатура kit-функции НЕ менялась —
   471→    /// §Контракт-1 FR-068).
   472→    ///
   473→    /// Standalone-семантика: измеритель и шрифтовой пул создаются на
   474→    /// вызов (immediate, без retained-состояния — решение FR-068 §W3);
   475→    /// горячий путь потребителя — [`button_layout`] с собственным пулом
   476→    /// (владелец пула — canvas-render::text, PRD-0009 §14/Q6).
   477→    fn layout(&self, _backend: &dyn LayoutBackend, slot: UiRect) -> Vec<UiRect> {
   478→        let mut m = TextMeasurer::new();
   479→        let mut fs = cosmic_text::FontSystem::new();
   480→        let bl = button_layout(
   481→            slot,
   482→            &self.props.label,
   483→            (HAlign::Center, VAlign::Center),
   484→            &mut m,
   485→            &mut fs,
   486→            BUTTON_FAMILY,
   487→            BUTTON_FONT_SIZE,
   488→        );
   489→        vec![bl.rect]
   490→    }
   491→
   492→    /// Рисует `rects[0]` (rect из [`Component::layout`]) контролом:
   493→    /// заливка/рамка/радиус — [`button_style`] по варианту и состоянию
   494→    /// [`WidgetState::kit_state`] (disabled прошит в state при
   495→    /// [`Button::new`]). Подпись — забота потребителя (`Painter::label`
   496→    /// слотом `style.text`; усечённая подпись живёт в [`button_layout`],
   497→    /// кит цвет от геометрии не отделяет — F-8). Пустой `rects` — no-op
   498→    /// (деградация, не паника).
   499→    fn paint(&self, painter: &mut Painter, rects: &[UiRect]) {
   500→        let Some(&rect) = rects.first() else {
   501→            return;
   502→        };
   503→        let style = button_style(
   504→            self.props.variant,
   505→            self.state.kit_state(),
   506→            &self.props.palette,
   507→        );
   508→        painter.control(rect, &style);
   509→    }
   510→
   511→    // hit_test — дефолтный из [`Component`] (первый rect, содержащий
   512→    // точку): у кнопки rect один — этого достаточно, переопределение
   513→    // не требуется (план FR-068 §W3).
   514→}
