   122→/// Состояние тура: индекс текущего шага (кламп `0..len-1` — инвариант
   123→/// карусели). `None` в `App` — тура нет.
   124→#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
   125→pub struct OnboardingState {
   126→    pub step: usize,
   127→}

   129→impl OnboardingState {
   130→    /// Число шагов.
   131→    pub fn len(&self) -> usize {
   132→        ONBOARDING_STEPS.len()
   133→    }
   134→
   135→    /// Тур пуст (недостижимо с непустым ONBOARDING_STEPS — для полноты).
   136→    pub fn is_empty(&self) -> bool {
   137→        ONBOARDING_STEPS.is_empty()
   138→    }
   139→
   140→    /// Шаг вперёд; false на последнем шаге (вызывающий завершает тур —
   141→    /// «Готово» ставит `onboarding_done` и сохраняет конфиг).
   142→    // Не Iterator::next — шаги карусели с клампом, а не выдача элементов
   143→    #[allow(clippy::should_implement_trait)]
   144→    pub fn next(&mut self) -> bool {
   145→        if self.step + 1 < ONBOARDING_STEPS.len() {
   146→            self.step += 1;
   147→            true
   148→        } else {
   149→            false
   150→        }
   151→    }
   152→
   153→    /// Шаг назад; false на первом (кнопки «Назад» там нет — hit-тест None).
   154→    pub fn prev(&mut self) -> bool {
   155→        if self.step == 0 {
   156→            return false;
   157→        }
   158→        self.step -= 1;
   159→        true
   160→    }
   161→
   162→    /// Последний шаг (подпись кнопки меняется на «Готово»).
   163→    pub fn is_last(&self) -> bool {
   164→        self.step + 1 >= ONBOARDING_STEPS.len()
   165→    }
   166→
   167→    /// Ключ подписи правой кнопки: «Далее» / «Готово» на последнем шаге
   168→    /// (текст — таблица [`crate::i18n`], FR-040). CR-031: на финальном шаге
   169→    /// «Далее» не рисуется (футер — только «Назад»), выбор — CTA-опции.
   170→    pub fn next_label_key(&self) -> &'static str {
   171→        if self.is_last() {
   172→            keys::ONBOARDING_DONE
   173→        } else {
   174→            keys::ONBOARDING_NEXT
   175→        }
   176→    }
   177→
   178→    /// Ключ подписи левой кнопки: «Назад» / нет на первом шаге.
   179→    pub fn prev_label_key(&self) -> Option<&'static str> {
   180→        (self.step > 0).then_some(keys::ONBOARDING_BACK)
   181→    }
   182→}

   231→/// Высота полноширинной CTA-опции финального шага (CR-031).
   232→pub const ONBOARDING_OPTION_H: f32 = 34.0;
   233→/// Зазор между CTA-опциями финала.
   234→pub const ONBOARDING_OPTION_GAP: f32 = kit::GAP_CONTROLS;
   235→/// Зона финальных CTA над футером: верхний зазор + 2 опции + зазор
   236→/// (резервируется из зоны тела — опции всегда видимы, тело скроллится выше).
   237→pub const ONBOARDING_OPTION_ZONE: f32 = 10.0 + 2.0 * ONBOARDING_OPTION_H + ONBOARDING_OPTION_GAP;
   238→
   239→/// Rect'ы полноширинных CTA-опций финального шага (CR-031): две кнопки
   240→/// над футером, ширина — внутренняя ширина текстовой колонки (S2).
   241→/// Порядок: «Открыть шаблонную схему» (primary), «Начать самому» (secondary).
   242→pub fn option_rects(card: [f32; 4]) -> [[f32; 4]; 2] {
   243→    let col = text_column(card);
   244→    let w = (col[2] - ONBOARDING_PAD * 2.0).max(0.0);
   245→    let bottom = col[1] + col[3] - ONBOARDING_PAD - ONBOARDING_FOOTER_H;
   246→    let gallery_y = bottom - 2.0 * ONBOARDING_OPTION_H - ONBOARDING_OPTION_GAP;
   247→    let empty_y = bottom - ONBOARDING_OPTION_H;
   248→    [
   249→        [col[0] + ONBOARDING_PAD, gallery_y, w, ONBOARDING_OPTION_H],
   250→        [col[0] + ONBOARDING_PAD, empty_y, w, ONBOARDING_OPTION_H],
   251→    ]
   252→}

   254→/// Строки тела шага после переноса по ширине карточки — измеренный
   255→/// `TextMeasurer::wrap` (W-e, CR-015: замена эвристики
   256→/// «символы × 0.62»; те же метрики, что у шейпинга рендера). Один источник
   257→/// для высоты карточки ([`card_layout`]) и рендера (overlays.rs) —
   258→/// раскладка и геометрия не разъезжаются.
   259→pub fn body_lines(
   260→    step: usize,
   261→    width: f32,
   262→    language: Language,
   263→    role: &str,
   264→    m: &mut TextMeasurer,
   265→    fs: &mut cosmic_text::FontSystem,
   266→) -> Vec<String> {
   267→    if ONBOARDING_STEPS.get(step).is_none() {
   268→        return Vec::new();
   269→    };
   270→    // CR-031/S2: перенос считается по текстовой колонке (правая половина
   271→    // в двухколоночном режиме) — один источник с [`body_area`]/рендером.
   272→    // CR-031/S3: тело шага 1 — по роли.
   273→    let col_w = text_column([0.0, 0.0, width, 0.0])[2];
   274→    let avail = (col_w - ONBOARDING_PAD * 2.0).max(10.0);
   275→    let wrapped = m.wrap(
   276→        fs,
   277→        i18n::tr(language, step_body_key(step, role)),
   278→        FAMILY,
   279→        ONBOARDING_BODY_FONT,
   280→        avail,
   281→    );
   282→    // W-e follow-up (CI windows): сверхширокий токен без переносов
   283→    // («markdown-разметкой» на Windows-метриках — 226.6 > бюджета 192)
   284→    // выходил за карточку и на рендере, и в тесте бюджета. Жёсткий
   285→    // посимвольный разрыв в бюджет: строка либо влезает, либо рвётся
   286→    // глиф за глифом (тексты онбординга — кириллица/латиница).
   287→    let mut lines: Vec<String> = Vec::with_capacity(wrapped.len());
   288→    for line in wrapped {
   289→        if m.width_of(fs, &line, FAMILY, ONBOARDING_BODY_FONT) <= avail {
   290→            lines.push(line);
   291→            continue;
   292→        }
   293→        let mut cur = String::new();
   294→        for ch in line.chars() {
   295→            let mut next = cur.clone();
   296→            next.push(ch);
   297→            if m.width_of(fs, &next, FAMILY, ONBOARDING_BODY_FONT) > avail && !cur.is_empty() {
   298→                lines.push(std::mem::take(&mut cur));
   299→            }
   300→            cur.push(ch);
   301→        }
   302→        if !cur.is_empty() {
   303→            lines.push(cur);
   304→        }
   305→    }
   306→    lines
   307→}

   309→/// Отступ от верха карточки до первой строки тела (заголовок +
   310→/// прогресс-точки + зазор) — общий для геометрии и рендера.
   311→pub fn body_top_offset() -> f32 {
   312→    ONBOARDING_PAD + ONBOARDING_TITLE_LINE_H + ONBOARDING_DOTS_TOP + ONBOARDING_DOT + 10.0
   313→}
   314→
   315→/// Текстовая колонка карточки (CR-031/S2): в двухколоночном режиме — правая
   316→/// половина (иллюстрация — левая, см. [`illustration_rect`]); в фолбэке —
   317→/// вся карточка. Все текстово-кнопочные зоны (заголовок, точки, тело,
   318→/// футер, финальные CTA) считаются от этой колонки — «50% картинка,
   319→/// 50% текст с кнопками».
   320→pub fn text_column(card: [f32; 4]) -> [f32; 4] {
   321→    if card[2] >= ONBOARDING_SPLIT_MIN_W {
   322→        [card[0] + card[2] / 2.0, card[1], card[2] / 2.0, card[3]]
   323→    } else {
   324→        card
   325→    }
   326→}
   327→
   328→/// Зона иллюстрации шага (CR-031/S2): левая половина карточки минус пад —
   329→/// только в двухколоночном режиме ([`text_column`]); в фолбэке — None
   330→/// (карточка узка, иллюстрация не помещается).
   331→pub fn illustration_rect(card: [f32; 4]) -> Option<[f32; 4]> {
   332→    if card[2] >= ONBOARDING_SPLIT_MIN_W {
   333→        Some([
   334→            card[0] + ONBOARDING_PAD,
   335→            card[1] + ONBOARDING_PAD,
   336→            card[2] / 2.0 - ONBOARDING_PAD * 2.0,
   337→            (card[3] - ONBOARDING_PAD * 2.0).max(0.0),
   338→        ])
   339→    } else {
   340→        None
   341→    }
   342→}
   343→
   344→/// Зона тела шага в карточке: от якоря первой строки до футера с CTA
   345→/// (минус нижний пад). Общий источник высоты окна видимости скролла для
   346→/// раскладки ([`card_layout`]), колеса ввода и отрисовки (видимые строки).
   347→/// CR-031: на финальном шаге снизу резервируется зона CTA-опций
   348→/// ([`ONBOARDING_OPTION_ZONE`]) — опции всегда видимы, тело скроллится выше.
   349→pub fn body_area(card: [f32; 4], final_step: bool) -> [f32; 4] {
   350→    let col = text_column(card);
   351→    let mut h = (col[3] - body_top_offset() - ONBOARDING_PAD - ONBOARDING_FOOTER_H).max(0.0);
   352→    if final_step {
   353→        h = (h - ONBOARDING_OPTION_ZONE).max(0.0);
   354→    }
   355→    [
   356→        col[0] + ONBOARDING_PAD,
   357→        col[1] + body_top_offset(),
   358→        (col[2] - ONBOARDING_PAD * 2.0).max(0.0),
   359→        h,
   360→    ]
   361→}

   363→/// Раскладка карточки шага (W-e): слот = вьюпорт минус поля
   364→/// ([`ONBOARDING_VIEWPORT_MARGIN`]), панель — `kit::modal` (центр слота,
   365→/// как dialog/autolink); ширина — кламп к слоту, высота — по измеренному
   366→/// телу с клампом к слоту (инвариант FR-028 «карточка целиком в окне» —
   367→/// теперь ещё и с полями). При клампе высоты контент ужимается:
   368→/// скролл-состояние тела синхронизируется с контентом/окном (футер с CTA
   369→/// всегда видим — в скролл-зону не входит). Один источник для реестра
   370→/// (hit-rect'ы), ввода (кнопки) и отрисовки — «ввод = тому, что видно».
   371→pub struct OnboardingLayout {
   372→    /// Rect карточки `[x, y, w, h]` (центр окна, кламп с полями).
   373→    pub card: [f32; 4],
   374→    /// Строки тела после измеренного переноса по ширине карточки.
   375→    pub lines: Vec<String>,
   376→}
   377→
   378→/// Раскладка карточки шага: карточка + строки тела + синхронизация
   379→/// скролл-состояния тела (кит [`ScrollState`] — offset сохраняется,
   380→/// content_h/viewport_h обновляются, позиция клампится; паттерн
   381→/// `flow_map_layout`).
   382→pub fn card_layout(
   383→    viewport: [f32; 2],
   384→    step: usize,
   385→    language: Language,
   386→    role: &str,
   387→    scroll: &mut ScrollState,
   388→    m: &mut TextMeasurer,
   389→    fs: &mut cosmic_text::FontSystem,
   390→) -> OnboardingLayout {
   391→    let margin = ONBOARDING_VIEWPORT_MARGIN;
   392→    let slot = UiRect::new(
   393→        margin,
   394→        margin,
   395→        (viewport[0] - margin * 2.0).max(0.0),
   396→        (viewport[1] - margin * 2.0).max(0.0),
   397→    );
   398→    let w = ONBOARDING_CARD_WIDTH.min(slot.w);
   399→    let lines = body_lines(step, w, language, role, m, fs);
   400→    let body_h = lines.len() as f32 * ONBOARDING_BODY_LINE_H;
   401→    // CR-031: финальный шаг несёт две полноширинные CTA-опции — зона
   402→    // резервируется в желаемой высоте карточки.
   403→    let final_step = step + 1 >= ONBOARDING_STEPS.len();
   404→    let options_h = if final_step {
   405→        ONBOARDING_OPTION_ZONE
   406→    } else {
   407→        0.0
   408→    };
   409→    let desired_h = body_top_offset() + body_h + ONBOARDING_PAD + ONBOARDING_FOOTER_H + options_h;
   410→    // Кламп высоты к слоту: desired пре-клампнут, поэтому min-инвариант
   411→    // модали (приоритетен — parity FR-060) не конфликтует с клампом FR-028.
   412→    let h = desired_h.min(slot.h);
   413→    let size = UiVec2::new(w, h);
   414→    let panel = kit::modal(slot, size, size, size).panel;
   415→    let card = [panel.x, panel.y, panel.w, panel.h];
   416→    // Скролл тела: при клампе высоты viewport_h < content_h — строки
   417→    // прокручиваются (колесо ввода; видимые строки — `kit::list_rows`).
   418→    let body = body_area(card, final_step);
   419→    scroll.content_h = lines.len() as f32 * ONBOARDING_BODY_LINE_H;
   420→    scroll.viewport_h = body[3];
   421→    scroll.clamp();
   422→    OnboardingLayout { card, lines }
   423→}

   425→/// Прогресс-точки: центры по горизонтали (рендер — квады-кружки).
   426→/// CR-031/S2: центрируются по текстовой колонке (правая половина), не по
   427→/// всей карточке.
   428→pub fn progress_dots(card: [f32; 4]) -> (Vec<f32>, f32) {
   429→    let col = text_column(card);
   430→    let n = ONBOARDING_STEPS.len() as f32;
   431→    let total = n * ONBOARDING_DOT + (n - 1.0) * ONBOARDING_DOT_GAP;
   432→    let start = col[0] + (col[2] - total) / 2.0;
   433→    let centers: Vec<f32> = (0..ONBOARDING_STEPS.len())
   434→        .map(|i| start + i as f32 * (ONBOARDING_DOT + ONBOARDING_DOT_GAP) + ONBOARDING_DOT / 2.0)
   435→        .collect();
   436→    let y = col[1] + ONBOARDING_PAD + ONBOARDING_TITLE_LINE_H + ONBOARDING_DOTS_TOP;
   437→    (centers, y)
   438→}

   440→/// Rect кнопки карточки (футер прибит к низу карточки — при клампе высоты
   441→/// кнопки остаются достижимы): Prev — слева, Next — справа (оба в футере);
   442→/// Skip — в правом верхнем углу карточки (выход виден всегда — NN/g).
   443→//
   444→// K5/FR-070: `Skip` — это подписанная прямоугольная кнопка 64×22
   445→// («Пропустить»), НЕ каноническая «×» в углу модали. Поэтому `Skip` НЕ
   446→// является миграционным кандидатом для `kit::stage_close_button` (тот
   447→// возвращает квадрат `ICON_BUTTON_SIZE`=26×26 под глиф «×», а Skip —
   448→// широкая кнопка-призрак с подписью). Этот TODO — НЕ close-button
   449→// migration (вопреки FR-070 тегу), а кнопочный layout-миграция.
   450→//
   451→// O2 (Task O / FR-UI-FOOTER): Prev/Next carousel footer migrated to
   452→// `kit::split_footer_buttons` (Agent L's split-footer kit component).
   453→// Pattern: Prev on LEFT (slot.x + inset), Next on RIGHT (slot.right -
   454→// inset - W), inset = ONBOARDING_PAD = 24, gap = GAP_CONTROLS (8,
   455→// unused — 1 button per side). Skip stays manual (top-right corner —
   456→// NOT in the footer slot). Positions are 1:1 with the previous
   457→// hand-rolled formula (footer_y = card.bottom - FOOTER_H + (FOOTER_H -
   458→// BTN_H)/2; Prev.x = card.x + PAD; Next.x = card.right - PAD - W) —
   459→// I-1: zero visual jump.
   460→//
   461→// Будущая миграция Skip — на `kit::button_layout` (измеренный размер
   462→// подписи + `BUTTON_PAD_H` × 2, высота `BUTTON_HEIGHT`=30). Требует:
   463→// 1. Сигнатуру `button_rect(card, button, m, fs, family, size)` —
   464→//    вместо текущей без замерщика (10+ callers в onboarding_ui.rs,
   465→//    app/overlays.rs:3669, app.rs:13310 — большая площадь рефактора).
   466→// 2. UX-ревью: размер меняется с 64×22 на ~(80–100)×30 (ширина
   467→//    измеренной подписи «Пропустить» + 2·BUTTON_PAD_H=12; высота — kit
   468→//    canonical `BUTTON_HEIGHT`=30). Это сдвиг позиции кнопки в карточке
   469→//    (примерно +6px по высоте, +16–36px по ширине).
   470→// 3. Стиль: `kit::button_style(ButtonVariant::Ghost, KitState::Normal,
   471→//    palette)` — ghost-кнопка без заливки/рамки, hover → accent border.
   472→//
   473→// Оставлено как TODO до отдельной волны onboarding button_layout
   474→// (I-1: ноль скачка). Паритет с другими подписанными кнопками-призраками
   475→// в карточках (например suggest-cards).
   476→pub fn button_rect(card: [f32; 4], button: OnboardingButton) -> [f32; 4] {
   477→    match button {
   478→        OnboardingButton::Prev | OnboardingButton::Next => {
   479→            // O2: split footer (Prev left / Next right) via kit.
   480→            // CR-031/S2: футер — в текстовой колонке (правая половина),
   481→            // не по всей карточке.
   482→            let col = text_column(card);
   483→            let slot = UiRect::new(
   484→                col[0],
   485→                col[1] + col[3] - ONBOARDING_FOOTER_H,
   486→                col[2],
   487→                ONBOARDING_FOOTER_H,
   488→            );
   489→            let btns = kit::split_footer_buttons(
   490→                slot,
   491→                &[ONBOARDING_BUTTON_W],
   492→                &[ONBOARDING_BUTTON_W],
   493→                kit::GAP_CONTROLS,
   494→                ONBOARDING_PAD,
   495→            );
   496→            // btns[0] = (Prev rect, FooterGroup::Left, 0);
   497→            // btns[1] = (Next rect, FooterGroup::Right, 0).
   498→            let idx = match button {
   499→                OnboardingButton::Prev => 0,
   500→                OnboardingButton::Next => 1,
   501→                _ => unreachable!(),
   502→            };
   503→            let r = btns[idx].0;
   504→            [r.x, r.y, r.w, r.h]
   505→        }
   506→        OnboardingButton::Skip => [
   507→            card[0] + card[2] - ONBOARDING_PAD - 64.0,
   508→            card[1] + 10.0,
   509→            64.0,
   510→            22.0,
   511→        ],
   512→        // CR-031: финальные CTA-опции — полноширинные кнопки над футером
   513→        // ([`option_rects`]; в футере их нет)
   514→        OnboardingButton::FinalGallery => option_rects(card)[0],
   515→        OnboardingButton::FinalEmpty => option_rects(card)[1],
   516→    }
   517→}
   518→
   519→/// Hit-test кнопки карточки. «Назад» на первом шаге отсутствует (None),
   520→/// «Далее»/«Готово» и «Пропустить» доступны всегда (инвариант карусели).
   521→/// CR-031: на финальном шаге «Далее» отсутствует (футер — только «Назад»),
   522→/// вместо него — полноширинные CTA-опции «Открыть шаблонную схему» /
   523→/// «Начать самому» (нижняя зона карточки). FR-097: на coarse-указателе
   524→/// тач-цели кнопок дотягиваются до 44 лог. px (hit-only, кламп в карточку —
   525→/// расширенная зона не выходит за оверлей; на точном указателе — прежние
   526→/// зоны).
   527→pub fn button_at(
   528→    card: [f32; 4],
   529→    state: &OnboardingState,
   530→    point: [f32; 2],
   531→) -> Option<OnboardingButton> {
   532→    // CR-031/S2: тач-расширение клампится в текстовую колонку (кнопки
   533→    // живут в колонке, а не во всей карточке).
   534→    let col = text_column(card);
   535→    let coarse = crate::touch_targets::pointer_coarse();
   536→    let hit = |rect: [f32; 4]| {
   537→        let rect = if coarse {
   538→            crate::touch_targets::intersect_xywh(
   539→                crate::touch_targets::expand_xywh(rect, crate::touch_targets::MIN_TOUCH_TARGET),
   540→                col,
   541→            )
   542→        } else {
   543→            rect
   544→        };
   545→        point[0] >= rect[0]
   546→            && point[0] <= rect[0] + rect[2]
   547→            && point[1] >= rect[1]
   548→            && point[1] <= rect[1] + rect[3]
   549→    };
   550→    // CR-031: финальный шаг — CTA-опции вместо «Далее»
   551→    if state.is_last() {
   552→        let [gallery, empty] = option_rects(card);
   553→        if hit(gallery) {
   554→            return Some(OnboardingButton::FinalGallery);
   555→        }
   556→        if hit(empty) {
   557→            return Some(OnboardingButton::FinalEmpty);
   558→        }
   559→    } else if hit(button_rect(card, OnboardingButton::Next)) {
   560→        return Some(OnboardingButton::Next);
   561→    }
   562→    if hit(button_rect(card, OnboardingButton::Skip)) {
   563→        return Some(OnboardingButton::Skip);
   564→    }
   565→    // «Назад» только не на первом шаге
   566→    if state.step > 0 && hit(button_rect(card, OnboardingButton::Prev)) {
   567→        return Some(OnboardingButton::Prev);
   568→    }
   569→    None
   570→}
   571→
   572→/// Клик внутри карточки (но не по кнопкам)? Ввод канваса глотается всей
   573→/// карточкой — клик мимо кнопок не проваливается под оверлей.
   574→pub fn point_in_card(card: [f32; 4], point: [f32; 2]) -> bool {
   575→    point[0] >= card[0]
   576→        && point[0] <= card[0] + card[2]
   577→        && point[1] >= card[1]
   578→        && point[1] <= card[1] + card[3]
   579→}

   737→    pub card: [f32; 4],

   739→    /// [`AiOnboardingChoice`]: Local, Cloud, SelfHosted.
   740→    pub mode_cards: [[f32; 4]; 3],

   742→    pub btn_privacy: [f32; 4],
   743→    pub btn_continue: [f32; 4],

   748→/// Вычислить раскладку экрана AI-онбординга (чистая функция, без замерщика —
   749→/// высоты строк фиксированы по прототипу F-8). Карточка центрируется во
   750→/// вьюпорте с полями `AI_ONB_VIEWPORT_MARGIN`; высота зависит от
   751→/// `state.privacy_open`.
   752→//
   753→// O1 (Task O / FR-UI-RADIO): mode-card rendering migrated to `kit::radio_card`
   754→// in `app/overlays.rs::ai_onboarding_overlay` (where palette is available —
   755→// this layout function stays a pure geometry function returning slot rects,
   756→// per task step 1 option (b); signature unchanged). The kit's
   757→// `radio_card(slot, label_w, desc_w, selected, state, palette)` is called
   758→// per-card in overlays.rs; `paint_radio_card` emits the card background +
   759→// indicator; `layout.label`/`layout.desc` rects are used for text positions.
   760→// Tag (right corner) stays separate — kit doesn't support tag.
   761→pub fn ai_onboarding_layout(viewport: [f32; 2], state: &AiOnboardingState) -> AiOnboardingLayout {
   762→    use canvas_ui::geometry::{UiRect, UiVec2};
   763→    use canvas_ui::kit;
   764→    let margin = AI_ONB_VIEWPORT_MARGIN;
   765→    let slot = UiRect::new(
   766→        margin,
   767→        margin,
   768→        (viewport[0] - margin * 2.0).max(0.0),
   769→        (viewport[1] - margin * 2.0).max(0.0),
   770→    );
   771→    let w = AI_ONB_CARD_W.min(slot.w);
   772→    // Высота: step + title + sub + 3 mode cards (с зазорами) + actions +
   773→    // опционально privacy block. Поля — top/bottom.
   774→    let modes_h = 3.0 * AI_ONB_MODE_H + 2.0 * AI_ONB_MODE_GAP;
   775→    let mut h = AI_ONB_PAD_TOP
   776→        + AI_ONB_STEP_H
   777→        + AI_ONB_TITLE_H
   778→        + AI_ONB_SUB_H
   779→        + modes_h
   780→        + 18.0 // зазор от режимов до actions (прототип: margin-top: 18px)
   781→        + AI_ONB_ACTIONS_H
   782→        + AI_ONB_PAD_BOTTOM;
   783→    if state.privacy_open {
   784→        h += 15.0 + AI_ONB_PRIV_H; // зазор + блок
   785→    }
   786→    let h = h.min(slot.h);
   787→    let panel = kit::modal(
   788→        slot,
   789→        UiVec2::new(w, h),
   790→        UiVec2::new(w, h),
   791→        UiVec2::new(w, h),
   792→    )
   793→    .panel;
   794→    let card = [panel.x, panel.y, panel.w, panel.h];
   795→    // Y-positions: шаг → заголовок → подзаголовок → 3 карточки → actions → privacy.
   796→    let x = card[0] + AI_ONB_PAD_X;
   797→    let inner_w = (card[2] - AI_ONB_PAD_X * 2.0).max(0.0);
   798→    let mut y = card[1] + AI_ONB_PAD_TOP;
   799→    y += AI_ONB_STEP_H + AI_ONB_TITLE_H + AI_ONB_SUB_H;
   800→    // 3 карточки-режима (Local / Cloud / SelfHosted — порядок AiOnboardingChoice).
   801→    let mut mode_cards = [[0.0f32; 4]; 3];
   802→    for (i, rect) in mode_cards.iter_mut().enumerate() {
   803→        *rect = [
   804→            x,
   805→            y + i as f32 * (AI_ONB_MODE_H + AI_ONB_MODE_GAP),
   806→            inner_w,
   807→            AI_ONB_MODE_H,
   808→        ];
   809→    }
   810→    let actions_y = y + 3.0 * AI_ONB_MODE_H + 2.0 * AI_ONB_MODE_GAP + 18.0;
   811→    // O2 (Task O / FR-UI-FOOTER): Privacy (left) + Continue (right) migrated
   812→    // to `kit::split_footer_buttons` (Agent L's split-footer kit component).
   813→    // Pattern: Privacy on LEFT (slot.x + inset=0), Continue on RIGHT
   814→    // (slot.right - inset=0 - W). Slot height = AI_ONB_BTN_H (30, no vertical
   815→    // centering — y stays at actions_y). Positions are 1:1 with the previous
   816→    // hand-rolled formula (btn_privacy.x = x = card.x + PAD_X;
   817→    // btn_continue.x = x + inner_w - CONTINUE_W = card.right - PAD_X -
   818→    // CONTINUE_W) — I-1: zero visual jump.
   819→    let actions_slot = UiRect::new(x, actions_y, inner_w, AI_ONB_BTN_H);
   820→    let ai_footer_btns = kit::split_footer_buttons(
   821→        actions_slot,
   822→        &[AI_ONB_BTN_PRIV_W],
   823→        &[AI_ONB_BTN_CONTINUE_W],
   824→        kit::GAP_CONTROLS, // unused — 1 button per side
   825→        0.0,               // no inset — both buttons flush to slot edges
   826→    );
   827→    // btns[0] = (Privacy rect, FooterGroup::Left, 0);
   828→    // btns[1] = (Continue rect, FooterGroup::Right, 0).
   829→    let priv_r = ai_footer_btns[0].0;
   830→    let cont_r = ai_footer_btns[1].0;
   831→    let btn_privacy = [priv_r.x, priv_r.y, priv_r.w, priv_r.h];
   832→    let btn_continue = [cont_r.x, cont_r.y, cont_r.w, cont_r.h];
   833→    let privacy_block = if state.privacy_open {
   834→        [
   835→            x,
   836→            actions_y + AI_ONB_ACTIONS_H + 15.0,
   837→            inner_w,
   838→            AI_ONB_PRIV_H,
   839→        ]
   840→    } else {
   841→        [0.0; 4]
   842→    };
   843→    AiOnboardingLayout {
   844→        card,
   845→        mode_cards,
   846→        btn_privacy,
   847→        btn_continue,
   848→        privacy_block,
   849→    }
   850→}

   852→/// Hit-test кнопки экрана AI-онбординга (карточки-режимы — всегда доступны;
   853→/// «Подробнее» — всегда; «Продолжить» — только если `selected` не None).
   854→pub fn ai_onboarding_button_at(
   855→    layout: &AiOnboardingLayout,
   856→    state: &AiOnboardingState,
   857→    point: [f32; 2],
   858→) -> Option<AiOnboardingButton> {
   859→    use canvas_ui::geometry::UiRect;
   860→    let hit = |rect: [f32; 4]| -> bool {
   861→        let r = UiRect::new(rect[0], rect[1], rect[2], rect[3]);
   862→        point[0] >= r.x && point[0] <= r.x + r.w && point[1] >= r.y && point[1] <= r.y + r.h
   863→    };
   864→    // Карточки-режимы (Local / Cloud / SelfHosted — порядок AiOnboardingChoice).
   865→    let choices = [
   866→        AiOnboardingChoice::Local,
   867→        AiOnboardingChoice::Cloud,
   868→        AiOnboardingChoice::SelfHosted,
   869→    ];
   870→    for (i, &choice) in choices.iter().enumerate() {
   871→        if hit(layout.mode_cards[i]) {
   872→            return Some(AiOnboardingButton::Mode(choice));
   873→        }
   874→    }
   875→    if hit(layout.btn_privacy) {
   876→        return Some(AiOnboardingButton::Privacy);
   877→    }
   878→    // «Продолжить» — только если режим выбран (иначе кнопка disabled).
   879→    if state.selected.is_some() && hit(layout.btn_continue) {
   880→        return Some(AiOnboardingButton::Continue);
   881→    }
   882→    None
   883→}

   887→pub fn ai_onboarding_point_in_card(card: [f32; 4], point: [f32; 2]) -> bool {
   888→    point[0] >= card[0]
   889→        && point[0] <= card[0] + card[2]
   890→        && point[1] >= card[1]
   891→        && point[1] <= card[1] + card[3]
   892→}
