    85→/// Состояние галереи схем (модальная; `None`-подобие — `open == false`).
    86→#[derive(Debug, Clone, Default, PartialEq)]
    87→pub struct SchemeGalleryState {
    88→    pub open: bool,
    89→    /// Ординал выбранной строки среди отфильтрованных.
    90→    pub selected: usize,
    91→    /// Активная категория (`None` — «Все»).
    92→    pub category: Option<String>,
    93→    /// Фильтр по названию/описанию (без регистрозависимости).
    94→    pub filter: String,
    95→    /// Верх строки окна видимости (индекс в отфильтрованном списке).
    96→    pub scroll_top: usize,
    97→}

    99→impl SchemeGalleryState {
   100→    pub fn open(&mut self) {
   101→        self.open = true;
   102→        self.selected = 0;
   103→        self.scroll_top = 0;
   104→    }
   105→
   106→    pub fn close(&mut self) {
   107→        self.open = false;
   108→        self.filter.clear();
   109→        self.category = None;
   110→        self.selected = 0;
   111→        self.scroll_top = 0;
   112→    }
   113→}

   115→/// Отфильтрованные строки: категория-чип + подстрока фильтра
   116→/// (название RU/EN, описание RU/EN, категория).
   117→///
   118→/// FR-087 (роль/класс): `visible` — allow-list raw-токенов категорий
   119→/// схем (`App::visible_scheme_categories`: ручной фильтр настроек >
   120→/// дефолт роли > все; `onboarding` виден всегда — добавляется
   121→/// вызывающим). Поиск ведётся только по видимым категориям.
   122→pub fn rows<'a>(
   123→    registry: &'a SchemeRegistry,
   124→    state: &SchemeGalleryState,
   125→    visible: &[String],
   126→) -> Vec<&'a SchemeManifest> {
   127→    let filter = state.filter.to_lowercase();
   128→    registry
   129→        .list()
   130→        .iter()
   131→        .filter(|s| visible.iter().any(|v| v == &s.category))
   132→        .filter(|s| match &state.category {
   133→            Some(cat) => s.category == *cat,
   134→            None => true,
   135→        })
   136→        .filter(|s| {
   137→            filter.is_empty()
   138→                || s.name_ru.to_lowercase().contains(&filter)
   139→                || s.name_en.to_lowercase().contains(&filter)
   140→                || s.description_ru.to_lowercase().contains(&filter)
   141→                || s.description_en.to_lowercase().contains(&filter)
   142→                || s.category.to_lowercase().contains(&filter)
   143→        })
   144→        .collect()
   145→}
   146→
   147→/// Уникальные категории реестра в стабильном порядке (чипы «Все» + N).
   148→/// FR-087: только видимые категории (роль/ручной фильтр) — чип скрытой
   149→/// категории не рисуется и не кликается.
   150→pub fn categories(registry: &SchemeRegistry, visible: &[String]) -> Vec<(String, String, String)> {
   151→    let mut out: Vec<(String, String, String)> = Vec::new();
   152→    for scheme in registry.list() {
   153→        if !visible.iter().any(|v| v == &scheme.category) {
   154→            continue;
   155→        }
   156→        if !out.iter().any(|(key, _, _)| *key == scheme.category) {
   157→            out.push((
   158→                scheme.category.clone(),
   159→                scheme.category_ru.clone(),
   160→                scheme.category_en.clone(),
   161→            ));
   162→        }
   163→    }
   164→    out
   165→}

   167→/// Сдвинуть окно видимости так, чтобы выбранная строка была видна.
   168→pub fn clamp_scroll(state: &mut SchemeGalleryState, visible: usize) {
   169→    if visible == 0 {
   170→        state.scroll_top = 0;
   171→        return;
   172→    }
   173→    if state.selected < state.scroll_top {
   174→        state.scroll_top = state.selected;
   175→    } else if state.selected >= state.scroll_top + visible {
   176→        state.scroll_top = state.selected + 1 - visible;
   177→    }
   178→    // Кламп скролла к размеру списка — вызывающий передаёт rows.len().
   179→}

   181→/// W-a (аудит §8 п.4): шаг колесa каталога — строк на щелчок колеса
   182→/// (паритет с шаблонным доком `template_ui::PANEL_WHEEL_LINES`).
   183→pub const WHEEL_ROWS_PER_LINE: f32 = 3.0;
   184→
   185→/// W-a (аудит §8 п.4): верх окна видимости после колес-скролла —
   186→/// `scroll_top + delta_rows` (строки), кламп на `0..=max_top`, где
   187→/// `max_top = список − окно видимости`: хвост каталога достижим,
   188→/// пустого окна нет (при пустом списке/окне — 0). Выбор не двигается
   189→/// (прецедент веток wheel-скролла input.rs); связка с [`clamp_scroll`]
   190→/// сохраняется — следующая клавиатурная навигация возвращает окно
   191→/// к выбранной строке.
   192→pub fn wheel_scroll_top(current: usize, delta_rows: f32, list_len: usize, visible: usize) -> usize {
   193→    let visible = visible.max(1);
   194→    let max_top = list_len.saturating_sub(visible);
   195→    // Округление суммы (не дельты): мелкие тачпад-шаги накапливаются,
   196→    // пока не наберут строку.
   197→    let next = (current as f32 + delta_rows).round();
   198→    if next <= 0.0 {
   199→        0
   200→    } else if next >= max_top as f32 {
   201→        max_top
   202→    } else {
   203→        next as usize
   204→    }
   205→}

   207→/// Раскладка галереи: панель по центру, шапка, фильтр, чипы, строки.

   218→    /// Индексы строк в общем отфильтрованном списке.

   223→/// Раскладка галереи (чистая функция; кламп к вьюпорту — инвариант
   224→/// 320×240, строки скроллятся окном видимости). FR-053: собрана
   225→/// примитивами `canvas-ui::layout` — вертикальный ритм дословно прежний
   226→/// (зазоры header→input 0, input→chips `SPACING_S`, chips→rows
   227→/// `SPACING_S`, строки примыкают к футеру), позиция панели/размер —
   228→/// Stack/Constrain. W3.2 (каталог `docs/plans/fr-068-w3-consumer-migration.md`):
   229→/// дети скелета/чипов/строк выражаются [`MeasuredItem`] —
   230→/// `Row/Column::lay_out_measured` без ручных фиксированных детей.
   231→///
   232→/// M1 (FR-UI-CHIP-STRIP): `ru` — выбор языка подписи чипа для замера
   233→/// ширины (кит `chip_strip` измеряет текст — нужна конкретная подпись).
   234→/// Сам язык строки (RU/EN) сохранён в `chip_rects` как `Option<String>`
   235→/// (ключ категории; подпись рендера выбирается по языку пользователя).
   236→pub fn layout(
   237→    viewport: [f32; 2],
   238→    list: &[&SchemeManifest],
   239→    state: &SchemeGalleryState,
   240→    ru: bool,
   241→    visible_cats: &[String],
   242→) -> GalleryLayout {
   243→    // W3.2: замерщик — канонические shared-точки на вызов (Text-детей
   244→    // нет — замерщик геометрию не читает).
   245→    let mut m = TextMeasurer::new();
   246→    let mut fs = canvas_render::text::measure_font_system();
   247→    layout_with(viewport, list, state, ru, &mut m, &mut fs, visible_cats)
   248→}
   249→
   250→/// То же с ЯВНЫМ замерщиком (для потребителей, уже держащих
   251→/// `measure_font_system` — двойной лок глобального FontSystem невозможен).
   252→/// FR-087: `visible` — allow-list категорий схем (роль/ручной фильтр) —
   253→/// чипы категорий строятся только из видимых.
   254→///
   255→/// M1 (FR-UI-CHIP-STRIP): `ru` — выбор языка подписи чипа для замера
   256→/// ширины через `kit::chip_strip`. См. [`layout`].
   257→pub fn layout_with(
   258→    viewport: [f32; 2],
   259→    list: &[&SchemeManifest],
   260→    state: &SchemeGalleryState,
   261→    ru: bool,
   262→    m: &mut TextMeasurer,
   263→    fs: &mut cosmic_text::FontSystem,
   264→    visible_cats: &[String],
   265→) -> GalleryLayout {
   266→    let max_w = (viewport[0] - canvas_core::tokens::SPACING_XL).max(280.0);
   267→    let panel_w = PANEL_W.min(max_w);
   268→    // Сколько строк влезает: высота панели — от вьюпорта. Маржа XL
   269→    // учитывается С ОБЕИХ сторон (панель центрируется по вертикали:
   270→    // иначе при ROW_H 62 панель 536 в 560-окне давала y=12 < маржи —
   271→    // поймано G4-линтом после wasm-аудита 2026-09-25).
   272→    let chrome = HEADER_H + INPUT_H + CHIP_H + FOOTER_H + PANEL_PAD * 3.0;
   273→    let max_h = (viewport[1] - canvas_core::tokens::SPACING_XL * 2.0).max(160.0);
   274→    // Панель растёт под список, но не выше вьюпорта (строки скроллятся).
   275→    let panel_h = max_h.min(chrome + ROW_H * list.len().max(1) as f32);
   276→    let avail_rows_h = (panel_h - chrome).max(0.0);
   277→    let visible = ((avail_rows_h / ROW_H).floor() as usize).max(1);
   278→    let shown = visible.min(list.len().saturating_sub(state.scroll_top));
   279→
   280→    // Панель — Stack по центру вьюпорта (позиция = (vw−w)/2, (vh−h)/2).
   281→    let panel = stack(
   282→        UiRect::new(0.0, 0.0, viewport[0], viewport[1]),
   283→        UiVec2::new(panel_w, panel_h),
   284→        HAlign::Center,
   285→        VAlign::Center,
   286→    );
   287→    let inner = pad(panel, EdgeInsets::uniform(PANEL_PAD));
   288→    let inner_w = panel_w - PANEL_PAD * 2.0;
   289→
   290→    // Вертикальный ритм панели: Column без базового зазора + явные
   291→    // распорки `SPACING_S` там, где прежняя геометрия имела зазор
   292→    // (header→input 0, input→chips 6, chips→rows 6, rows→footer 0 —
   293→    // ноль визуального скачка). W3.2: дети — MeasuredItem (Fixed/Spacer);
   294→    // Семантика Spacer в колонке — нулевая высота (main-ось — высота;
   295→    // см. оракул measured_column_matches_manual_fixed_oracle) — бит-в-бит
   296→    // с прежней проводкой.
   297→    let items = vec![
   298→        // Шапка: место под заголовок (кнопка «×» — в правом крае панели).
   299→        MeasuredItem::Fixed {
   300→            w: inner_w - 32.0,
   301→            h: HEADER_H,
   302→        },
   303→        MeasuredItem::Fixed {
   304→            w: inner_w,
   305→            h: INPUT_H,
   306→        },
   307→        MeasuredItem::Spacer(canvas_core::tokens::SPACING_S),
   308→        MeasuredItem::Fixed {
   309→            w: inner_w,
   310→            h: CHIP_H,
   311→        },
   312→        MeasuredItem::Spacer(canvas_core::tokens::SPACING_S),
   313→        MeasuredItem::Fixed {
   314→            w: inner_w,
   315→            h: avail_rows_h,
   316→        },
   317→        MeasuredItem::Fixed {
   318→            w: inner_w,
   319→            h: FOOTER_H,
   320→        },
   321→    ];
   322→    let col = Column {
   323→        gap: 0.0,
   324→        cross: CrossAlign::Start,
   325→        ..Column::default()
   326→    }
   327→    .lay_out_measured(inner, &items, m, fs, FAMILY, 12.0);
   328→    let as_rect = |r: &UiRect| [r.x, r.y, r.w, r.h];
   329→    let header_rect = as_rect(&col[0]);
   330→    let input_rect = as_rect(&col[1]);
   331→    let chips_slot = col[3];
   332→    let rows_slot = col[5];
   333→    let footer_rect = as_rect(&col[6]);
   334→
   335→    // FR-070: каноническая «×» в правом-верхнем углу панели —
   336→    // kit::stage_close_button (audit §6.1 — бывшая раскладка через
   337→    // `stack(inner, 24×24, End, Start)` с офсетом +4; size 24 →
   338→    // ICON_BUTTON_SIZE 26, inset 6 (через inner+4) → SPACING_SM 8 от panel —
   339→    // single source of truth, смещение ≈ 2px вправо/вниз — допустимо по
   340→    // AGENTS.md §«UI-кит»; кнопка переезжает из шапки в угол панели
   341→    // (канонический паттерн «× в углу модали»).
   342→    let close = canvas_ui::kit::stage_close_button(panel);
   343→
   344→    // Чипы: «Все» + категории — M1 (FR-UI-CHIP-STRIP): мигрировано на
   345→    // `kit::chip_strip`. Кит измеряет подпись (CHIP_FAMILY = SANS_FAMILY,
   346→    // CHIP_FONT_SIZE = 12 — паритет метрик рендера), ширина чипа =
   347→    // text_w + 2·CHIP_PAD_H (16), высота = CHIP_HEIGHT = 24, зазор =
   348→    // CHIP_GAP = SPACING_S — single source of truth (AGENTS.md §4).
   349→    // squeeze=false → RowPolicy::Fit (паритет прежней ручной Row(Fit):
   350→    // переполнение НЕ маскируется — ловится G4-линтом; прежний молчаливый
   351→    // `break`-кламп удалён). Канонизация: высота чипа 28→24 (4px короче
   352→    // — panel chrome ниже, shifts everything below by 4px; acceptable per
   353→    // audit §6.1, I-1 FR-046 — canonicalization to kit tokens desired).
   354→    //
   355→    // Язык подписи (`ru`) выбирает «Все»/«All» + category_ru/category_en
   356→    // для ЗАМЕРА ширины; сам ключ категории сохранён отдельно
   357→    // (`chip_keys: Option<String>`, None для «Все») — рендер в overlays.rs
   358→    // выбирает подпись по языку пользователя.
   359→    let all_label = if ru { "Все" } else { "All" };
   360→    let cats = categories(SchemeRegistry::embedded(), visible_cats);
   361→    let mut chip_labels: Vec<String> = vec![all_label.to_owned()];
   362→    let mut chip_keys: Vec<Option<String>> = vec![None];
   363→    for (key, cat_ru, cat_en) in &cats {
   364→        chip_labels.push(if ru { cat_ru.clone() } else { cat_en.clone() });
   365→        chip_keys.push(Some(key.clone()));
   366→    }
   367→    let labels_ref: Vec<&str> = chip_labels.iter().map(|s| s.as_str()).collect();
   368→    let chips = kit::chip_strip(chips_slot, &labels_ref, false, m, fs);
   369→    // Порядок items сохранён kit'ом (документировано в chip_strip): result[i]
   370→    // ↔ items[i]. Zip с chip_keys для проброса ключа категории в рендер.
   371→    let chip_rects: Vec<([f32; 4], Option<String>)> = chips
   372→        .iter()
   373→        .zip(chip_keys)
   374→        .map(|((r, _), key)| ([r.x, r.y, r.w, r.h], key))
   375→        .collect();
   376→
   377→    // Строки окна видимости: Column с зазором `SPACING_S`, видимая часть
   378→    // строки `ROW_INNER_H` (полный шаг ROW_H — дословно прежний ритм).
   379→    let row_children: Vec<MeasuredItem> = (0..shown)
   380→        .map(|_| MeasuredItem::Fixed {
   381→            w: inner_w,
   382→            h: ROW_INNER_H,
   383→        })
   384→        .collect();
   385→    let row_layout = Column {
   386→        gap: canvas_core::tokens::SPACING_S,
   387→        cross: CrossAlign::Start,
   388→        ..Column::default()
   389→    }
   390→    .lay_out_measured(rows_slot, &row_children, m, fs, FAMILY, 12.0);
   391→    let mut row_rects = Vec::with_capacity(shown);
   392→    let mut visible_rows = Vec::with_capacity(shown);
   393→    // Инвариант: `shown <= list.len() - scroll_top` (кламп выше), поэтому
   394→    // scroll_top + i < list.len() для всех i < shown — клампов в цикле нет.
   395→    for (i, r) in row_layout.iter().enumerate() {
   396→        let index = state.scroll_top + i;
   397→        row_rects.push(as_rect(r));
   398→        visible_rows.push(index);
   399→    }
   400→

   402→        panel_rect: [panel.x, panel.y, panel.w, panel.h],
   403→        header_rect,
   404→        close_rect: [close.x, close.y, close.w, close.h],
   405→        input_rect,
   406→        chip_rects,
   407→        row_rects,

   409→        footer_rect,

   425→/// Контракт W3.2: вызывающий держит guard `measure_font_system` (передаёт
   426→/// `fs` явно) — внутренняя раскладка обязана брать `layout_with`, а НЕ
   427→/// лочащую `layout`. Прецедент бага 2026-09-26: вызов `layout` здесь давал
   428→/// второй лок того же глобального FontSystem — wasm падал паникой

   434→/// (чипы не влияют на геометрию строк — колонка держит один слот CHIP_H,
   435→/// но передаем тот же список для консистентности с рендером).
   436→pub fn row_labels(
   437→    viewport: [f32; 2],
   438→    list: &[&SchemeManifest],
   439→    state: &SchemeGalleryState,
   440→    ru: bool,
   441→    measurer: &mut TextMeasurer,
   442→    fs: &mut cosmic_text::FontSystem,
   443→    visible_cats: &[String],
   444→) -> Vec<RowLabel> {
   445→    let lay = layout_with(viewport, list, state, ru, measurer, fs, visible_cats);
   446→    let max_w = (lay.row_rects.first().map(|r| r[2]).unwrap_or(0.0) - ROW_TEXT_PAD * 2.0).max(0.0);
   447→    lay.visible_rows
   448→        .iter()
   449→        .map(|&index| {
   450→            let Some(scheme) = list.get(index) else {
   451→                return RowLabel {
   452→                    title: String::new(),
   453→                    desc: String::new(),
   454→                };
   455→            };
   456→            RowLabel {
   457→                title: measurer.ellipsis(fs, scheme.display_name(ru), FAMILY, ROW_FONT, max_w),
   458→                desc: measurer.ellipsis(
   459→                    fs,
   460→                    if ru {
   461→                        &scheme.description_ru
   462→                    } else {
   463→                        &scheme.description_en
   464→                    },
   465→                    FAMILY,
   466→                    ROW_DESC_FONT,
   467→                    max_w,
   468→                ),
   469→            }
   470→        })
   471→        .collect()
   472→}

   529→    m: &mut TextMeasurer,
   530→    fs: &mut cosmic_text::FontSystem,
   531→) -> ([f32; 4], [f32; 4]) {

   565→/// Точка в прямоугольнике `[x, y, w, h]` (общий хелпер модуля).
   566→pub fn point_in_rect(rect: [f32; 4], point: [f32; 2]) -> bool {
   567→    point[0] >= rect[0]
   568→        && point[0] <= rect[0] + rect[2]
   569→        && point[1] >= rect[1]
   570→        && point[1] <= rect[1] + rect[3]
   571→}

   797→    /// конца колесом (вниз — до последней строки, вверх — до первой),
   798→    /// связка с clamp_scroll сохраняется: клавиатурная навигация
   799→    /// возвращает окно к выбранной строке.
   800→    #[test]
   801→    fn wheel_reaches_catalog_end_on_800x560() {
   802→        let registry = SchemeRegistry::embedded();
   803→        let mut st = state();
   804→        let list = rows(registry, &st, &all_visible(registry));
   805→        assert!(list.len() >= 10, "каталог 10+ схем (аудит 2026-09)");
   806→        let lay = layout([800.0, 560.0], &list, &st, true, &all_visible(registry));
   807→        let visible = lay.visible_rows.len();
   808→        assert!(visible < list.len(), "окно меньше списка — скролл нужен");
   809→        // Колесо вниз до упора: щелчок = −y · WHEEL_ROWS_PER_LINE.
   810→        let mut guard = 0;
   811→        loop {
   812→            let next = wheel_scroll_top(st.scroll_top, WHEEL_ROWS_PER_LINE, list.len(), visible);
   813→            if next == st.scroll_top {
   814→                break;
   815→            }
   816→            st.scroll_top = next;
   817→            guard += 1;
   818→            assert!(guard <= list.len(), "скролл обязан сойтись");
   819→        }
   820→        let lay_end = layout([800.0, 560.0], &list, &st, true, &all_visible(registry));
   821→        assert_eq!(
   822→            st.scroll_top,
   823→            list.len() - visible,
   824→            "кламп на хвост каталога"
   825→        );
   826→        assert!(
   827→            lay_end.visible_rows.contains(&(list.len() - 1)),
   828→            "последняя схема видна"
   829→        );
   830→        // Колесо вверх до упора — обратно к началу.
   831→        let mut guard = 0;
   832→        loop {
   833→            let next = wheel_scroll_top(st.scroll_top, -WHEEL_ROWS_PER_LINE, list.len(), visible);
   834→            if next == st.scroll_top {
   835→                break;
   836→            }
   837→            st.scroll_top = next;
   838→            guard += 1;
   839→            assert!(guard <= list.len(), "скролл обязан сойтись");
   840→        }
   841→        assert_eq!(st.scroll_top, 0, "окно вернулось к началу");
   842→        // Связка с clamp_scroll: колесо не ломает клавиатурную семантику —
   843→        // после wheel-скролла окно следует за выбранной строкой.
   844→        st.scroll_top = list.len() - visible;
   845→        st.selected = 0;
   846→        clamp_scroll(&mut st, visible);
   847→        let lay_sel = layout([800.0, 560.0], &list, &st, true, &all_visible(registry));
   848→        assert!(
   849→            lay_sel.visible_rows.contains(&0),
   850→            "clamp_scroll вернул выбранную строку в окно"
   851→        );
   852→    }
