   153→/// FR-LLM-D / PRD-0010 F-4.9: результат валидации tool_calls (Q3 whitelist).
   154→#[derive(Debug, Clone)]
   155→pub struct ValidationResult {
   156→    /// Все ли вызовы валидны (создание новых / модификация selected).
   157→    /// `true` по умолчанию (нет отклонённых вызовов → валидно); `false`
   158→    /// когда `rejected` непустой (см. [`Self::default`] ниже — ручная
   159→    /// реализация, т.к. `bool::default()` = `false`, что неверно для
   160→    /// семантики «нет отклонений → валидно»).
   161→    pub valid: bool,
   162→    /// Отклонённые вызовы: (tool_name, причина). `valid=false` → непусто.
   163→    pub rejected: Vec<(String, String)>,
   164→}
   165→
   166→impl Default for ValidationResult {
   167→    /// Дефолт: `valid = true`, `rejected = []` (нет вызовов — нет проблем).
   168→    /// Вручную: `#[derive(Default)]` дал бы `valid = false` (неверно).
   169→    fn default() -> Self {
   170→        Self {
   171→            valid: true,
   172→            rejected: Vec::new(),
   173→        }
   174→    }
   175→}

   182→        let mut quads = Vec::new();
   183→        let mut texts = Vec::new();
   184→        if !self.agent_panel.open {
   185→            return (quads, texts);
   186→        }
   187→        let viewport = self.viewport_logical();
   188→        if viewport[0] < AGENT_PANEL_MIN_VIEWPORT_W || viewport[1] <= 0.0 {

   190→        }

   193→        // FR-LLM-D: KitDraw-адаптер (как ai_status_panel.rs/overlays.rs).

   209→            UiRect::new(panel[0], panel[1], panel[2], panel[3]),

   216→            UiRect::new(panel[0], panel[1], 1.0, panel[3]),

   218→            [0.0; 4],

   240→            UiRect::new(
   241→                panel_x + PAD + 18.0,
   242→                head_y + 16.0,
   243→                panel_w - PAD * 2.0 - 18.0,
   244→                12.0,
   245→            ),
   246→            "tool-calling · 42 MCP-инструмента · PRD-0010 F-4",
   247→            kit_palette.text_muted,
   248→            9.0,
   249→        );
   250→        // Кнопка закрытия ✕ (правый край шапки).
   251→        let close_x = panel_x + panel_w - PAD - CLOSE_BTN_SIZE;
   252→        let close_hovered = point_in_rect(
   253→            [close_x, head_y + 2.0, CLOSE_BTN_SIZE, CLOSE_BTN_SIZE],
   254→            self.cursor,
   255→        );
   256→        let close_state = if close_hovered {
   257→            canvas_ui::kit::KitState::Hovered
   258→        } else {
   259→            canvas_ui::kit::KitState::Normal
   260→        };
   261→        let close_style = canvas_ui::kit::icon_button_style(close_state, &kit_palette);
   262→        let close_rect = UiRect::new(close_x, head_y + 2.0, CLOSE_BTN_SIZE, CLOSE_BTN_SIZE);

   268→        let ctx = self.agent_selection_context();
   269→        let ctx_label = match &ctx {
   270→            AgentContext::EntireCanvas => "Context: entire canvas".to_owned(),
   271→            AgentContext::SelectedNodes(ids) => {
   272→                if ids.len() == 1 {
   273→                    if let Some(&i) = ids.first() {
   274→                        let title = self
   275→                            .scene
   276→                            .canvas
   277→                            .nodes
   278→                            .get(i)

   280→                                n.label
   281→                                    .clone()
   282→                                    .or_else(|| n.title().map(|s| s.to_owned()))

   284→                            })
   285→                            .unwrap_or_else(|| "?".into());
   286→                        format!("Selected: 1 — {}", title)
   287→                    } else {
   288→                        "Selected: 1".to_owned()
   289→                    }
   290→                } else {
   291→                    format!("Selected: {} nodes", ids.len())

   294→        };

   296→        let prov_label = crate::settings_ui::ai_provider_label(
   297→            self.settings.language,
   298→            self.settings.llm.provider_agent,
   299→        );

   301→            "ChatGPT: {}/80",
   302→            self.ai_chatgpt_rate_used.saturating_sub(0)

   305→            self.settings.llm.provider_agent == canvas_llm::LlmProviderId::ChatGptOAuth;

   307→        let chip_y = ctx_y + 6.0;

   309→        // Layout: [Context ...] [Provider 100px] [Rate 100px?]

   313→        let ctx_x = panel_x + PAD;

   334→                canvas_ui::kit::chip_style(canvas_ui::kit::KitState::Normal, &kit_palette);

   345→        d.rect(
   346→            log_rect,
   347→            [
   348→                kit_palette.panel_fill[0],
   349→                kit_palette.panel_fill[1],
   350→                kit_palette.panel_fill[2],
   351→                kit_palette.panel_fill[3] * 0.5,
   352→            ],
   353→            [0.0; 4],
   354→            canvas_core::tokens::RADIUS_PANEL,
   355→        );
   356→        // Empty state — подсказка пользователю что делать.
   357→        let mut msg_y = log_y + 8.0;
   358→        // CR-015 fix (Task Q1+Q2): TextMeasurer для реального шейпинга текста
   359→        // bubble (вместо эвристик `text.len() as f32 * 6.0` / `text.len() /
   360→        // 32.0` / `text.len() / 48` — те ломаются на Cyrillic/emoji: byte_count
   361→        // ≠ glyph_count). Один measurer+FontSystem на цикл сообщений (не на
   362→        // сообщение — кэш переиспользуется). Семейство/кегль — те же, что у
   363→        // `d.label_left` рендера bubble (FR-053: метрики раскладки = метрики
   364→        // рендера). Контракт «вложенный лок FontSystem запрещён» (app.rs:209):
   365→        // `agent_panel_overlay` не вызывается внутри другого FontSystem-лока
   366→        // (handler.rs:314 —顶层 render path, без шейпинг-локов рядом).
   367→        let mut measurer = canvas_ui::measure::TextMeasurer::new();
   368→        let mut fs = canvas_render::text::measure_font_system();
   369→        let bubble_family = canvas_render::text::SANS_FAMILY; // «Noto Sans Display»

   373→                UiRect::new(
   374→                    log_rect.x + 8.0,

   381→                 покажет preview ghost-нодами на канвасе и спросит Accept/Reject. \

   383→                kit_palette.text_muted,
   384→                11.0,
   385→            );
   386→        } else {

   389→                match msg {
   390→                    AgentMessage::User(text) => {
   391→                        // User bubble — выравнивание справа, акцентный фон.
   392→                        // CR-015 fix (Task Q1): real glyph shaping via
   393→                        // TextMeasurer (was `text.len() as f32 * 6.0` and
   394→                        // `(text.len() as f32 / 32.0).ceil()` heuristics —
   395→                        // break on Cyrillic/emoji: byte_count ≠ glyph_count,
   396→                        // multi-byte UTF-8 over-estimates width and line
   397→                        // count). Семейство/кегль — те же, что у `d.label_left`
   398→                        // bubble text (FR-053: метрики раскладки = метрики

   404→                        let bubble_x = log_rect.x + log_rect.w - 8.0 - bubble_w;
   405→                        // Wrap по реальной ширине строки внутри bubble

   779→    /// FR-LLM-D / PRD-0010 F-4.9: валидация tool_calls против selection
   780→    /// whitelist (Q3). Разрешено: создавать новые, модифицировать selected.
   781→    /// REJECT: modify non-selected.
   782→    ///
   783→    /// Это статический анализ tool_calls ДО применения. Реальная защита —
   784→    /// `graph_apply` на серверной стороне (MCP) тоже проверяет.
   785→    pub(super) fn agent_validate_tool_calls(
   786→        &self,
   787→        calls: &[canvas_llm::ToolCall],
   788→        selected: &[usize],
   789→    ) -> ValidationResult {
   790→        let selected_ids: std::collections::HashSet<String> = selected
   791→            .iter()
   792→            .filter_map(|&i| self.scene.canvas.nodes.get(i).map(|n| n.id.clone()))
   793→            .collect();
   794→        let mut rejected = Vec::new();
   795→        for call in calls {
   796→            // REJECT-паттерны: node_edit / node_delete / node_move /
   797→            // node_resize / node_set_color / edge_delete с id НЕ из selected.
   798→            let modifying_id = call.name.as_str();
   799→            let target_id: Option<&str> = match modifying_id {
   800→                "node_edit" | "node_delete" | "node_move" | "node_resize" | "node_set_color"
   801→                | "edge_delete" => call
   802→                    .arguments
   803→                    .as_object()
   804→                    .and_then(|o| o.iter().find(|(k, _)| k == "id"))
   805→                    .and_then(|(_, v)| v.as_string()),
   806→                _ => None,
   807→            };
   808→            if let Some(id) = target_id {
   809→                if !selected_ids.contains(id) {
   810→                    rejected.push((
   811→                        call.name.clone(),
   812→                        format!(
   813→                            "попытка модифицировать непросленную ноду/ребро «{}» (Q3 whitelist)",
   814→                            id
   815→                        ),
   816→                    ));
   817→                }
   818→            }
   819→        }
   820→        ValidationResult {
   821→            valid: rejected.is_empty(),
   822→            rejected,
   823→        }
   824→    }

   880→    /// W2 п.2: preview из реальных tool_calls модели. Разбираются вызовы
   881→    /// `n` (graph_apply батч, FR-033): `n_note {ref, x, y, text}` → ноды,
   882→    /// `edge_create {fromRef, toRef}` → рёбра (по ref-адресам батча).
   883→    /// Ноды без x/y (модель не задала позицию) — цепочка от якоря (как
   884→    /// [`Self::agent_build_preview`]). Вызовы без нод — пустой preview
   885→    /// (панель покажет «нечего применять»).
   886→    pub(super) fn agent_preview_from_calls(&self, calls: &[canvas_llm::ToolCall]) -> AgentPreview {
   887→        // Anchor-геометрия — как mock-preview.
   888→        let anchor = self.selected.and_then(|s| match s {
   889→            Selection::Node(i) => self.scene.canvas.nodes.get(i),
   890→            _ => None,
   891→        });
   892→        let (base_x, base_y) = if let Some(a) = anchor {
   893→            (a.x + a.width + 90.0, a.y - 10.0)
   894→        } else {
   895→            let center = self.camera.position();
   896→            (center[0] - 155.0, center[1] - 60.0)
   897→        };
   898→
   899→        let mut nodes: Vec<AgentPreviewNode> = Vec::new();
   900→        let mut refs: Vec<String> = Vec::new(); // ref → индекс в nodes
   901→        let mut edges: Vec<(usize, usize, String)> = Vec::new();
   902→        let mut pending_edges: Vec<(String, String)> = Vec::new();
   903→
   904→        for call in calls {
   905→            if call.name != "graph_apply" {
   906→                continue;
   907→            }
   908→            let ops = call
   909→                .arguments
   910→                .to_serde()
   911→                .get("operations")
   912→                .and_then(|v| v.as_array())
   913→                .cloned()
   914→                .unwrap_or_default();
   915→            for op in &ops {
   916→                let op_name = op.get("op").and_then(|v| v.as_str()).unwrap_or("");
   917→                match op_name {
   918→                    "n_note" | "n_file" => {
   919→                        let title = op
   920→                            .get("text")
   921→                            .and_then(|v| v.as_str())
   922→                            .map(|t| t.lines().next().unwrap_or(t).to_string())
   923→                            .or_else(|| op.get("path").and_then(|v| v.as_str()).map(String::from))
   924→                            .unwrap_or_else(|| "Новая нода".into());
   925→                        let i = nodes.len();
   926→                        let auto_x = base_x + i as f32 * 320.0;
   927→                        nodes.push(AgentPreviewNode {
   928→                            title,
   929→                            x: op
   930→                                .get("x")
   931→                                .and_then(|v| v.as_f64())
   932→                                .map(|v| v as f32)
   933→                                .unwrap_or(auto_x),
   934→                            y: op
   935→                                .get("y")
   936→                                .and_then(|v| v.as_f64())
   937→                                .map(|v| v as f32)
   938→                                .unwrap_or(base_y),
   939→                            width: op
   940→                                .get("width")
   941→                                .and_then(|v| v.as_f64())
   942→                                .map(|v| v as f32)
   943→                                .unwrap_or(240.0),
   944→                            height: op
   945→                                .get("height")
   946→                                .and_then(|v| v.as_f64())
   947→                                .map(|v| v as f32)
   948→                                .unwrap_or(120.0),
   949→                        });
   950→                        if let Some(r) = op.get("ref").and_then(|v| v.as_str()) {
   951→                            refs.push(r.to_string());
   952→                        } else {
   953→                            refs.push(format!("__idx{i}"));
   954→                        }
   955→                    }
   956→                    "edge_create" => {
   957→                        // Адресация fromRef/toRef резолвится ПОСЛЕ прохода
   958→                        // нод (ref может ссылаться на более ранний op).
   959→                        let from = op.get("fromRef").and_then(|v| v.as_str());
   960→                        let to = op.get("toRef").and_then(|v| v.as_str());
   961→                        if let (Some(f), Some(t)) = (from, to) {
   962→                            pending_edges.push((f.to_string(), t.to_string()));
   963→                        }
   964→                    }
   965→                    _ => {}
   966→                }
   967→            }
   968→        }
   969→        // Резолв ребер по ref → индекс; неразрешенные пропускаются.
   970→        // (второй проход: ref-таблица уже заполнена)
   971→        let resolve = |r: &String| -> Option<usize> { refs.iter().position(|x| x == r) };
   972→        for (f, t) in pending_edges.drain(..) {
   973→            if let (Some(fi), Some(ti)) = (resolve(&f), resolve(&t)) {
   974→                edges.push((fi, ti, String::new()));
   975→            }
   976→        }
   977→        AgentPreview { nodes, edges }
   978→    }

  1024→    /// W2 п.2: применить preview через `mcp_dispatch("n", …)` — батч
  1025→    /// операций FR-033 («всё или ничего», один undo-шаг). Возвращает
  1026→    /// (создано нод, создано связей) или человекочитаемую ошибку.
  1027→    pub(super) fn agent_apply_preview_ops(
  1028→        &mut self,
  1029→        preview: &AgentPreview,
  1030→    ) -> Result<(usize, usize), String> {
  1031→        let mut operations: Vec<serde_json::Value> = Vec::new();
  1032→        let refs: Vec<String> = (0..preview.nodes.len()).map(|i| format!("n{i}")).collect();
  1033→        for (i, node) in preview.nodes.iter().enumerate() {
  1034→            operations.push(serde_json::json!({
  1035→                "op": "n_note",
  1036→                "ref": refs[i],
  1037→                "x": node.x,
  1038→                "y": node.y,
  1039→                "width": node.width,
  1040→                "height": node.height,
  1041→                "text": node.title,
  1042→            }));
  1043→        }
  1044→        for (from, to, label) in &preview.edges {
  1045→            let Some(f) = refs.get(*from) else { continue };
  1046→            let Some(t) = refs.get(*to) else { continue };
  1047→            let mut op = serde_json::json!({
  1048→                "op": "edge_create",
  1049→                "fromRef": f,
  1050→                "toRef": t,
  1051→            });
  1052→            if !label.is_empty() {
  1053→                op["label"] = serde_json::Value::String(label.clone());
  1054→            }
  1055→            operations.push(op);
  1056→        }
  1057→        if operations.is_empty() {
  1058→            return Err("нет операций для применения".into());
  1059→        }
  1060→        let result = canvas_scene::mcp_dispatch(
  1061→            &mut self.scene,
  1062→            &self.templates,
  1063→            "n",
  1064→            &serde_json::json!({ "operations": operations }),
  1065→        );
  1066→        match result {
  1067→            Ok(resp) => {
  1068→                let ok = resp.get("ok").and_then(|v| v.as_bool()).unwrap_or(false);
  1069→                if ok {
  1070→                    Ok((preview.nodes.len(), preview.edges.len()))
  1071→                } else {
  1072→                    Err(resp
  1073→                        .get("message")
  1074→                        .and_then(|v| v.as_str())
  1075→                        .unwrap_or("graph_apply: операция отклонена")
  1076→                        .to_string())
  1077→                }
  1078→            }
  1079→            Err(e) => Err(e),
  1080→        }
  1081→    }

  1355→    /// FR-LLM-D / PRD-0010 F-4: hit-test агент-панели — определить элемент
  1356→    /// под курсором (для обработчика ввода). `None` — клик мимо панели.
  1357→    pub(crate) fn agent_panel_hit(&self, point: [f32; 2]) -> Option<AgentPanelHit> {
  1358→        let panel = self.agent_panel_rect()?;
  1359→        if !point_in_rect(panel, point) {
  1360→            return None;
  1361→        }
  1362→        let [panel_x, panel_y, panel_w, panel_h] = panel;
  1363→        // Кнопка закрытия ✕ (правый край шапки).
  1364→        let head_y = panel_y + 8.0;
  1365→        let close_x = panel_x + panel_w - PAD - CLOSE_BTN_SIZE;
  1366→        if point_in_rect(
  1367→            [close_x, head_y + 2.0, CLOSE_BTN_SIZE, CLOSE_BTN_SIZE],
  1368→            point,
  1369→        ) {
  1370→            return Some(AgentPanelHit::Close);
  1371→        }
  1372→        // Input area.
  1373→        let input_y = panel_y + panel_h - QUICK_H - INPUT_H + 4.0;
  1374→        let send_w = 36.0;
  1375→        let input_w = panel_w - PAD * 2.0 - send_w - GAP;
  1376→        let input_rect = [panel_x + PAD, input_y, input_w, INPUT_H - 8.0];
  1377→        if point_in_rect(input_rect, point) {
  1378→            return Some(AgentPanelHit::Input);
  1379→        }
  1380→        // Send button.
  1381→        let send_rect = [
  1382→            panel_x + PAD + input_w + GAP,
  1383→            input_y,
  1384→            send_w,
  1385→            INPUT_H - 8.0,
  1386→        ];
  1387→        if point_in_rect(send_rect, point) {
  1388→            return Some(AgentPanelHit::Send);
  1389→        }
  1390→        // Quick actions (3 preset).
  1391→        let quick_y = panel_y + panel_h - QUICK_H + 4.0;
  1392→        let quick_w = (panel_w - PAD * 2.0 - GAP * 2.0) / 3.0;
  1393→        for i in 0..3 {
  1394→            let qx = panel_x + PAD + i as f32 * (quick_w + GAP);
  1395→            if point_in_rect([qx, quick_y, quick_w, 24.0], point) {
  1396→                return Some(AgentPanelHit::QuickAction(i));
  1397→            }
  1398→        }
  1399→        // Accept/Reject кнопки (если есть preview в последнем сообщении).
  1400→        let log_y = panel_y + HEAD_H + CTX_H;
  1401→        let log_h = panel_h - HEAD_H - CTX_H - EST_H - INPUT_H - QUICK_H - GAP;
  1402→        if self.agent_panel.preview.is_some() {
  1403→            // Грубая проверка — кнопки где-то в нижней половине лога.
  1404→            // FR-LLM-D-TODO: точная геометрия — через макет сообщений.
  1405→            let btn_y = log_y + log_h - PREVIEW_BTN_H - 16.0;
  1406→            let btn_w = (panel_w - PAD * 2.0 - 8.0) / 2.0;
  1407→            let accept_rect = [panel_x + PAD + 16.0, btn_y, btn_w, PREVIEW_BTN_H];
  1408→            let reject_rect = [
  1409→                panel_x + PAD + 16.0 + btn_w + 8.0,
  1410→                btn_y,
  1411→                btn_w,
  1412→                PREVIEW_BTN_H,
  1413→            ];
  1414→            if point_in_rect(accept_rect, point) {
  1415→                return Some(AgentPanelHit::Accept);
  1416→            }
  1417→            if point_in_rect(reject_rect, point) {
  1418→                return Some(AgentPanelHit::Reject);
  1419→            }
  1420→        }
  1421→        // Клик мимо активных элементов (тело панели) — глотаем ввод.
  1422→        Some(AgentPanelHit::NoOp)
  1423→    }

  1425→    /// W2 п.1: диспетчер клика по агент-панели (вызов из обработчика ввода
  1426→    /// ПЕРЕД canvas-pick — панель транзиентна, как ai-status-panel).
  1427→    /// `true` — клик поглощён панелью.
  1428→    pub(super) fn agent_panel_click(&mut self, point: [f32; 2]) -> bool {
  1429→        let Some(hit) = self.agent_panel_hit(point) else {
  1430→            return false;
  1431→        };
  1432→        match hit {
  1433→            AgentPanelHit::Close => {
  1434→                self.agent_panel.open = false;
  1435→                self.request_redraw();
  1436→            }
  1437→            AgentPanelHit::Input => {

  1439→                // (esc_stack верхний owner); символьный ввод панели —

  1441→                self.request_redraw();
  1442→            }
  1443→            AgentPanelHit::Send => {
  1444→                let text = self.agent_panel.input.clone();
  1445→                self.agent_send(&text);

  1447→            AgentPanelHit::QuickAction(i) => {

  1454→            AgentPanelHit::Reject => self.agent_reject_preview(),
  1455→            AgentPanelHit::NoOp => {}
  1456→        }
  1457→        true
  1458→    }

  1461→/// FR-LLM-D: русская плюрализация (как прототип `pluralRu`).
  1462→/// Возвращает один из вариантов по числу: 1 — `one`, 2-4 — `few`,
  1463→/// 5-0/11-14 — `many`. Входные строки пробрасываются по lifetime `'a` —
  1464→/// вызывающий передаёт строковые литералы (`&'static str`), lifetimes
  1465→/// согласованы.
  1466→pub(crate) fn plural_ru<'a>(n: usize, one: &'a str, few: &'a str, many: &'a str) -> &'a str {
  1467→    let n10 = n % 10;
  1468→    let n100 = n % 100;
  1469→    if n10 == 1 && n100 != 11 {
  1470→        one
  1471→    } else if (2..=4).contains(&n10) && !(12..=14).contains(&n100) {
  1472→        few
  1473→    } else {
  1474→        many
  1475→    }
  1476→}

  1478→/// FR-LLM-D: расширение `JsonVal` — доступ к object-pairs (для валидации
  1479→/// tool_calls). `JsonVal` уже имеет `as_object` через pattern-match, но
  1480→/// нам нужен便捷-метод — добавим trait-extension ниже.
  1481→trait JsonValExt {
  1482→    fn as_object(&self) -> Option<&[(String, canvas_llm::JsonVal)]>;
  1483→    fn as_string(&self) -> Option<&str>;
  1484→}

  1489→            canvas_llm::JsonVal::Object(pairs) => Some(pairs),
  1490→            _ => None,
  1491→        }
  1492→    }
  1493→    fn as_string(&self) -> Option<&str> {
  1494→        match self {
  1495→            canvas_llm::JsonVal::String(s) => Some(s),
  1496→            _ => None,
  1497→        }
  1498→    }
  1499→}

  1521→/// W2 п.2: MCP-инструменты (`canvas_mcp::tools_list()`, tools/list JSON)
  1522→/// → `Vec<ToolDef>` для `LlmProvider::tool_calling`. Схема — JsonVal
  1523→/// (from_serde), имена/описания — строки реестра.
  1524→pub(super) fn mcp_tools_as_tooldefs() -> Vec<canvas_llm::ToolDef> {
  1525→    let list = canvas_mcp::tools_list();
  1526→    let tools = list.get("tools").and_then(|v| v.as_array());
  1527→    tools
  1528→        .map(|arr| {
  1529→            arr.iter()
  1530→                .filter_map(|t| {
  1531→                    let name = t.get("name")?.as_str()?.to_string();
  1532→                    let description = t
  1533→                        .get("description")
  1534→                        .and_then(|v| v.as_str())
  1535→                        .unwrap_or_default()
  1536→                        .to_string();
  1537→                    let input_schema = t
  1538→                        .get("inputSchema")
  1539→                        .cloned()
  1540→                        .unwrap_or(serde_json::Value::Object(serde_json::Map::new()));
  1541→                    Some(canvas_llm::ToolDef {
  1542→                        name,
  1543→                        description,
  1544→                        input_schema: canvas_llm::JsonVal::from_serde(&input_schema),
  1545→                    })
  1546→                })
  1547→                .collect()
  1548→        })
  1549→        .unwrap_or_default()
  1550→}
